\chapter{Electromagnetism}

Electromagnetism appears at all length scales.

\section{Electrostatics}

\begin{definition}[Electrostatics]
The physics of \textbf{stationary charges}. Every particle has an antiparticle of opposite charge (electron/positron, proton/antiproton), and charge itself is both \underline{conserved} and \underline{quantized}.
\end{definition}

\subsection{Conservation of Charge}

\begin{theorem}[Conservation of Charge]
If a system is \underline{isolated}, the total charge of that system \underline{does not change}; it is the algebraic sum of $+$ and $-$ charges.
\end{theorem}

Photons carry no charge and so may freely enter or leave the system. Charged particles can be created and destroyed, but only in \underline{pairs} --- $e^+e^-$ pair production being the standard example --- so the algebraic total never shifts.

\textit{Nonconservation of charge would be quite incompatible with the structure of our present electromagnetic theory: the equations of electromagnetism require it, and any violation would break the internal consistency of the entire framework.}

\subsection{Quantization of Charge}

\begin{definition}[Quantization of Charge]
$e$: smallest discrete unit of charge; in one electron ($e \approx 1.6 \times 10^{-19}$ C). Protons carry exactly the same magnitude, $+e$. Throughout we treat charged particles as points, which is harmless because a proton's charge occupies only $\sim 10^{-15}$ m --- far below any scale of interest here.
\end{definition}

\section{Coulomb's Law}

\begin{formula}[Coulomb's Law]
The force on charge $q_1$ due to charge $q_2$ (where $\hat{r}_{21}$ points from source $q_2$ to test $q_1$):
\[
    \vect{F}_{12} = k \frac{q_1 q_2}{r_{21}^2}\, \hat{r}_{21}, \qquad \text{where } k = \frac{1}{4\pi\varepsilon_0}
\]
By Newton's third law, $\vect{F}_1 = -\vect{F}_2$ (assumes both charges are point particles).
\end{formula}

\subsection{Units and the Constant $k$}

The SI unit of charge is the \underline{Coulomb} (C). Two like charges of $1\,$C held $1\,$m apart repel with a force of about $8.99\times 10^9\,$N, which is just to say that $k \approx 8.988 \times 10^9\;\text{N}\cdot\text{m}^2/\text{C}^2$. On that scale the elementary charge is minute, $e \approx 1.602 \times 10^{-19}\,$C, so a single Coulomb amounts to some $6.242 \times 10^{18}$ electrons.

\paragraph{Gaussian System.} In Gaussian units, $k = 1$, so
\[
    \vect{F} = \frac{q_1 q_2}{r^2}\hat{r} \quad\text{(in esu)},
\]
where 1 esu $= \sqrt{\text{dyne}}\cdot\text{cm}$.

\paragraph{SI System.} Alternatively,
\[
    k = \frac{1}{4\pi\varepsilon_0} \implies \varepsilon_0 = \frac{1}{4\pi k} \approx 8.854 \times 10^{-12}\;\frac{\text{C}^2}{\text{N}\cdot\text{m}^2},
\]
so Coulomb's law becomes
\[
    \vect{F} = \frac{1}{4\pi\varepsilon_0}\frac{q_1 q_2}{r_{21}^2}\,\hat{r}_{21}.
\]

\subsection{Significance of Coulomb's Law: Additivity}

Electric charge is \textbf{additive}, but establishing this requires actually measuring charge: place $q_1$, $q_2$, and $q_3$ in series and verify that $q_3 = q_1 + q_2$.

\begin{theorem}[Superposition Principle]
Combining two sets of sources into one system by placing the second system on top of the first. Forces then add linearly --- the whole is the sum of its parts:
\[
    \vect{F}_a = \sum_{i=1}^{N} \vect{F}_i = \sum_{i=1}^{N} k\frac{q_i q_a}{r_{ia}^2}\,\hat{r}_{ia}.
\]
\end{theorem}

\textit{Superposition is not a logical necessity — it must be verified experimentally. Crucially, the force between two charges is unaffected by the presence of a third charge; fields simply add.}

\begin{example}[Coulomb's Law: Three Charges]
Place $q_1$ at the origin, $q_2$ at $(d,0)$, $q_3$ at $(0,d)$. The force on $q_1$ from $q_2$ points along $\hat{r}_{12} = -\ihat$:
\[
    \vect{F}_{12} = -\frac{q_1 q_2}{4\pi\varepsilon_0 d^2}\,\ihat.
\]

\section{Energy in a System of Charges}

Because electrical forces are conservative, the work done in assembling a charge configuration is path-independent and equals the potential energy stored in the system.

\begin{formula}[Work to Assemble Two Charges]
\[
    W = \int_{r=\infty}^{r=r_0} \left(-\frac{1}{4\pi\varepsilon_0}\frac{q_1 q_2}{r^2}\right)dr
    = \frac{1}{4\pi\varepsilon_0}\frac{q_1 q_2}{r_0}.
\]
\end{formula}

If you drop a 3rd charge from infinity, the work done by the system becomes:
\begin{align*}
    W_3 &= -\int_{\infty}^{P_3} \vect{F}_{P_3}\cdot d\lvect{s} \\
        &= -\int_{\infty}^{P_3} \left(\vect{F}_{q_1} + \vect{F}_{q_2}\right)\cdot d\lvect{s} \\
        &= -\int_{\infty}^{P_3} \vect{F}_{q_1}\cdot d\lvect{s} - \int_{\infty}^{P_3} \vect{F}_{q_2}\cdot d\lvect{s} \\
        &= \frac{q_1 q_3}{4\pi\varepsilon_0 r_{13}} + \frac{q_2 q_3}{4\pi\varepsilon_0 r_{23}}.
\end{align*}

Thus total work in constructing the system is:
\begin{align*}
    W &= W_{12} + W_{13} + W_{23} \\
      &= \frac{1}{4\pi\varepsilon_0}\left(\frac{q_1 q_2}{r_{12}} + \frac{q_1 q_3}{r_{13}} + \frac{q_2 q_3}{r_{23}}\right) \equiv U_e,
\end{align*}
the \emph{energy of the system of charges}.

\begin{formula}[Electric Potential Energy of $N$ Point Charges]
\[
    U = \frac{1}{2}\sum_{i=1}^{N}\sum_{\substack{j=1\\j\neq i}}^{N}
    \frac{1}{4\pi\varepsilon_0}\frac{q_i q_j}{r_{ij}}
    = \frac{1}{8\pi\varepsilon_0}\sum_{i=1}^{N}\sum_{\substack{j=1\\j\neq i}}^{N}
    \frac{q_i q_j}{r_{ij}}.
\]
\end{formula}

\subsection{Electrical Energy in a Crystal Lattice}

An ionic crystal such as NaCl consists of alternating positive and negative ions arranged on a regular lattice; the goal is to find the total electrostatic energy, whose sign determines whether the crystal is stable against dissociation into free ions. Because every ion of a given type occupies an equivalent site, the contribution from each ion is the same, so the total energy is simply $N$ times the energy of a single reference ion with all others: $U = \frac{1}{2}N\displaystyle\sum_{j \neq 1} \frac{q_1 q_j}{4\pi\varepsilon_0\, r_{1j}}$.

Using the central atom as reference (nearest-neighbor distance $a$):
\[
    U = \frac{1}{2}N\frac{e^2}{4\pi\varepsilon_0}\left(-\frac{6e^2}{a} + \frac{12e^2}{\sqrt{2}\,a} + \frac{8e^2}{\sqrt{3}\,a} - \frac{6e^2}{2a} + \cdots\right)
\]
(6 closest, 12 next-closest, 8 furthest-before-that, etc.)
\[
    \approx -\frac{0.874\, Ne^2}{4\pi\varepsilon_0\, a} \quad\text{(negative $\Rightarrow$ formation is energetically favorable)}.
\]

\section{Electric Fields}

\begin{definition}[Electric Field]
Suppose we fix charges $q_1,\ldots,q_N$ in place and want to measure the force on a test charge $q_0$:
\[
    \vect{F} = \sum_{i=1}^{N} k\frac{q_i q_0}{r_{i0}^2}\hat{r}_{i0}
    = q_0\,\vect{E} \quad \Longrightarrow \quad \vect{E} \text{ is the \textbf{Electric Field}.}
\]
The electric field encodes all information about the source charges in a single local vector function of position. Once $\vect{E}$ is known, the force on \emph{any} test charge follows immediately from $\vect{F} = q_0\vect{E}$, regardless of how the sources are distributed. The field exists independently of whether a test charge is present.
\end{definition}

\subsection{Visualizing Electric Fields}

Field lines are curves whose tangent at every point gives the direction of $\vect{E}$, with their density representing the field's magnitude; they originate on positive charges and terminate on negative charges (or extend to infinity if no opposite charge is present).

\section{Continuous Charge Distributions}

\begin{definition}[Charge Density]
$\rho = f(x,y,z)$ has units of charge/volume or charge/area.
\[
    \rho\,dV = dQ, \quad dV = dx\,dy\,dz.
\]
\end{definition}

\begin{formula}[Electric Field from a Continuous Distribution]
\[
    \vect{E}(x,y,z) = \frac{1}{4\pi\varepsilon_0}
    \int \frac{\rho(x',y',z')\,\hat{r}\,dV'}{r^2}
    \quad \text{(volume integral)},
\]
where $\hat{r}$ points \emph{from source} $(x',y',z')$ \emph{to test point} $(x,y,z)$ (i.e.\ from source to test charge; $\hat{r}$ is always this direction).
\end{formula}

Using spherical coordinates, the area element is $d\sigma = r^2\sin\theta\,d\theta\,d\phi$, so $dV = r^2\sin\theta\,dr\,d\theta\,d\phi$. Setting $dQ = \rho\,dV$, by symmetry the horizontal components of $\vect{E}$ cancel due to rotational symmetry, giving:
\[
    dE_z = \frac{\rho(2\pi r^2 \sin\theta\,dr\,d\theta)\cos\theta}{4\pi\varepsilon_0\, r^2}.
\]
\[
    E_z = \int_0^{\pi/2}\int \frac{\rho\cdot 2\pi r^2 \sin\theta\cos\theta}{4\pi\varepsilon_0\, r^2}\,dr\,d\theta
        = \frac{\rho}{2\varepsilon_0}\cdot R\cdot \frac{\sin^2\theta}{2}\Bigg|_0^{\pi/2}
        = \frac{\rho R}{4\varepsilon_0}.
\]

\textbf{Note:} Because the volume of a ring grows as $r^2$, this cancels the $\frac{1}{r^2}$ denominator, which removes the singularity.
\end{example}

\end{figure}

Using the ring result:
\[
    \vect{E} = E_x\,\ihat = \frac{\sigma x}{2\varepsilon_0}
    \int_0^R \frac{r\,dr}{(r^2+x^2)^{3/2}}\,\ihat.
\]

Evaluate the integral:
\[
    E_x = \frac{\sigma}{2\varepsilon_0}\left(1 - \frac{1}{\sqrt{1+R^2/x^2}}\right).
\]

\paragraph{Infinite plane limit ($R\to\infty$):}
\[
    E_x \to \frac{\sigma}{2\varepsilon_0}.
\]

\paragraph{Far-field limit ($x \gg R$, finite disk):}
Using $(1+a)^n \approx 1 + na$ for small $a$:
\begin{align*}
    \sqrt{1+R^2/x^2} &\approx 1 + \frac{1}{2}\frac{R^2}{x^2}, \\
    E_x &\approx \frac{\sigma}{2\varepsilon_0}\left(1 - \frac{1}{1 + \frac{R^2}{2x^2}}\right)
          \approx \frac{\sigma}{2\varepsilon_0}\cdot\frac{R^2}{2x^2}
          = \frac{\sigma \pi R^2}{4\pi\varepsilon_0 x^2}
          = \frac{Q}{4\pi\varepsilon_0 x^2}.
\end{align*}
As expected, the finite disk looks like a point charge from far away.
\end{example}

\section{Flux and Gauss's Law}

\subsection{Flux}

\begin{definition}[Electric Flux]
Approximate a smooth, closed 3D shape by a bunch of flat rectangles with normals $\hat{n}$. Let $\vect{E}$ be the electric field at $d\vect{A}$. The flux through a surface element is $\vect{E}\cdot d\vect{A}$.

Generalizing, the flux through a surface $S$ is:
\[
    \Phi = \int_S \vect{E}\cdot d\vect{A} = \int_S \partial A\,(\hat{n}\cdot\vect{E}).
\]
\end{definition}

Flux measures the magnitude of field passing through the surface, and for a closed surface it depends on nothing but the charge inside:
\[
    \oint_S \vect{E}\cdot d\vect{A} = \frac{Q_\text{enc}}{\varepsilon_0}.
\]

\subsection{Derivation of Gauss's Law from Coulomb's Law}

\paragraph{A sphere around a point charge.} Take a single point charge $q$ and a sphere of radius $r$ centred on it. On the sphere $\vect{E} = \frac{q}{4\pi\varepsilon_0 r^2}\,\rhat$ is normal to the surface and constant in magnitude, so
\[
    \Phi = \oint \vect{E}\cdot d\vect{A} = E\cdot 4\pi r^2 = \frac{q}{\varepsilon_0},
\]
independent of $r$.

\paragraph{Any closed surface.} The sphere was convenient but not essential. For an arbitrary surface element, only its projection perpendicular to $\lvect{r}$ contributes to the flux (Fig.~\ref{fig:em-solidangle}):
\[
    \vect{E}\cdot d\vect{A} = \frac{q}{4\pi\varepsilon_0}\,\frac{\cos\theta\; da}{r^2} = \frac{q}{4\pi\varepsilon_0}\,d\Omega ,
\]
since $\cos\theta\,da = r^2\,d\Omega$ by the definition of solid angle. The distance cancels, and this is the one place the inverse-square law enters: for any other power law the flux would depend on the shape of the surface.

Integrating over a closed surface, the total solid angle is $4\pi$ if the charge is inside. If it is outside, every ray that enters the surface also leaves it, and the two crossings cancel:
\[
    \oint_S \vect{E}\cdot d\vect{A} = \begin{cases} q/\varepsilon_0, & q \text{ inside } S,\\[2pt] 0, & q \text{ outside } S.\end{cases}
\]
Flux is linear in $\vect{E}$, so by superposition the result holds for any collection of charges.

\begin{fact}[Gauss's Law]
\[
    \oint_S \vect{E}\cdot d\vect{A} = \frac{Q_\text{enc}}{\varepsilon_0}
    = \frac{1}{\varepsilon_0}\sum_{i\,\in\,S} q_i
    = \frac{1}{\varepsilon_0}\int_V \rho\,dV.
\]
Coulomb's law takes charges to the field; Gauss's law takes the field back to the charges.
\end{fact}

\subsection{Applying Gauss's Law: Only Three Useful Cases}

Gauss's law determines $E$ only when symmetry makes $|\vect{E}|$ constant over a suitable closed surface, so that $\Phi = E\cdot A$. A cube around a point charge fails: $\vect{E}$ varies over its faces and the flux integral does not collapse. Only three geometries are symmetric enough --- spherical, cylindrical, and planar:
\[
    |\vect{E}_\text{sphere}| = \frac{Q}{4\pi\varepsilon_0 r^2}, \qquad
    |\vect{E}_\text{line}| = \frac{\lambda}{2\pi\varepsilon_0 r}, \qquad
    |\vect{E}_\text{sheet}| = \frac{\sigma}{2\varepsilon_0}.
\]

\subsection{Example: Uniformly Charged Ball}
\label{sec:em-uniform-ball}

\begin{example}[Uniformly charged ball]
A ball of radius $R$ carries charge $Q$ with uniform density $\rho = Q/\tfrac43\pi R^3$. Take a Gaussian sphere of radius $r$. By symmetry $\vect{E}=E(r)\,\rhat$, so $E\cdot4\pi r^2 = Q_\text{enc}/\varepsilon_0$.

\textbf{Inside} ($r<R$): the enclosed charge is $Q\,r^3/R^3$, so
\[
    E(r) = \frac{Q\,r}{4\pi\varepsilon_0 R^3} = \frac{\rho\,r}{3\varepsilon_0}.
\]
\textbf{Outside} ($r\ge R$): all the charge is enclosed, and
\[
    E(r) = \frac{Q}{4\pi\varepsilon_0 r^2},
\]
as if the charge were concentrated at the centre. The field grows linearly inside, peaks at $r=R$, and falls as $1/r^2$ outside.

\begin{example}[Ball with a spherical cavity]
A ball of uniform density $\rho$ has a spherical cavity whose centre is displaced by $\lvect{b}$ from the ball's centre. Treat it as a full ball of density $\rho$ plus a small ball of density $-\rho$ filling the cavity. At a point inside the cavity, at position $\lvect{r}$ from the big centre and $\lvect{r}\,'=\lvect{r}-\lvect{b}$ from the cavity centre, the inside-ball result gives
\[
    \vect{E} = \frac{\rho}{3\varepsilon_0}\,\lvect{r} - \frac{\rho}{3\varepsilon_0}\,\lvect{r}\,' = \frac{\rho}{3\varepsilon_0}\,\lvect{b}.
\]
The field in the cavity is \emph{uniform}, pointing along the offset $\lvect{b}$, whatever the sizes of the ball and cavity.
\end{example}

\subsection{Example: Infinite Line Charge}

\begin{example}[Infinite line charge]
A line carries charge $\lambda$ per unit length. By symmetry $\vect{E}$ is radial.

\textbf{Directly from Coulomb's law:} with $dq=\lambda\,dx$ at distance $R$ from the field point and $\theta$ the angle from the perpendicular,
\[
    E = \int_{-\infty}^{\infty} \frac{\lambda\cos\theta\,dx}{4\pi\varepsilon_0\, R^2}
      = \int_{-\pi/2}^{\pi/2} \frac{\lambda\cos\theta}{4\pi\varepsilon_0\, r}\,d\theta
      = \frac{\lambda}{2\pi\varepsilon_0\, r},
\]
using $R = r/\cos\theta$ and $dx = r\,d\theta/\cos^2\theta$.

\textbf{From Gauss's law:} a coaxial cylinder of radius $r$ and length $L$ has flux $2\pi rL\,E$ through its curved side and none through its ends, so $2\pi rL\,E = \lambda L/\varepsilon_0$, giving the same result in one line.

\section{Force on a Layer of Charge}

Take a hollow sphere with surface charge density $\sigma$. Gauss's law gives $E=0$ just inside and $E=\sigma/\varepsilon_0$ just outside. What force do the charges of the layer feel?

A patch of the layer cannot push on itself, so it feels only the field of all the \emph{other} charges. The patch's own field is $\pm\sigma/2\varepsilon_0$ on its two sides, so the field of the rest is the average of the fields on the two sides:
\[
    d\vect{F} = \tfrac12\bigl(E_\text{inside}+E_\text{outside}\bigr)\,\sigma\,dA\;\uvect{n}
    = \frac{\sigma^2}{2\varepsilon_0}\,dA\;\uvect{n}.
\]
The same holds for a layer of finite thickness: with $dE/dx = \rho/\varepsilon_0$ across it,
\[
    \frac{F}{A} = \int E\rho\,dx = \varepsilon_0\int_{E_1}^{E_2}E\,dE = \frac{\varepsilon_0}{2}\bigl(E_2^2-E_1^2\bigr)
    = \tfrac12(E_1+E_2)\,\sigma ,
\]
using $E_2-E_1=\sigma/\varepsilon_0$.

\begin{formula}[Force per unit area on a charge layer]
\[
    \frac{F}{A} = \tfrac12\bigl(E_1+E_2\bigr)\,\sigma = \frac{\varepsilon_0}{2}\bigl(E_2^2-E_1^2\bigr),
\]
directed outward whatever the sign of $\sigma$.
\end{formula}

\section{Energy Density of the Electric Field}

The energy of an electrostatic system can be assigned to the field itself, with an energy $\frac{\varepsilon_0}{2}E^2$ per unit volume. This is more than bookkeeping: once fields vary in time, this energy can propagate as radiation.

To see where the formula comes from, squeeze a charged spherical shell inward by $dr$. The outward pressure is $\sigma^2/2\varepsilon_0$ from the previous section, so the work done is
\[
    dW = \frac{\sigma^2}{2\varepsilon_0}\,4\pi r^2\,dr = \frac{\varepsilon_0}{2}\Bigl(\frac{\sigma}{\varepsilon_0}\Bigr)^2 dV,
\]
where $dV = 4\pi r^2dr$ is the thin shell of space which previously had no field and now has field $E=\sigma/\varepsilon_0$.

\begin{formula}[Energy density of the electric field]
\[
    u_E = \frac{\varepsilon_0}{2}E^2, \qquad U = \frac{\varepsilon_0}{2}\int E^2\,dV .
\]
\end{formula}

\subsection{Example: Self-Energy of a Shell and of a Ball}

\begin{example}[Self-energy of a spherical shell and of a uniform ball]
\textbf{Shell.} For a shell of radius $R$ and charge $Q$, $E=0$ inside and $E = Q/4\pi\varepsilon_0 r^2$ outside, so all the energy lies outside:
\[
    U_\text{shell} = \frac{\varepsilon_0}{2}\int_R^{\infty}\left(\frac{Q}{4\pi\varepsilon_0 r^2}\right)^2 4\pi r^2\,dr
                 = \frac{Q^2}{8\pi\varepsilon_0 R}.
\]
\textbf{Uniform ball.} Outside, the field is the same as for the shell, contributing the same $Q^2/8\pi\varepsilon_0 R$. Inside, $E = Qr/4\pi\varepsilon_0R^3$, so
\[
    U_\text{int} = \frac{\varepsilon_0}{2}\int_0^R \left(\frac{Qr}{4\pi\varepsilon_0 R^3}\right)^2 4\pi r^2\,dr
                 = \frac{Q^2}{8\pi\varepsilon_0 R^6}\int_0^R r^4\,dr
                 = \frac{1}{5}\,\frac{Q^2}{8\pi\varepsilon_0 R}.
\]
The same result follows from assembling the ball shell by shell: bringing a shell of charge $dq = \rho\,4\pi r^2dr$ from infinity to the surface of a ball of radius $r$ and charge $q(r)=Qr^3/R^3$ costs $dW = q(r)\,dq/4\pi\varepsilon_0 r$, so
\[
    W = \int_0^R \frac{1}{4\pi\varepsilon_0 r}\,\frac{Qr^3}{R^3}\,\frac{3Q r^2}{R^3}\,dr = \frac{3Q^2}{4\pi\varepsilon_0R^6}\int_0^R r^4\,dr = \frac{3}{5}\,\frac{Q^2}{4\pi\varepsilon_0 R}.
\]
\begin{formula}[Self-energies]
\[
    U_\text{shell} = \frac{Q^2}{8\pi\varepsilon_0 R},
    \qquad
    U_\text{ball} = \frac{6}{5}\,\frac{Q^2}{8\pi\varepsilon_0 R} = \frac{3}{5}\,\frac{Q^2}{4\pi\varepsilon_0 R}.
\]
\end{formula}
\end{example}

\section{Chapter Summary}

\subsection{Forces Between Two Charges}
\[
    \vect{F} = \frac{1}{4\pi\varepsilon_0}\frac{q_1 q_2}{r_{21}^2}\,\hat{r}_{21}.
\]

\subsection{Potential Energy of a System of Charges}
\[
    U = \frac{1}{2}\sum_{j=1}^{N}\sum_{\substack{k=1\\k\neq j}}^{N}
    \frac{1}{4\pi\varepsilon_0}\frac{q_j q_k}{r_{jk}}.
\]

\subsection{Electric Field Due to a Charge Distribution}
\[
    \vect{E} = \frac{1}{4\pi\varepsilon_0}\int \frac{\rho(x',y',z')\,\hat{r}\,dx'\,dy'\,dz'}{r^2}
    \quad \text{or} \quad
    \frac{1}{4\pi\varepsilon_0}\sum_{j=1}^{N}\frac{q_j\,\hat{r}_j}{r_j^2}.
\]
\[
    \vect{F} = q\vect{E} \quad \text{(force on a test charge)}.
\]

\subsection{Flux of an Electric Field through Surface $S$}
\[
    \Phi = \int_S \vect{E}\cdot d\vect{A}.
\]

\begin{fact}[Gauss's Law (Summary)]
Flux of an $\vect{E}$ through any closed surface:
\[
    \oint \vect{E}\cdot d\vect{A} = \frac{1}{\varepsilon_0}\int \rho\,dV = \frac{1}{\varepsilon_0}\sum q_i = \frac{Q_\text{enc}}{\varepsilon_0}.
\]
Pre-determined shapes:
\[
    E_\text{sphere} = \frac{Q}{4\pi\varepsilon_0 r^2}, \quad
    E_\text{line} = \frac{\lambda}{2\pi\varepsilon_0 r}, \quad
    E_\text{sheet} = \frac{\sigma}{2\varepsilon_0}.
\]
\end{fact}

The force per unit area on a layer of charge follows from averaging the field across it,
\[
    \frac{F}{A} = \frac{1}{2}(E_\text{inside} + E_\text{outside})\,\sigma,
\]
and since an electric field carries energy density $\varepsilon_0 E^2/2$, integrating that density
\[
    U = \frac{\varepsilon_0}{2}\int E^2\,dV
\]
gives the total work required to assemble the charge distribution in the first place.

\subsection{Handy Trig Substitutions}

For integrals of the form $\dfrac{1}{(x^2+a^2)^n}$, use $x = a\tan(u)$:
\[
    \implies dx = a\sec^2(u)\,du.
\]

\textbf{Strategy:} Find the field due to the entire plane, then subtract the field from the disk of radius $R$.

By Gauss's Law for the entire plane:
\[
    2(\vect{E}\cdot\pi R^2) = \frac{\sigma\pi R^2}{\varepsilon_0}
    \implies \vect{E} = \frac{\sigma}{2\varepsilon_0}.
\]

Minus the disk: $dq = \sigma\cdot 2\pi r\,dr$.
\[
    \text{thus } dE = \frac{1}{4\pi\varepsilon_0}\frac{2\pi r\,\sigma\,dr\,z}{(z^2+r^2)^{3/2}}.
\]
\[
    E_\text{disk} = \frac{\sigma z}{2\varepsilon_0}\int_0^R\frac{r\,dr}{(z^2+r^2)^{3/2}}
                  = \frac{\sigma z}{2\varepsilon_0}\cdot\frac{-1}{(z^2+r^2)^{1/2}}\Bigg|_0^R
                  = \frac{\sigma z}{2\varepsilon_0}\left(\frac{-1}{(R^2+z^2)^{1/2}} + \frac{1}{z}\right).
\]
\[
    \text{Thus} \quad \boxed{E_\text{hole} = \frac{\sigma z}{2\varepsilon_0(z^2+R^2)^{1/2}}}.
\]
\end{example}
