\section{Energy Stored in a Capacitor}
%%─────────────────────────────────────────────

\begin{itemize}
  \item Suppose we have a capacitor of capacitance $C$, potential difference $\Delta\phi$, charge $Q$,
        and $Q = C\,\Delta\phi$.
  \item $+Q$, $-Q$ on plates.
  \item Work required to move a charge $Q$ from one plate to the other:
\end{itemize}

\[
  W = \frac{1}{C}\int_0^{Q_f} Q\,dQ = \frac{Q_f^2}{2C} \;\defeq\; U_C
  \qquad \text{-- Energy stored in a Capacitor}
\]

\begin{formula}[Energy stored in a capacitor]
\[
  U_C = \frac{Q^2}{2C} = \frac{Q\Phi}{2} = \frac{1}{2}C\Phi^2
\]
\end{formula}

\noindent For a parallel-plate capacitor with area $A$ and separation $s$:

\begin{figure}[H]
  \centering
  \caption{Parallel-plate capacitor with plate area $A$ and separation $s$.}
\end{figure}

\[
  C = \frac{\varepsilon_0 A}{s}
  \implies
  U_C = \frac{1}{2}\!\left(\frac{\varepsilon_0 A}{s}\right)\!(E\,s)^2
       = \frac{\varepsilon_0 E^2}{2}\cdot \text{Volume}
\]
where $E = \Phi/s$.

\medskip
\noindent This equation also applies to an \emph{isolated charged conductor}. For an isolated sphere:
\[
  C = 4\pi\varepsilon_0 a, \qquad
  U_C = \frac{1}{2}Cq^2 \;=\; \frac{1}{2}(4\pi\varepsilon_0 a)\Phi^2
      = \frac{1}{2}\cdot\frac{q^2}{4\pi\varepsilon_0 a}
\]
which agrees with the earlier calculation for energy stored in the electric field of a charged sphere.

\section{Force Required to Keep Capacitor Plates Apart}
%%─────────────────────────────────────────────

\begin{itemize}
  \item Suppose we have a general capacitor whose capacitance $C$ depends on some linear coordinate
        that measures the distance between the two plates, $x$.
  \item Let $F$ be the magnitude of the force required to keep the conductors apart.
  \item Let $x$ be increased by an increment $\Delta x$; $Q$ remains constant, one plate is fixed.
  \item $F_\text{ext}$ does work $F\,\Delta x = \Delta U = \dfrac{\partial U}{\partial x}\,\Delta x$:
\end{itemize}

\begin{align*}
  F_\text{ext}\,\Delta x &= \Delta U = \frac{Q^2}{2}\,\frac{\partial}{\partial x}\!\left(\frac{1}{C}\right)\Delta x \\[4pt]
  \implies\quad F_\text{ext} &= \frac{Q^2}{2}\,\frac{d}{dx}\!\left(\frac{1}{C}\right)
\end{align*}

\begin{itemize}
  \item \textbf{Conductivity} (not surface charge) $\to$ \textbf{$\sigma$} (Ohm's law):
\end{itemize}

\begin{formula}[Ohm's Law (microscopic)]
\[
  \vect{J} = \sigma\,\vect{E}
\]
\end{formula}

\noindent where $\vect{J}$ is current density (Amps/m$^2$) and the units are
$[\sigma] = \tfrac{\text{Amp}}{\text{m}^2}\big/\tfrac{\text{V}}{\text{m}}$.

\[
  I = \vect{J}\cdot\hat{A} = S\,\vect{E}\cdot\hat{A}
  \qquad \text{(Amps = Coulombs/sec)}
\]
\[
  I = \int \vect{J}\cdot d\vect{A} \quad\text{-- mean/average vector}
\]

\begin{itemize}
  \item Charges have defects/blockages causing them to have a finite speed (drift velocity):
\end{itemize}

\begin{figure}[H]
  \centering
  \caption{Charge carrier box with drift velocity $\vect{v} = V/s$ (voltage over length).}
\end{figure}

\[
  I = \vect{J}\cdot A = S\,\vect{E}\cdot\hat{A} = \left(\frac{\sigma A}{s}\right)V
\]

\noindent From $I = GV$, $V = \tfrac{1}{G} \cdot I$, so $G = \tfrac{\sigma A}{s}$;
resistance $R = \tfrac{1}{G} = \tfrac{s}{\sigma A}$.
\[
  I = \frac{V}{R}, \qquad [R] \;=\; \left(\frac{6.37}{\text{Amp}}\right)\cdot\frac{1\;\text{Volt}}{1\;\text{Amp}}
\]
\[
  \rho = \frac{1}{\sigma} = \text{resistivity}; \quad R = \frac{\rho\,\ell}{A}
\]

\section{Conductors in Equilibrium --- Uniqueness Theorem}
%%─────────────────────────────────────────────

\noindent Inside a perfect conductor $E = 0$, so $\Phi =$ constant inside the boundary; it must also
be constant inside the boundary:
\[
  \frac{\partial^2\phi}{\partial x^2} + \frac{\partial^2\phi}{\partial y^2} + \frac{\partial^2\phi}{\partial z^2} = 0
\]

\begin{theorem}[Uniqueness Theorem]
$\nabla^2\Phi = -\dfrac{\rho}{\varepsilon_0}$. Suppose there are two solutions that satisfy:
\[
  \nabla^2\phi_1 = -\frac{\rho}{\varepsilon_0}, \qquad \nabla^2\phi_2 = -\frac{\rho}{\varepsilon_0}
\]
with $\phi_1(\vec{r}) = \phi_2(\vec{r})$ on the boundaries. Then $\nabla^2(\phi_1 - \phi_2) = 0$
everywhere. Let $\phi_3 = \phi_1 - \phi_2$; then
\[
  \nabla^2\phi_3 = 0 \implies \phi_3 = 0 \implies \phi_2 = \phi_1
\]
\textbf{``Uniqueness theorem.''}
\end{theorem}

\begin{fact}[Corollary]
If we know $\rho(\vec{r})$ inside a boundary and we know the perpendicular electric field at the
boundary, we know the potential inside the boundary to within a constant.
\end{fact}

\noindent Need: $\nabla\cdot(\phi_3\,\nabla\phi_3) = (\nabla\phi_3)^2 + \phi_3\,\nabla^2\phi_3$.

\noindent Noting that $\nabla^2\phi_3 = 0$ (with $\phi_3 = \phi_2 - \phi_1$)\ldots \ \emph{nevermind.}

\subsubsection*{Case (i): Charge above a Grounded Plane}

Setup: charge $Q$ at $(0,0,d)$, image charge $-Q$ at $(0,0,-d)$, observation point $\vec{r}=(x,y,z)$, surface at $z=0$.

\[
  \vect{E} = \frac{Q}{4\pi\varepsilon_0}\,\frac{x\,\hat{x} + y\,\hat{y} + (z-d)\,\hat{z}}{\bigl(x^2+y^2+(z-d)^2\bigr)^{3/2}}
  + \frac{-Q}{4\pi\varepsilon_0}\,\frac{x\,\hat{x} + y\,\hat{y} + (z+d)\,\hat{z}}{\bigl(x^2+y^2+(z+d)^2\bigr)^{3/2}}
\]

\noindent At $z=0$:
\[
  \vect{E}(x,y,0) = \frac{Q}{4\pi\varepsilon_0}\cdot\frac{-2d\,\hat{z}}{(x^2+y^2+d^2)^{3/2}}
                  = \frac{-Qd\,\hat{z}}{2\pi\varepsilon_0(x^2+y^2+d^2)^{3/2}}
\]

\noindent In cylindrical coordinates ($r,\theta,\phi$):
\[
  E(r,\theta) = \frac{-Qd\,\hat{z}}{2\pi\varepsilon_0(r^2+d^2)^{3/2}}
\]

\noindent From the boundary condition $E = \sigma/\varepsilon_0$:
\[
  \sigma = \varepsilon_0\,\vect{E}\cdot\hat{z} = \frac{-Qd}{2\pi(r^2+d^2)^{3/2}}
\]

\noindent To find total charge:
\[
  \int_0^\infty \sigma\cdot 2\pi r\,dr = \int_0^\infty \frac{-Qd}{2\pi(r^2+d^2)}\cdot 2\pi r\,dr
  + Qd\cdot\frac{1}{(r^2-d^2)^{1/2}}\bigg|
\]
(The total induced charge is $-Q$; it is repelled to $-z$ infinity.)

\section{Capacitance (Recitation 5)}
%%─────────────────────────────────────────────

\begin{figure}[H]
  \centering
  \caption{Two conductors at potentials $\phi_1$ and $\phi_1'$ each carrying charge $+1$. $\phi$ is NOT the work the conductor does on the charge; it is the work the charge does on the conductor.}
\end{figure}

\begin{definition}[Capacitor]
A capacitor consists of two conductors with potentials $\phi_1$ and $\phi_2$, charges $+q$ and $-q$:
\[
  Q = C\bigl(|\phi_1 - \phi_2|\bigr) \quad \text{only valid if you move } Q
  \text{ from one previously neutral conductor}
\]
\[
  Q \approx C\Phi = C|\Delta\phi|
\]
\end{definition}

\begin{example}[Coaxial Cylindrical Capacitor]
Inner radius $a$ (charge $+Q$), outer radius $b$ (charge $-Q$), length $L\to\infty$;
linear charge density $\lambda = Q/L$.

By Gauss's law:
\[
  |\vect{E}|\cdot 2\pi r L = \frac{Q}{\varepsilon_0}
  \implies |\vect{E}| = \frac{Q}{2\pi r L\varepsilon_0} = \frac{\lambda}{2\pi r\varepsilon_0}
\]
\[
  |\Delta\phi| = \int_a^b \vect{E}\,dr = \frac{\lambda}{2\pi\varepsilon_0}\ln\!\left(\frac{b}{a}\right)
\]
Capacitance per unit length:
\[
  \frac{C}{L} \equiv \frac{\text{capacitance}}{\text{unit length}} = \frac{Q}{L\,|\Delta\phi|}
            = \frac{\lambda\cdot 2\pi\varepsilon_0}{\lambda\ln(b/a)} = \frac{2\pi\varepsilon_0}{\ln(b/a)}
\]
\end{example}

\subsection{Energy Stored per Unit Length (Coaxial Capacitor)}
%%─────────────────────────────────────────────

\[
  U = \frac{Q^2}{2C}, \qquad Q = \lambda L
\]
\[
  \frac{U}{L} = \frac{\ln(b/a)\,\lambda^2}{4\pi\varepsilon_0}, \qquad
  \frac{U}{L} = \frac{1}{LC}\cdot\frac{Q^2}{2} = \frac{1}{LC}\cdot\frac{\lambda^2 L^2}{2} = \frac{L\lambda^2}{2C} = \frac{L\lambda^2}{2C}
\]

\begin{figure}[H]
