% =====================================================================
\chapter{Multipoles and Dielectrics}
% =====================================================================

The dipole of Section~\ref{sec:em-dipole} is the first correction to the field of a neutral pair of charges. This chapter makes that systematic --- the multipole expansion of an arbitrary localised charge distribution --- and then uses the dipole term to describe insulating matter, whose response to a field is to polarise.

% =====================================================================
\section{The Multipole Expansion}
\label{sec:em-multipole}
% =====================================================================

For a localised charge distribution $\rho(\lvect{r}')$ the potential is
\[
    \phi(\lvect{r}) = \frac{1}{4\pi\varepsilon_0}\int\frac{\rho(\lvect{r}')}{|\lvect{r}-\lvect{r}'|}\,d^3r' .
\]
Let $\alpha$ be the angle between $\lvect{r}$ and $\lvect{r}'$. By the law of cosines,
\[
    |\lvect{r}-\lvect{r}'| = r\sqrt{1+\epsilon},
    \qquad
    \epsilon = \frac{r'}{r}\left(\frac{r'}{r}-2\cos\alpha\right).
\]
Far outside the distribution $r'\ll r$, so $\epsilon\ll1$ and the binomial series applies:
\[
    \frac{1}{|\lvect{r}-\lvect{r}'|} = \frac1r\left(1-\tfrac12\epsilon+\tfrac38\epsilon^2-\tfrac{5}{16}\epsilon^3+\cdots\right).
\]
Collecting powers of $r'/r$, the coefficients turn out to be the Legendre polynomials:

\begin{formula}[Legendre generating function]
\[
    \frac{1}{|\lvect{r}-\lvect{r}'|} = \frac1r\sum_{n=0}^{\infty}\left(\frac{r'}{r}\right)^{n} P_n(\cos\alpha),
    \qquad r'<r,
\]
with $P_0(x)=1$, $P_1(x)=x$, $P_2(x)=\tfrac12(3x^2-1)$, $P_3(x)=\tfrac12(5x^3-3x)$.
\end{formula}

\begin{formula}[Multipole expansion]
\[
    \phi(\lvect{r}) = \frac{1}{4\pi\varepsilon_0}\sum_{n=0}^{\infty}\frac{1}{r^{n+1}}\int (r')^{n}P_n(\cos\alpha)\,\rho(\lvect{r}')\,d^3r' .
\]
\end{formula}

The $n=0$ (monopole) term is $\phi_\text{mon} = Q/4\pi\varepsilon_0 r$ with $Q=\int\rho\,d^3r'$, and dominates at large $r$ unless $Q=0$. The next, $n=1$, term uses $r'\cos\alpha = \rhat\cdot\lvect{r}'$:
\[
    \phi_\text{dip}(\lvect{r}) = \frac{1}{4\pi\varepsilon_0}\frac{1}{r^2}\,\rhat\cdot\int\lvect{r}'\rho(\lvect{r}')\,d^3r' .
\]

\begin{definition}[Dipole moment]
\[
    \lvect{p} = \int\lvect{r}'\rho(\lvect{r}')\,d^3r',
    \qquad
    \phi_\text{dip}(\lvect{r}) = \frac{1}{4\pi\varepsilon_0}\frac{\lvect{p}\cdot\rhat}{r^2}.
\]
For point charges $\lvect{p}=\sum_i q_i\lvect{r}_i$; for a pair $\pm q$ separated by $\lvect{d}$ (from $-q$ to $+q$), $\lvect{p}=q\lvect{d}$, recovering Section~\ref{sec:em-dipole}.
\end{definition}

\begin{fact}[Origin dependence]
Shifting the origin by $\lvect{a}$ changes the dipole moment to $\bar{\lvect{p}} = \int(\lvect{r}'-\lvect{a})\rho\,d^3r' = \lvect{p}-Q\lvect{a}$. The dipole moment of a \emph{neutral} distribution is therefore independent of the origin.
\end{fact}

\subsection{The Pure Dipole}

A \emph{pure} (point) dipole is one whose potential is exactly $\phi_\text{dip}$, with no higher multipoles: the limit $q\to\infty$, $d\to0$ of a physical dipole at fixed $p=qd$. Its field was computed in Section~\ref{sec:em-dipole}, $\vect{E} = \frac{p}{4\pi\varepsilon_0 r^3}(2\cos\theta\,\rhat+\sin\theta\,\tthat)$. Writing $\lvect{p} = p\cos\theta\,\rhat - p\sin\theta\,\tthat$, one has $3(\lvect{p}\cdot\rhat)\rhat-\lvect{p} = 2p\cos\theta\,\rhat+p\sin\theta\,\tthat$, so

\begin{formula}[Dipole field, coordinate-free]
\[
    \vect{E}_\text{dip}(\lvect{r}) = \frac{1}{4\pi\varepsilon_0 r^3}\bigl[3(\lvect{p}\cdot\rhat)\rhat-\lvect{p}\bigr].
\]
\end{formula}

\section{Dielectrics and Polarisation}
% =====================================================================

A conductor has an unlimited supply of free charge. In an \emph{insulator} every charge is bound to an atom or molecule and can only shift slightly within it; an insulator whose charges shift to form induced dipoles is a \emph{dielectric}.

\subsection{Atomic Polarisability}

\begin{example}[Toy model of an atom]
Model an atom as a point nucleus $+q$ inside a uniformly charged electron cloud of charge $-q$ and radius $a$. An external field $\vect{E}$ displaces the nucleus by $d$ relative to the cloud's centre, where the cloud's field (Gauss's law inside a uniform ball) is
\[
    E_\text{cloud} = \frac{1}{4\pi\varepsilon_0}\frac{qd}{a^3}.
\]
Equilibrium requires $E_\text{cloud}=E$, so $d = 4\pi\varepsilon_0 a^3E/q$ and the induced moment is
\[
    p = qd = 4\pi\varepsilon_0a^3\,E .
\]
\end{example}

\begin{definition}[Atomic polarisability]
For a weak field the induced moment is linear, $\lvect{p}=\alpha\vect{E}$; $\alpha$ is the \emph{atomic polarisability}. The toy model gives $\alpha=4\pi\varepsilon_0a^3$.
\end{definition}

For hydrogen, with $a$ the Bohr radius $5.3\times10^{-11}$\,m, the toy model predicts $\alpha/4\pi\varepsilon_0\approx1.5\times10^{-31}\,\text{m}^3$ against the measured $6.7\times10^{-31}\,\text{m}^3$ --- right to within a factor of about four, which is good for so crude a model.

\begin{definition}[Polarisation]
In a field, a dielectric fills with aligned induced dipoles and is said to be \emph{polarised}. The \emph{polarisation} $\vect{P}(\lvect{r})$ is the dipole moment per unit volume.
\end{definition}

\subsection{Bound Charges}

Summing the dipole potential over the volume elements $\vect{P}\,d^3r'$ of a polarised object,
\[
    \phi(\lvect{r}) = \frac{1}{4\pi\varepsilon_0}\int_V\frac{\vect{P}(\lvect{r}')\cdot(\lvect{r}-\lvect{r}')}{|\lvect{r}-\lvect{r}'|^3}\,d^3r'
    = \frac{1}{4\pi\varepsilon_0}\int_V\vect{P}\cdot\grad'\!\left(\frac{1}{|\lvect{r}-\lvect{r}'|}\right)d^3r' .
\]
The product rule $\grad'\cdot(\vect{P}/|\lvect{r}-\lvect{r}'|) = \vect{P}\cdot\grad'(1/|\lvect{r}-\lvect{r}'|) + (\grad'\cdot\vect{P})/|\lvect{r}-\lvect{r}'|$ and the divergence theorem turn this into
\[
    \phi(\lvect{r}) = \frac{1}{4\pi\varepsilon_0}\oint_S\frac{\vect{P}\cdot\uvect{n}'}{|\lvect{r}-\lvect{r}'|}\,da'
    + \frac{1}{4\pi\varepsilon_0}\int_V\frac{-\grad'\cdot\vect{P}}{|\lvect{r}-\lvect{r}'|}\,d^3r' .
\]

\begin{theorem}[Bound charges]
The potential of a polarised object is that of a surface charge and a volume charge,
\[
    \sigma_b = \vect{P}\cdot\uvect{n},
    \qquad
    \rho_b = -\div\vect{P}.
\]
\end{theorem}

These are genuine accumulations of charge: $\sigma_b$ is the uncompensated end of each column of dipoles at the surface, and $\rho_b$ appears where the polarisation is non-uniform so that neighbouring dipoles fail to cancel. In practice one computes the bound charges, forgets the material, and solves an ordinary electrostatics problem.

\subsection{Example: The Uniformly Polarised Ball}
\label{sec:em-polarised-ball}

A ball of radius $R$ carries uniform polarisation $\vect{P}=P\khat$. Then $\rho_b=0$ and $\sigma_b = P\cos\theta$. Rather than integrate Coulomb's law, solve Laplace's equation inside and outside and match at the surface.

Using the axisymmetric solution of Section~\ref{sec:em-separation}, with $A_\ell r^\ell$ inside and $B_\ell r^{-(\ell+1)}$ outside, the matching formula there with $\sigma_0=\sigma_b=P\cos\theta=P\,P_1(\cos\theta)$ leaves only $\ell=1$: $A_1=P/3\varepsilon_0$ and $B_1=A_1R^3=PR^3/3\varepsilon_0$.

\begin{formula}[Uniformly polarised ball]
\[
    \phi(r,\theta) =
    \begin{cases}
        \dfrac{P}{3\varepsilon_0}\,r\cos\theta, & r\le R,\\[3mm]
        \dfrac{PR^3}{3\varepsilon_0}\dfrac{\cos\theta}{r^2}, & r\ge R.
    \end{cases}
\]
Inside, $r\cos\theta=z$ and the field is \emph{uniform}, $\vect{E}=-\vect{P}/3\varepsilon_0$. Outside, the potential is exactly that of a pure dipole at the centre with the ball's total moment $\lvect{p}=\tfrac43\pi R^3\vect{P}$.
\end{formula}

\section{The Electric Displacement}
\label{sec:em-displacement}

\subsection{Gauss's Law in Matter}

Inside a dielectric the total charge density is $\rho = \rho_b + \rho_f$, where $\rho_b = -\div\vect{P}$ is the bound charge of the previous section and the \emph{free} charge $\rho_f$ is everything else: the charge we control. Gauss's law becomes
\[
    \varepsilon_0\div\vect{E} = -\div\vect{P} + \rho_f
    \qquad\Longrightarrow\qquad
    \div\bigl(\varepsilon_0\vect{E} + \vect{P}\bigr) = \rho_f .
\]

\begin{definition}[Electric displacement]
\[
    \vect{D} \defeq \varepsilon_0\vect{E} + \vect{P},
    \qquad
    \div\vect{D} = \rho_f,
    \qquad
    \oint_S\vect{D}\cdot d\lvect{a} = Q_{f,\text{enc}} .
\]
Only the free charge appears.
\end{definition}

\subsection{\texorpdfstring{$\vect{D}$}{D} Is Not the Field of the Free Charge}

The curl of $\vect{D}$ need not vanish:
\[
    \curl\vect{D} = \varepsilon_0\curl\vect{E} + \curl\vect{P} = \curl\vect{P}.
\]
Consequently there is no potential with $\vect{D} = -\grad\phi'$, no Poisson equation $\laplacian\phi' = -\rho_f$, and no Coulomb law:
\[
    \vect{D}(\lvect{r}) \neq \frac{1}{4\pi}\int\frac{\lvect{r}-\lvect{r}'}{|\lvect{r}-\lvect{r}'|^3}\,\rho_f(\lvect{r}')\,d^3r' .
\]
The right-hand side is curl-free, which contradicts $\curl\vect{D} = \curl\vect{P}\neq0$ in general. By the Helmholtz theorem a field needs both its divergence and its curl to be determined, and knowing $\rho_f$ supplies only the first.

\begin{fact}[When Gauss's law for $\vect{D}$ is useful]
Gauss's law is one scalar equation. It fixes $\vect{D}$ only when symmetry (spherical, cylindrical or planar) already fixes its direction and dependence. Then
\[
    \oint\vect{D}\cdot d\lvect{a} = Q_{f,\text{enc}}\;\Rightarrow\;\vect{D},
    \qquad
    \vect{E} = \frac{\vect{D}-\vect{P}}{\varepsilon_0}\quad\Bigl(=\frac{\vect{D}}{\varepsilon}\ \text{if linear}\Bigr).
\]
Without such symmetry, solve for $\phi$ with boundary conditions instead.
\end{fact}

\subsection{Boundary Conditions}

At an interface with normal $\uvect{n}$ pointing from ``below'' to ``above'', a pillbox gives the normal components from the divergences and a thin loop gives the tangential components from the curls.

\begin{formula}[Boundary conditions in matter]
\begin{align*}
    D^\perp_{\text{above}} - D^\perp_{\text{below}} &= \sigma_f, &
    E^\perp_{\text{above}} - E^\perp_{\text{below}} &= \frac{\sigma_f+\sigma_b}{\varepsilon_0},\\
    \vect{D}^\parallel_{\text{above}} - \vect{D}^\parallel_{\text{below}} &= \vect{P}^\parallel_{\text{above}} - \vect{P}^\parallel_{\text{below}}, &
    \vect{E}^\parallel_{\text{above}} - \vect{E}^\parallel_{\text{below}} &= 0.
\end{align*}
In terms of the potential, which is continuous across the interface, in linear media
\[
    \varepsilon_{\text{above}}\pdv{\phi_{\text{above}}}{n} - \varepsilon_{\text{below}}\pdv{\phi_{\text{below}}}{n} = -\sigma_f .
\]
\end{formula}

\subsection{Linear Dielectrics}

\begin{definition}[Linear dielectric]
For fields that are not too strong, most materials respond linearly:
\[
    \vect{P} = \varepsilon_0\chi_e\vect{E},
    \qquad
    \vect{D} = \varepsilon\vect{E},
    \qquad
    \varepsilon = \varepsilon_0(1+\chi_e),
    \qquad
    \varepsilon_r = 1+\chi_e = \frac{\varepsilon}{\varepsilon_0}.
\]
$\chi_e$ is the \emph{electric susceptibility}, $\varepsilon$ the \emph{permittivity} and $\varepsilon_r$ the \emph{dielectric constant}. For hydrogen gas $\varepsilon_r\approx1.00025$; for water $\varepsilon_r\approx80.1$.
\end{definition}

Even in linear media, $\curl\vect{D} = 0$ only \emph{inside} each homogeneous region. A loop straddling two regions with different $\varepsilon$ has $\oint\vect{E}\cdot d\lvect{l} = 0$, but in general $\oint\vect{D}\cdot d\lvect{l}\neq0$, because the ratio $D/E = \varepsilon$ jumps.

\begin{fact}[All space filled with one dielectric]
If a single homogeneous linear dielectric fills all space, then $\curl\vect{D} = 0$ everywhere, so $\vect{D}$ is $\varepsilon_0$ times the vacuum field of the free charge. Every vacuum result then holds with $\varepsilon_0\to\varepsilon$:
\[
    \vect{E} = \frac{1}{4\pi\varepsilon}\frac{q}{r^2}\rhat,
    \qquad
    C = \varepsilon_r\,C_{\text{vac}}\quad\text{(filled capacitor)}.
\]
Inside a homogeneous linear dielectric, bound charge sits only where free charge is:
\[
    \rho_b = -\div\Bigl(\frac{\varepsilon_0\chi_e}{\varepsilon}\vect{D}\Bigr) = -\frac{\chi_e}{1+\chi_e}\,\rho_f .
\]
\end{fact}

\subsection{The Clausius--Mossotti Relation}

Naively $\vect{P} = N\lvect{p} = N\alpha\vect{E}$, giving $\chi_e = N\alpha/\varepsilon_0$. But two different fields are in play:
\begin{itemize}
    \item $\vect{E}_{\text{else}}$, the field of everything \emph{except} a given atom (external sources plus all the other atoms). This is what polarises the atom: $\lvect{p} = \alpha\vect{E}_{\text{else}}$.
    \item $\vect{E}$, the macroscopic field: the total field averaged over a region, \emph{including} the atom's own field. This is what enters $\vect{P} = \varepsilon_0\chi_e\vect{E}$.
\end{itemize}

\begin{theorem}[Average field over a sphere]
For charges inside a sphere of radius $R$ with total dipole moment $\lvect{p}$,
\[
    \langle\vect{E}\rangle_{\text{inside}} = -\frac{1}{4\pi\varepsilon_0}\frac{\lvect{p}}{R^3},
\]
independent of the details of the distribution. For charges \emph{outside}, the average over the sphere equals the field at its centre.
\end{theorem}
\begin{proof}[Sketch]
A single charge $q$ at $\lvect{r}$ produces the same average field over the sphere as the field at $\lvect{r}$ of a uniform ball of density $-q/\tfrac43\pi R^3$, namely $-q\lvect{r}/4\pi\varepsilon_0R^3$. Superposing over all the enclosed charges gives $-\lvect{p}/4\pi\varepsilon_0R^3$. The statement for outside charges is the mean value property of the harmonic potential (Section~\ref{sec:em-meanvalue}).
\end{proof}

Give each atom a cell of radius $R$ with $N = 1/(\tfrac43\pi R^3)$, so that $1/(4\pi R^3) = N/3$ (Figure~\ref{fig:em-cm-cells}). Averaging the total field over one cell,
\[
    \vect{E} = \langle\vect{E}\rangle_{\text{cell}} = \vect{E}_{\text{else}} + \vect{E}_{\text{self}},
    \qquad
    \vect{E}_{\text{self}} = -\frac{1}{4\pi\varepsilon_0}\frac{\lvect{p}}{R^3} = -\frac{\alpha\vect{E}_{\text{else}}}{4\pi\varepsilon_0R^3} = -\frac{N\alpha}{3\varepsilon_0}\vect{E}_{\text{else}} .
\]
The atom's field outside its own cell is part of its neighbours' $\vect{E}_{\text{else}}$, so nothing is double counted. Hence
\[
    \vect{E} = \Bigl(1-\frac{N\alpha}{3\varepsilon_0}\Bigr)\vect{E}_{\text{else}} :
\]
the macroscopic field is weaker than the field each atom feels, because the atom's own average field opposes $\lvect{p}$. Setting $\vect{P} = N\alpha\vect{E}_{\text{else}} = \varepsilon_0\chi_e\vect{E}$:

\begin{formula}[Clausius--Mossotti relation]
\[
    \chi_e = \frac{N\alpha/\varepsilon_0}{1-N\alpha/3\varepsilon_0}
    \qquad\Longleftrightarrow\qquad
    \alpha = \frac{3\varepsilon_0}{N}\left(\frac{\varepsilon_r-1}{\varepsilon_r+2}\right).
\]
\end{formula}

For a dilute gas ($N\alpha/\varepsilon_0\ll1$) this reduces to the naive $\chi_e\approx N\alpha/\varepsilon_0$. For dense liquids and solids the correction matters, and it always makes $\chi_e$ larger than the naive value.

\subsection{Examples}

\begin{example}[Insulated wire]
A wire carries line charge $\lambda$ and is wrapped in rubber insulation of permittivity $\varepsilon$ out to radius $b$. By symmetry $\vect{D} = D(s)\,\uvect{s}$, and a Gaussian cylinder of radius $s$ and length $l$ gives $D\,2\pi sl = \lambda l$:
\[
    \vect{D} = \frac{\lambda}{2\pi s}\uvect{s}\quad\text{(all $s$)},
    \qquad
    \vect{E} =
    \begin{cases}
        \dfrac{\lambda}{2\pi\varepsilon s}\uvect{s}, & s<b,\\[3mm]
        \dfrac{\lambda}{2\pi\varepsilon_0 s}\uvect{s}, & s>b.
    \end{cases}
\]
\end{example}

\begin{example}[Dielectric shell around a charged sphere]
A metal sphere of radius $a$ carries charge $Q$ and is surrounded by a linear dielectric $\varepsilon$ for $a<r<b$. Then
\[
    \vect{D} = \frac{Q}{4\pi r^2}\rhat\ (r>a),
    \qquad
    \vect{E} = \frac{Q}{4\pi\varepsilon r^2}\rhat\ (a<r<b),
    \qquad
    \vect{E} = \frac{Q}{4\pi\varepsilon_0 r^2}\rhat\ (r>b).
\]
Integrating in from infinity, the potential at the centre (equal to that of the conductor) is
\[
    \phi(0) = \frac{Q}{4\pi}\left(\frac{1}{\varepsilon_0 b} + \frac{1}{\varepsilon a} - \frac{1}{\varepsilon b}\right).
\]
\end{example}

\begin{example}[Polarised cylindrical shell with no free charge]
Let $\vect{P} = (k/s)\,\uvect{s}$ for $a<s<b$. The bound charges are
\[
    \rho_b = -\frac1s\pdv{}{s}\Bigl(s\cdot\frac ks\Bigr) = 0,
    \qquad
    \sigma_b(b) = \frac kb,
    \qquad
    \sigma_b(a) = -\frac ka .
\]
Symmetry gives $\vect{D} = D(s)\,\uvect{s}$, and $Q_f = 0$, so $D\,2\pi sl = 0$ and $\vect{D} = 0$ everywhere. Hence
\[
    \vect{E} = -\frac{\vect{P}}{\varepsilon_0} = -\frac{k}{\varepsilon_0 s}\uvect{s}\quad(a<s<b),
    \qquad
    \vect{E} = 0\ \text{otherwise},
\]
in agreement with summing the fields of the bound surface charges directly.
\end{example}
