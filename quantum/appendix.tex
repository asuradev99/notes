\section{Integrals}

\begin{formula}[Gaussian integrals]
\[
    \int_{-\infty}^\infty e^{-\alpha x^2}dx = \sqrt{\frac\pi\alpha},
    \qquad
    \int_{-\infty}^\infty x^2e^{-\alpha x^2}dx = \frac12\sqrt{\frac{\pi}{\alpha^3}},
    \qquad
    \int_{-\infty}^\infty x^{2n}e^{-\alpha x^2}dx = \frac{(2n-1)!!}{(2\alpha)^n}\sqrt{\frac\pi\alpha},
\]
\[
    \int_{-\infty}^\infty e^{-\alpha x^2+\beta x}dx = \sqrt{\frac\pi\alpha}\,e^{\beta^2/4\alpha}\quad(\mathrm{Re}\,\alpha>0,\ \beta\in\mathbb{C}),
    \qquad
    \int_0^\infty x^ne^{-\lambda x}dx = \frac{n!}{\lambda^{n+1}}.
\]
\end{formula}
The first follows by squaring and using polar coordinates, $\bigl(\int e^{-\alpha x^2}dx\bigr)^2 = \int_0^{2\pi}d\phi\int_0^\infty e^{-\alpha r^2}r\,dr = \pi/\alpha$; the moments follow by differentiating with respect to $\alpha$, and the shifted one by completing the square.

\begin{formula}[Delta function]
\[
    \delta(x) = \int\frac{dk}{2\pi}e^{ikx} = \lim_{\epsilon\to0}\frac{1}{\sqrt{\pi}\epsilon}e^{-x^2/\epsilon^2},
    \qquad
    \delta(ax) = \frac{\delta(x)}{|a|},
    \qquad
    \delta(g(x)) = \sum_i\frac{\delta(x-x_i)}{|g'(x_i)|},
\]
\[
    \int f(x)\delta^{(n)}(x-a)\,dx = (-1)^nf^{(n)}(a),
    \qquad
    \laplacian\frac1r = -4\pi\delta^3(\lvect{r}).
\]
\end{formula}

\section{Operator Toolkit}

\begin{formula}[Summary of key operator relations]
\[
    [\hat{x},\hat{p}] = i\hbar,
    \quad
    [\hat{a},\hat{a}^\dagger] = 1,
    \quad
    [\hat{J}_i,\hat{J}_j] = i\hbar\epsilon_{ijk}\hat{J}_k,
    \quad
    [\hat{J}_z,\hat{J}_\pm] = \pm\hbar\hat{J}_\pm,
    \quad
    \sigma_i\sigma_j = \delta_{ij} + i\epsilon_{ijk}\sigma_k,
\]
\[
    \hat{a}|n\rangle = \sqrt n|n-1\rangle,
    \quad
    \hat{a}^\dagger|n\rangle = \sqrt{n+1}|n+1\rangle,
    \quad
    \hat{J}_\pm|j,m\rangle = \hbar\sqrt{j(j+1)-m(m\pm1)}|j,m\pm1\rangle,
\]
\[
    \frac{d}{dt}\langle\hat{Q}\rangle = \frac{i}{\hbar}\langle[\hat{H},\hat{Q}]\rangle + \Bigl\langle\pdv{\hat{Q}}{t}\Bigr\rangle,
    \qquad
    \Delta A\,\Delta B\ge\tfrac12|\langle[\hat{A},\hat{B}]\rangle|.
\]
\end{formula}
