\chapter{Quantum Dynamics}
\label{ch:dynamics}

In non-relativistic quantum mechanics, time is a parameter labelling how states evolve, not an operator like $\hat{x}$. Measured quantities are expectation values $\langle\Psi|\hat{A}|\Psi\rangle$, and their time dependence can be placed in the state (the \emph{Schr\"odinger picture}) or in the operators (the \emph{Heisenberg picture}).

% ============================================================
\section{Schr\"odinger Dynamics}
% ============================================================

In the Schr\"odinger picture the state traces out a path in $\mathcal{H}$, $|\Psi,t\rangle = \sum_nc_n(t)|e_n\rangle$, while operators are fixed. Preserving the norm requires unitary evolution:
\[
    |\Psi,t\rangle = U(t,t_0)|\Psi,t_0\rangle,
    \qquad
    U^\dagger U = I .
\]
$U$ is unique and independent of the state, $U(t_0,t_0) = I$, it composes, $U(t_2,t_0) = U(t_2,t_1)U(t_1,t_0)$, and Hermitian conjugation reverses time, $U(t_0,t_1) = U^\dagger(t_1,t_0)$.

Differentiating, $\partial_t|\Psi,t\rangle = (\partial_tU)U^\dagger|\Psi,t\rangle$. Define
\[
    \hat{H}(t) \defeq i\hbar\,\pdv{U(t,t_0)}{t}U^\dagger(t,t_0).
\]
Differentiating $UU^\dagger = I$ shows $\hat{H} = \hat{H}^\dagger$, and the composition law shows it is independent of $t_0$. This gives the Schr\"odinger equation.

\begin{formula}[Schr\"odinger equation]
\[
    i\hbar\pdv{t}|\Psi,t\rangle = \hat{H}(t)|\Psi,t\rangle,
    \qquad
    i\hbar\pdv{t}U(t,t_0) = \hat{H}(t)U(t,t_0).
\]
\end{formula}

Any Hermitian operator can generate time evolution; the question is which physical system it describes. Classically $\frac{d}{dt}Q(x,p) = \{Q,H\}_{\text{PB}}$; quantum mechanically, for $\partial_t\hat{Q} = 0$,
\[
    \frac{d}{dt}\langle\hat{Q}\rangle = \frac{i}{\hbar}\langle[\hat{H},\hat{Q}]\rangle .
\]
So a classical system is quantized by promoting $x,p\to\hat{x},\hat{p}$ in $H(x,p)$. No classical correspondence is needed, though: for present-day quantum hardware the question is rather which Hamiltonians can be implemented reliably.

\begin{formula}[Solving for $U$]
\begin{enumerate}
    \item \textbf{Time-independent $\hat{H}$} (e.g.\ the oscillator): $U(t,t_0) = \exp\bigl[-\frac{i}{\hbar}\hat{H}(t-t_0)\bigr]$.
    \item \textbf{$[\hat{H}(t_1),\hat{H}(t_2)] = 0$} (e.g.\ spin in $\vect{B} = B(t)\khat$): $U(t,t_0) = \exp\bigl[-\frac{i}{\hbar}\int_{t_0}^tdt'\,\hat{H}(t')\bigr]$.
    \item \textbf{General case} (e.g.\ $\vect{B} = B_x(t)\ihat + B_z(t)\khat$): the time-ordered exponential
    \[
        U(t,t_0) = \mathcal{T}\exp\Bigl[-\frac{i}{\hbar}\int_{t_0}^tdt'\,\hat{H}(t')\Bigr]
        = 1 + \Bigl(-\frac{i}{\hbar}\Bigr)\int_{t_0}^tdt_1\hat{H}(t_1) + \Bigl(-\frac{i}{\hbar}\Bigr)^2\int_{t_0}^tdt_1\hat{H}(t_1)\int_{t_0}^{t_1}dt_2\hat{H}(t_2) + \cdots,
    \]
    ordered $t\ge t_1\ge t_2\ge\cdots\ge t_0$ (with no $1/n!$, since the ordering removes the permutations).
\end{enumerate}
\end{formula}
Case 2 works because $G = -\frac{i}{\hbar}\int\hat{H}$ commutes with $G' = -\frac{i}{\hbar}\hat{H}(t)$, so $\partial_te^G = G'e^G$ term by term. In case 3 this fails.

\begin{example}[Time evolution with a time-independent Hamiltonian]
If $\hat{H}|n\rangle = E_n|n\rangle$, then $U(t,0) = e^{-i\hat{H}t/\hbar} = \sum_ne^{-iE_nt/\hbar}|n\rangle\langle n|$ by the spectral theorem, and
\[
    |\Psi,t\rangle = \sum_ne^{-iE_nt/\hbar}|n\rangle\langle n|\Psi,0\rangle,
\]
recovering the expansion-in-eigenstates recipe of Chapter~\ref{ch:schrodinger}. In position space, $\langle x|U(t,0)|x'\rangle = K(x,t;x',0)$ is the \emph{propagator}: $\Psi(x,t) = \int dx'\,K(x,t;x',0)\Psi(x',0)$. For a free particle, inserting momentum states,
\[
    K = \int\frac{dp}{2\pi\hbar}e^{ip(x-x')/\hbar - ip^2t/2m\hbar} = \sqrt{\frac{m}{2\pi i\hbar t}}\exp\Bigl[\frac{im(x-x')^2}{2\hbar t}\Bigr].
\]
\end{example}

\begin{proof}[Why the time-ordered exponential works]
Differentiating the series term by term, $\partial_t$ acts only on the upper limit of the outermost integral, giving $-\frac{i}{\hbar}\hat{H}(t)$ times the series with one fewer integral, i.e.\ $-\frac{i}{\hbar}\hat{H}(t)U(t,t_0)$, with $\hat{H}(t)$ correctly on the left. Writing the $n$th term with all $n$ integrals running from $t_0$ to $t$ would overcount by $n!$, and require the integrand to be ordered, which is what $\mathcal{T}$ means.
\end{proof}

\section{Heisenberg Dynamics}
% ============================================================

Since $\langle\Psi,t|\hat{A}_S|\Psi,t\rangle = \langle\Psi,t_0|U^\dagger\hat{A}_SU|\Psi,t_0\rangle$, we can instead evolve the operators.

\begin{definition}[Heisenberg operator]
\[
    \hat{A}_H(t) = U^\dagger(t,t_0)\,\hat{A}_S\,U(t,t_0).
\]
\end{definition}

Products and commutators carry over, $(\hat{A}\hat{B})_H = \hat{A}_H\hat{B}_H$ and $[\hat{A}_H,\hat{B}_H] = U^\dagger[\hat{A}_S,\hat{B}_S]U$; in particular $[\hat{x}_H(t),\hat{p}_H(t)] = i\hbar$. If $\hat{A}_S$ commutes with $\hat{H}_S$, then $\hat{A}_H = \hat{A}_S$; and $\hat{H}_H = \hat{H}_S$ when $\hat{H}$ commutes with itself at different times.

\begin{formula}[Heisenberg equation of motion]
\[
    i\hbar\frac{d}{dt}\hat{A}_H(t) = [\hat{A}_H(t),\hat{H}_H(t)] + i\hbar\Bigl(\pdv{\hat{A}_S}{t}\Bigr)_H .
\]
\end{formula}

\begin{theorem}[Ehrenfest's theorem]
For $\hat{H} = \hat{p}^2/2m + V(\hat{x})$,
\[
    \frac{d}{dt}\langle\hat{x}\rangle = \frac{\langle\hat{p}\rangle}{m},
    \qquad
    \frac{d}{dt}\langle\hat{p}\rangle = -\Bigl\langle\pdv{V}{x}\Bigr\rangle,
    \qquad
    m\frac{d^2}{dt^2}\langle\hat{x}\rangle = -\Bigl\langle\pdv{V}{x}\Bigr\rangle,
\]
the quantum analogue of Newton's second law. (Use $[\hat{x},\hat{p}^2] = 2i\hbar\hat{p}$ and $[\hat{p},V(\hat{x})] = -i\hbar V'(\hat{x})$.)
\end{theorem}

\begin{proof}[Derivation of the Heisenberg equation]
Differentiate $\hat{A}_H = U^\dagger\hat{A}_SU$, using $i\hbar\partial_tU = \hat{H}_SU$ and $-i\hbar\partial_tU^\dagger = U^\dagger\hat{H}_S$:
\[
    i\hbar\frac{d\hat{A}_H}{dt} = -U^\dagger\hat{H}_S\hat{A}_SU + U^\dagger\hat{A}_S\hat{H}_SU + i\hbar U^\dagger(\partial_t\hat{A}_S)U .
\]
Inserting $UU^\dagger = I$ between the operators turns the first two terms into $[\hat{A}_H,\hat{H}_H]$, with $\hat{H}_H = U^\dagger\hat{H}_SU$.
\end{proof}

Ehrenfest's theorem does not mean $\langle x\rangle$ follows a classical trajectory: in general $\langle V'(\hat{x})\rangle\neq V'(\langle x\rangle)$. They agree exactly for potentials at most quadratic (free particle, constant force, oscillator), and approximately when the wavepacket is narrow compared with the scale on which $V'$ varies.

\begin{example}[Free particle in the Heisenberg picture]
For $\hat{H} = \hat{p}^2/2m$, $\dot{\hat{p}}_H = 0$ and $\dot{\hat{x}}_H = \hat{p}_H/m$, so $\hat{x}_H(t) = \hat{x} + \hat{p}t/m$. Then
\[
    \Delta x^2(t) = \Delta x^2(0) + \frac{t}{m}\bigl(\langle\hat{x}\hat{p}+\hat{p}\hat{x}\rangle - 2\langle x\rangle\langle p\rangle\bigr) + \frac{t^2}{m^2}\Delta p^2,
\]
which for an initial Gaussian (no $x$--$p$ correlation) is $a^2 + \hbar^2t^2/4m^2a^2$, the spreading found before, now in two lines.
\end{example}

\begin{example}[Oscillator in the Heisenberg picture]
For the oscillator (Chapter~\ref{ch:sho}, with $|n\rangle$ the number states), $\frac{d}{dt}\hat{x}_H = \hat{p}_H/m$ and $\frac{d}{dt}\hat{p}_H = -m\omega^2\hat{x}_H$, so
\[
    \hat{x}_H(t) = \hat{x}\cos\omega t + \frac{\hat{p}}{m\omega}\sin\omega t,
    \qquad
    \hat{p}_H(t) = \hat{p}\cos\omega t - m\omega\hat{x}\sin\omega t,
\]
and $[\hat{H},\hat{a}] = -\hbar\omega\hat{a}$ gives $\hat{a}_H(t) = e^{-i\omega t}\hat{a}$, $\hat{a}^\dagger_H(t) = e^{i\omega t}\hat{a}^\dagger$. These look like the classical equations of motion, but in an energy eigenstate $\langle n|\hat{x}_H(t)|n\rangle = 0$: energy eigenstates do not behave classically.
\end{example}

\begin{example}[Spin precession in the Heisenberg picture]
For $\hat{H} = -\gamma B\hat{S}_z$, using $[\hat{S}_z,\hat{S}_x] = i\hbar\hat{S}_y$ and $[\hat{S}_z,\hat{S}_y] = -i\hbar\hat{S}_x$,
\[
    \dot{\hat{S}}_x = \frac{i}{\hbar}[\hat{H},\hat{S}_x] = \gamma B\hat{S}_y,
    \qquad
    \dot{\hat{S}}_y = -\gamma B\hat{S}_x,
    \qquad
    \dot{\hat{S}}_z = 0,
\]
so $\hat{S}_x(t) = \hat{S}_x\cos\omega t + \hat{S}_y\sin\omega t$ and $\hat{S}_y(t) = \hat{S}_y\cos\omega t - \hat{S}_x\sin\omega t$ with $\omega = \gamma B$. The operators themselves rotate, and any expectation value follows by taking $\langle\Psi,0|\cdot|\Psi,0\rangle$: the same Larmor precession found in the Schr\"odinger picture (Chapter~\ref{ch:twostate}), without ever writing down the state at time $t$.
\end{example}

\section{The Interaction Picture}
% ============================================================

For $\hat{H} = \hat{H}_0 + \hat{V}(t)$ with $\hat{H}_0$ solvable, a third picture is useful: let $\hat{H}_0$ evolve the operators and $\hat{V}$ evolve the states. Define
\[
    |\Psi_I,t\rangle = e^{i\hat{H}_0t/\hbar}|\Psi,t\rangle,
    \qquad
    \hat{V}_I(t) = e^{i\hat{H}_0t/\hbar}\hat{V}(t)e^{-i\hat{H}_0t/\hbar}.
\]
Differentiating, the $\hat{H}_0$ terms cancel:
\[
    i\hbar\frac{d}{dt}|\Psi_I,t\rangle = \hat{V}_I(t)|\Psi_I,t\rangle .
\]
If $\hat{V} = 0$ the interaction-picture state is constant, and all change comes from the interaction. In the basis of $\hat{H}_0$ eigenstates, $\langle k|\hat{V}_I|n\rangle = e^{i(E_k-E_n)t/\hbar}V_{kn}$, which is exactly how the oscillating phases $e^{\pm2i\omega_0t}$ arose in the maser and the rotating frame arose in NMR. The interaction picture is the natural starting point for time-dependent perturbation theory.

\section{Coherent States}
% ============================================================

Coherent states behave classically in many ways, and describe lasers and much of quantum optics.

\begin{definition}[Translation operator]
\[
    \hat{T}_{x_0} = \exp\Bigl(-\frac{i}{\hbar}x_0\hat{p}\Bigr).
\]
Translations form a commutative group, $\hat{T}_{x_0}\hat{T}_{y_0} = \hat{T}_{x_0+y_0}$, and $\hat{T}_{-x_0} = \hat{T}_{x_0}^\dagger = \hat{T}_{x_0}^{-1}$ is unitary. They translate the position operator, $\hat{T}_{x_0}^\dagger\hat{x}\hat{T}_{x_0} = \hat{x} + x_0$, and the states, $\hat{T}_{x_0}|x_1\rangle = |x_1+x_0\rangle$, so $\langle x|\hat{T}_{x_0}|\psi\rangle = \psi(x-x_0)$.
\end{definition}

\begin{definition}[Coherent state]
The coherent (Glauber) state is the translated oscillator ground state,
\[
    |\tilde{x}_0\rangle = \hat{T}_{x_0}|0\rangle,
    \qquad
    \langle x|\tilde{x}_0\rangle = \psi_0(x-x_0).
\]
\end{definition}

It is normalized, with $\langle\hat{x}\rangle = x_0$, $\langle\hat{p}\rangle = 0$ and $\langle\hat{H}\rangle = \frac12\hbar\omega + \frac12m\omega^2x_0^2$: the zero-point energy plus the classical potential energy at $x_0$. In the Heisenberg picture,
\[
    \langle\hat{x}_H(t)\rangle = x_0\cos\omega t,
    \qquad
    \langle\hat{p}_H(t)\rangle = -m\omega x_0\sin\omega t,
\]
classical harmonic motion, and $\Delta x^2 = \hbar/2m\omega$, $\Delta p^2 = m\hbar\omega/2$ at all times. So $\Delta x\,\Delta p = \hbar/2$ for all $t$: the Gaussian oscillates in the potential without changing shape.

With $d = \sqrt{\hbar/m\omega}$, write $\hat{T}_{x_0} = \exp\bigl(\frac{x_0}{\sqrt2d}(\hat{a}^\dagger-\hat{a})\bigr)$ and use the Baker--Campbell--Hausdorff formula $e^{A+B} = e^Ae^Be^{-\frac12[A,B]}$ (valid when $[A,B]$ is a number):
\[
    |\tilde{x}_0\rangle = e^{-x_0^2/4d^2}\sum_{n=0}^\infty\frac{1}{\sqrt{n!}}\Bigl(\frac{x_0}{\sqrt2d}\Bigr)^n|n\rangle .
\]
The occupation probabilities are Poisson, $P_n = \lambda^ne^{-\lambda}/n!$ with $\lambda = x_0^2/2d^2$, so $\langle N\rangle = \lambda$, $\Delta N = \sqrt\lambda$, and $\Delta H = \hbar\omega\sqrt\lambda$. For $x_0\gg d$, $\hbar\omega\ll\Delta H\ll\langle H\rangle$: the energy is sharp relative to its size, and the state becomes classical (Bohr's correspondence principle).

\begin{formula}[General coherent states]
With the displacement operator $D(\alpha) = \exp(\alpha\hat{a}^\dagger - \alpha^*\hat{a})$, $\alpha\in\mathbb{C}$,
\[
    |\tilde\alpha\rangle = D(\alpha)|0\rangle = e^{-|\alpha|^2/2}e^{\alpha\hat{a}^\dagger}|0\rangle,
    \qquad
    \hat{a}|\tilde\alpha\rangle = \alpha|\tilde\alpha\rangle,
\]
\[
    \langle\hat{x}\rangle = \sqrt2d\,\mathrm{Re}\,\alpha,
    \qquad
    \langle\hat{p}\rangle = \frac{\sqrt2\hbar}{d}\mathrm{Im}\,\alpha,
    \qquad
    |\tilde\alpha,t\rangle = e^{-i\omega t/2}|\widetilde{\alpha e^{-i\omega t}}\rangle .
\]
Coherent states are eigenstates of $\hat{a}$, which can serve as their definition; complex eigenvalues are allowed since $\hat{a}$ is not Hermitian.
\end{formula}

\begin{fact}[Coherent states are an overcomplete basis]
They are not orthogonal, $|\langle\tilde\alpha|\tilde\beta\rangle|^2 = e^{-|\alpha-\beta|^2}$, but they are complete,
\[
    I = \int\frac{d^2\alpha}{\pi}|\tilde\alpha\rangle\langle\tilde\alpha|,
\]
with $d^2\alpha/\pi = d\langle x\rangle\,d\langle p\rangle/2\pi\hbar$ the phase-space volume. In phase space a coherent state is a blob of fixed size $\Delta x\Delta p\sim\hbar$ circling the origin at frequency $\omega$, with number--phase uncertainty $\Delta N\sim|\alpha|$, $\Delta\varphi\sim1/|\alpha|$. A number state instead fills an annulus, $|\langle\tilde\alpha|n\rangle|^2 = e^{-|\alpha|^2}|\alpha|^{2n}/n!$.
\end{fact}

\end{proof}

\section{Squeezed States}
% ============================================================

Another class of minimum-uncertainty states, \emph{squeezed states}, matters in quantum optics and is critical to Advanced LIGO. Suddenly change an oscillator from $(m_1,\omega_1)$ to $(m_2,\omega_2)$. The old ground state $|0\rangle_{(1)}$ is no longer stationary but still has $\Delta x\Delta p = \hbar/2$, with $\Delta x = e^{-\gamma}\sqrt{\hbar/2m_2\omega_2}$ and $e^\gamma = \sqrt{m_1\omega_1/m_2\omega_2}$. The two sets of ladder operators are related by a Bogoliubov transformation,
\[
    \begin{pmatrix}\hat{a}_1\\\hat{a}_1^\dagger\end{pmatrix} = \begin{pmatrix}\cosh\gamma&\sinh\gamma\\\sinh\gamma&\cosh\gamma\end{pmatrix}\begin{pmatrix}\hat{a}_2\\\hat{a}_2^\dagger\end{pmatrix}.
\]
Solving $\hat{a}_1|0\rangle_{(1)} = 0$ with the ansatz $|0\rangle_{(1)} = N\exp(-\frac12f\hat{a}_2^{\dagger2})|0\rangle_{(2)}$ gives $f = \tanh\gamma$ and $N = 1/\sqrt{\cosh\gamma}$.

\begin{formula}[Squeezed states]
For any oscillator,
\[
    |0\rangle_\gamma = \frac{1}{\sqrt{\cosh\gamma}}\exp\Bigl(-\frac12\tanh\gamma\,\hat{a}^{\dagger2}\Bigr)|0\rangle = S(\gamma)|0\rangle,
    \qquad
    S(\gamma) = \exp\Bigl(-\frac12(\gamma^*\hat{a}^2 - \gamma\hat{a}^{\dagger2})\Bigr),
\]
with $S(\gamma)$ unitary. More generally $|\tilde\alpha\rangle_\gamma = D(\alpha)S(\gamma)|0\rangle$, an eigenstate of $\hat{a}\cosh\gamma + \hat{a}^\dagger\sinh\gamma$ with eigenvalue $\alpha$. All are minimum-uncertainty states.
\end{formula}

As $\gamma\to\infty$, $|0\rangle_\infty\propto e^{-\hat{a}^{\dagger2}/2}|0\rangle$ is annihilated by $\hat{x}\propto\hat{a}+\hat{a}^\dagger$: it is the position eigenstate $|x = 0\rangle$, and translations give the others. As $\gamma\to-\infty$ one obtains momentum eigenstates. In phase space, squeezing deforms the round blob of the coherent state into an ellipse.

\paragraph{Quadrature variances.} Define the quadratures $\hat{X}_1 = \frac12(\hat{a}+\hat{a}^\dagger)$ and $\hat{X}_2 = \frac{1}{2i}(\hat{a}-\hat{a}^\dagger)$, dimensionless versions of $\hat{x}$ and $\hat{p}$ with $[\hat{X}_1,\hat{X}_2] = \frac i2$, so $\Delta X_1\Delta X_2\ge\frac14$. For a coherent state $\Delta X_1 = \Delta X_2 = \frac12$. The squeezed vacuum is annihilated by $\hat{b} = \hat{a}\cosh\gamma + \hat{a}^\dagger\sinh\gamma$, which satisfies $[\hat{b},\hat{b}^\dagger] = 1$; inverting, $\hat{a} = \hat{b}\cosh\gamma - \hat{b}^\dagger\sinh\gamma$. Then $\hat{X}_1 = e^{-\gamma}\cdot\frac12(\hat{b}+\hat{b}^\dagger)$ and $\hat{X}_2 = e^{\gamma}\cdot\frac{1}{2i}(\hat{b}-\hat{b}^\dagger)$, and since the state is the vacuum of $\hat{b}$,
\[
    \Delta X_1 = \tfrac12e^{-\gamma},
    \qquad
    \Delta X_2 = \tfrac12e^{\gamma}:
\]
one quadrature is squeezed below the vacuum noise and the other stretched, keeping the product at its minimum. Advanced LIGO injects squeezed vacuum into its interferometer to reduce the phase noise in the quadrature it measures.

\section{The Quantized Electromagnetic Field}
% ============================================================

A single mode of the field in a cubic cavity of volume $V = L^3$ has
\[
    E_x(z,t) = \sqrt{\frac{2\omega^2}{\varepsilon_0V}}\,q(t)\sin kz,
    \qquad
    B_y(z,t) = \frac{\mu_0\varepsilon_0}{k}\sqrt{\frac{2\omega^2}{\varepsilon_0V}}\,p(t)\cos kz,
    \qquad
    k = \frac\omega c = \frac{n\pi}{L},
\]
where Maxwell's equations require $p = \dot{q}$. The energy is
\[
    H = \frac12\int dV\Bigl[\varepsilon_0E^2 + \frac{1}{\mu_0}B^2\Bigr] = \frac12\bigl(p^2 + \omega^2q^2\bigr),
\]
an oscillator with $m = 1$. Quantize by promoting $q,p$ to operators with $[\hat{q},\hat{p}] = i\hbar$, and define $\hat{a} = (\omega\hat{q} + i\hat{p})/\sqrt{2\hbar\omega}$, which removes a photon.

\paragraph{Why $p = \dot{q}$ and $H = \frac12(p^2+\omega^2q^2)$.} Faraday's law $\partial_zE_x = -\partial_tB_y$ with the mode functions above gives $-k\sqrt{\cdot}\,q\cos kz = -\frac{\mu_0\varepsilon_0}{k}\sqrt{\cdot}\,\dot{p}\cos kz$, so $\dot{p} = -\omega^2q$ (using $\omega = ck$), and the Amp\`ere--Maxwell law similarly gives $\dot{q} = p$. These are Hamilton's equations for $H = \frac12(p^2+\omega^2q^2)$, and the energy integral confirms it: $\int\sin^2kz\,dV = \int\cos^2kz\,dV = V/2$, so $\frac12\varepsilon_0\int E^2 = \frac12\omega^2q^2$ and $\frac{1}{2\mu_0}\int B^2 = \frac12p^2$.

\begin{formula}[Field operators]
\[
    \hat{E}_x(z,t) = \mathcal{E}_0\bigl(\hat{a}e^{-i\omega t} + \hat{a}^\dagger e^{i\omega t}\bigr)\sin kz,
    \qquad
    \mathcal{E}_0 = \sqrt{\frac{\hbar\omega}{\varepsilon_0V}} .
\]
\end{formula}

In a number state, $\langle n|\hat{E}_x|n\rangle = 0$ and $\Delta E_x = \mathcal{E}_0|\sin kz|\sqrt{2n+1}$: nothing like a classical field. (Since $\hat{a}$ and $\hat{a}^\dagger$ change $n$, their diagonal elements vanish; $\langle n|(\hat{a}e^{-i\omega t}+\hat{a}^\dagger e^{i\omega t})^2|n\rangle = \langle n|\hat{a}\hat{a}^\dagger + \hat{a}^\dagger\hat{a}|n\rangle = 2n+1$.) Even the vacuum $n = 0$ has $\Delta E_x\neq0$: these vacuum fluctuations drive spontaneous emission and the Lamb shift. In a coherent state,
\[
    \langle\tilde\alpha|\hat{E}_x|\tilde\alpha\rangle = 2\mathcal{E}_0|\alpha|\sin kz\cos(\omega t-\theta),
    \qquad
    \Delta E_x = \mathcal{E}_0|\sin kz|,
\]
with $\alpha = |\alpha|e^{i\theta}$, an oscillating field with fixed, $\alpha$-independent noise. For $|\alpha|\gg1$ it behaves like the classical field, and the single-mode field of a laser is idealized as a coherent state. Squeezed states trade noise between quadratures: amplitude-squeezed or phase-squeezed light. A general mode with wavevector $\lvect{k}$ and polarization $\lvect{e}_s$ is $\hat{\vect{E}}_{\lvect{k},s}\propto\lvect{e}_s\bigl(\hat{a}_{\lvect{k}}e^{i(\lvect{k}\cdot\lvect{r}-\omega t)} + \hat{a}_{\lvect{k}}^\dagger e^{-i(\lvect{k}\cdot\lvect{r}-\omega t)}\bigr)$, with $\omega = c|\lvect{k}|$.

\section{Potentials as Phases: Gauge Transformations and the Aharonov--Bohm Effect}
\label{sec:gauge-ab}
% ============================================================

This section uses Gaussian units, following Sakurai. The particle's charge $e$ carries its sign ($e<0$ for the electron), and the flux quantum is
\[
    \Phi_0 = \frac{2\pi\hbar c}{|e|}\approx4.14\times10^{-7}\ \text{G\,cm}^2
    \qquad(\text{SI: }h/|e|\approx4.14\times10^{-15}\ \text{Wb}).
\]
A charge in electromagnetic potentials has
\[
    \hat{H} = \frac{1}{2m}\Bigl(\hat{\lvect{p}} - \frac{e\vect{A}}{c}\Bigr)^2 + e\phi + V,
    \qquad
    \vect{E} = -\grad\phi - \frac1c\partial_t\vect{A},
    \quad
    \vect{B} = \curl\vect{A}.
\]
The central idea: potentials enter quantum mechanics only through phases. A single phase is never observable, but the \emph{difference} of the phases accumulated along two paths that recombine is, and it is given by a closed-loop integral of the potentials, which is gauge invariant modulo $2\pi$.

\subsection{A Potential That Exerts No Force Still Produces a Phase}

Shifting the potential by a constant, $V\to V+V_0$, multiplies every state by $e^{-iV_0(t-t_0)/\hbar}$ and shifts every energy by $V_0$. Nothing observable changes, because observables depend only on energy differences. Now let the shift depend on time but not on position, $V\to V+V_0(t)$, and look for a solution of the shifted problem of the form $e^{if(t)}|\alpha;t\rangle$, where $|\alpha;t\rangle$ solves the unshifted problem. The time derivative brings down the derivative of the phase,
\[
    i\hbar\,\partial_t\bigl(e^{if}|\alpha;t\rangle\bigr) = \bigl(-\hbar\dot{f} + \hat{H}\bigr)e^{if}|\alpha;t\rangle,
\]
which equals $(\hat{H}+V_0)e^{if}|\alpha;t\rangle$ provided $-\hbar\dot{f} = V_0(t)$.

\begin{formula}[Time-dependent uniform potential]
\[
    |\alpha;t\rangle\ \to\ \exp\Bigl[-\frac{i}{\hbar}\int_{t_0}^tV_0(t')\,dt'\Bigr]|\alpha;t\rangle .
\]
\end{formula}

This works only because the phase commutes with the kinetic term. A position-dependent phase does not commute with $\hat{\lvect{p}}$; moving $\hat{\lvect{p}}$ through it leaves behind the gradient of the phase (Chapter~\ref{ch:formalism}):
\begin{equation}
    \hat{\lvect{p}}\,e^{i\theta(\lvect{x})} = e^{i\theta(\lvect{x})}\bigl(\hat{\lvect{p}} + \hbar\grad\theta\bigr).
    \label{eq:p-through-phase}
\end{equation}
So a position-dependent phase changes the kinetic energy, and absorbing it requires a compensating shift of the momentum, i.e.\ a vector potential. This is the seed of the gauge transformation below.

\paragraph{The electrostatic interference experiment.} A beam is split into two wavepackets, each entering a metal cage. While the packets are inside, potentials $V_1(t)$ and $V_2(t)$ are applied to the cages and switched off before the packets leave. Inside each cage the potential is uniform, so the particle feels \emph{no force at any time}. By the result above, packet $j$ acquires the phase
\begin{equation}
    \phi_j = -\frac1\hbar\int_{t_i}^{t_f}V_j(t)\,dt,
    \qquad
    \Delta\phi\equiv\phi_1-\phi_2 = \frac1\hbar\int_{t_i}^{t_f}\bigl[V_2(t)-V_1(t)\bigr]dt .
    \label{eq:cage-phase}
\end{equation}
Only the potential difference enters, consistent with the arbitrariness of the zero of energy.

While the packets are apart they occupy orthogonal states, $|\text{up}\rangle$ and $|\text{down}\rangle$, and $|\psi\rangle = \frac{1}{\sqrt2}\bigl(|\text{up}\rangle + e^{i\Delta\phi}|\text{down}\rangle\bigr)$ has norm one for every $\Delta\phi$: the relative phase is invisible. Interference requires mapping both paths onto \emph{common} final states, which a 50/50 beam splitter does. Being unitary, it must send orthogonal inputs to orthogonal outputs; with equal weights this forces a relative sign:
\[
    |\text{up}\rangle\to\frac{|a\rangle+|b\rangle}{\sqrt2},
    \qquad
    |\text{down}\rangle\to\frac{|a\rangle-|b\rangle}{\sqrt2},
    \qquad
    U = \frac{1}{\sqrt2}\begin{pmatrix}1&1\\1&-1\end{pmatrix}.
\]
Then
\[
    |\psi_f\rangle = \tfrac12\bigl[(1+e^{i\Delta\phi})|a\rangle + (1-e^{i\Delta\phi})|b\rangle\bigr],
    \qquad
    P_a = \frac{1+\cos\Delta\phi}{2},
    \quad
    P_b = \frac{1-\cos\Delta\phi}{2},
\]
and $P_a+P_b = 1$: the voltage redistributes particles between the outputs, and creates or destroys none.

\begin{fact}[A common slip]
Writing both branches as the same ket, $|\alpha\rangle + e^{i\Delta\phi}|\alpha\rangle$, gives a ``norm'' $1+\cos\Delta\phi$ that varies between 0 and 2. That expression is $2P_a$: it projects both paths onto one output without the beam splitter's factor. Adding a state to a phase-shifted copy of itself is not a unitary operation.
\end{fact}

If instead the packets simply overlap on a screen, like a double slit (Section~\ref{sec:double-slit}), the intensity at position $x$ is
\[
    I(x)\propto1 + \cos\Bigl(\frac{kd\,x}{L} + \Delta\phi\Bigr),
\]
so the voltage slides the fringe pattern rigidly without changing its spacing. Equivalently, at a fixed detector the counts oscillate as the voltage is varied, with period $\Delta V\,\tau = 2\pi\hbar$ for a constant difference applied for a time $\tau$.

\paragraph{The classical limit.} Both fringe spacings are proportional to $\hbar$: $\Delta x_{\text{fringe}} = 2\pi\hbar L/pd$ in space and $2\pi\hbar/\tau$ in voltage. As $\hbar\to0$, any detector of finite size and any voltage with a little noise average over many fringes, $\langle\cos\Delta\phi\rangle\to0$, and the intensity reduces to the classical sum of the two beams. The effect is purely quantum: a potential that exerts no force is observable, but only through a phase, and only when the paths recombine.

\subsection{Gauge Transformations}
\label{sec:gauge}

The potentials $\vect{A}$ and $\tilde{\vect{A}} = \vect{A} + \grad\Lambda$ produce the same field. We look for the operator $\mathcal{G}$ relating the kets that describe the same physical situation in the two gauges, $|\tilde\alpha\rangle = \mathcal{G}|\alpha\rangle$. The physical requirements are
\begin{align*}
    &\text{normalization:} && \mathcal{G}^\dagger\mathcal{G} = 1,\\
    &\text{position:} && \mathcal{G}^\dagger\hat{\lvect{x}}\mathcal{G} = \hat{\lvect{x}},\\
    &\text{kinetic momentum } \bm{\Pi} = m\dot{\lvect{x}}: && \mathcal{G}^\dagger\Bigl(\hat{\lvect{p}} - \frac{e\tilde{\vect{A}}}{c}\Bigr)\mathcal{G} = \hat{\lvect{p}} - \frac{e\vect{A}}{c}.
\end{align*}
The canonical momentum $\hat{\lvect{p}}$ is \emph{not} required to be invariant: it is tied to the choice of gauge by $\hat{\lvect{p}} = \bm{\Pi} + e\vect{A}/c$.

\paragraph{The operator must be a phase.} Multiplying the position condition by $\mathcal{G}$ on the left gives $[\hat{\lvect{x}},\mathcal{G}] = 0$. An operator that commutes with position is diagonal in the position basis, so it acts on wavefunctions by multiplication: $\langle\lvect{x}'|\mathcal{G}|\lvect{x}''\rangle = g(\lvect{x}')\delta^3(\lvect{x}'-\lvect{x}'')$, i.e.\ $(\mathcal{G}\psi)(\lvect{x}) = g(\lvect{x})\psi(\lvect{x})$. Unitarity then requires $|g(\lvect{x})|^2 = 1$ \emph{at every point separately}, since every diagonal entry of a unitary diagonal matrix lies on the unit circle. Hence $g(\lvect{x}) = e^{i\theta(\lvect{x})}$.

\paragraph{Which phase.} Conjugate $\hat{\lvect{p}}$ by the phase using the BCH expansion $e^XYe^{-X} = Y + [X,Y] + \frac{1}{2!}[X,[X,Y]] + \cdots$ with $X = -i\theta(\hat{\lvect{x}})$ and $Y = \hat{\lvect{p}}$. Since $[\theta(\hat{\lvect{x}}),\hat{\lvect{p}}] = i\hbar\grad\theta$,
\[
    [X,Y] = -i\,(i\hbar\grad\theta) = \hbar\grad\theta,
    \qquad
    [X,[X,Y]] = [-i\theta,\hbar\grad\theta] = 0,
\]
because functions of position commute. The series terminates, giving $e^{-i\theta}\hat{\lvect{p}}e^{i\theta} = \hat{\lvect{p}} + \hbar\grad\theta$, which is \eqref{eq:p-through-phase} again. Inserting this into the kinetic-momentum condition,
\[
    \hat{\lvect{p}} + \hbar\grad\theta - \frac{e\vect{A}}{c} - \frac{e\grad\Lambda}{c} = \hat{\lvect{p}} - \frac{e\vect{A}}{c}
    \qquad\Longrightarrow\qquad
    \theta(\lvect{x}) = \frac{e\Lambda(\lvect{x})}{\hbar c} + \text{const}.
\]
The constant is a global phase and is unobservable.

\begin{theorem}[Gauge transformation]
\[
    \vect{A}\to\vect{A} + \grad\Lambda,
    \qquad
    \psi(\lvect{x})\to\exp\Bigl[\frac{ie\Lambda(\lvect{x})}{\hbar c}\Bigr]\psi(\lvect{x}).
\]
Nothing was assumed about the form of $\mathcal{G}$: invariance of position, kinetic momentum and normalization force it.
\end{theorem}

\paragraph{Consistency with the dynamics.} The defining property can be written as an intertwining relation, $(\hat{\lvect{p}} - e\tilde{\vect{A}}/c)\mathcal{G} = \mathcal{G}(\hat{\lvect{p}} - e\vect{A}/c)$. Applying it twice gives the same relation for $\bm{\Pi}^2$, hence $\tilde{H}\mathcal{G} = \mathcal{G}H$. For a time-independent $\Lambda$, $\mathcal{G}$ commutes with $\partial_t$, so if $\psi$ solves the Schr\"odinger equation with $\vect{A}$, then $\mathcal{G}\psi$ solves it with $\tilde{\vect{A}}$.

\begin{formula}[Time-dependent gauge functions]
For $\Lambda(\lvect{x},t)$ the full transformation also shifts the scalar potential:
\[
    \vect{A}\to\vect{A} + \grad\Lambda,
    \qquad
    \phi\to\phi - \frac1c\partial_t\Lambda,
    \qquad
    \psi\to e^{ie\Lambda/\hbar c}\psi,
\]
leaving $\vect{E}$ and $\vect{B}$ unchanged. The extra term $-\frac ec\partial_t\Lambda$ from $i\hbar\partial_t$ acting on the phase is exactly cancelled by the shift of $e\phi$. The uniform potential above is the special case $\Lambda = \Lambda(t)$.
\end{formula}

\subsection{Field-Free Regions: the Vector Potential Is Locally Pure Gauge}
\label{sec:field-free}

Suppose $\curl\vect{A} = 0$ throughout a region, while $\vect{A}\neq0$. In a \emph{simply connected} region define $\chi(\lvect{x}) = \int_{\lvect{x}_0}^{\lvect{x}}\vect{A}\cdot d\lvect{s}$. Two paths from $\lvect{x}_0$ to $\lvect{x}$ form a loop bounding a surface inside the region, where the field vanishes, so by Stokes the integral is path independent and $\chi$ is a well-defined function. Comparing $\chi(\lvect{x}+d\lvect{x})$ and $\chi(\lvect{x})$ along a path that ends with the step $d\lvect{x}$ gives $\grad\chi = \vect{A}$: the vector potential is a pure gradient. Taking $\psi_0$ to solve the problem with no vector potential and gauge-transforming with $\Lambda = \chi$ (from $\vect{0}$ to $\grad\chi = \vect{A}$):

\begin{formula}[Phase from a field-free vector potential]
\[
    \psi(\lvect{x},t) = \exp\Bigl[\frac{ie}{\hbar c}\int_{\lvect{x}_0}^{\lvect{x}}\vect{A}\cdot d\lvect{s}\Bigr]\psi_0(\lvect{x},t).
\]
\end{formula}

\paragraph{The dictionary, explicitly.} In a field-free, simply connected region three objects are the same thing:
\[
    \underbrace{\Lambda(\lvect{x})}_{\text{gauge function}}
    \;=\;
    \underbrace{\int_{\lvect{x}_0}^{\lvect{x}}\vect{A}\cdot d\lvect{s}}_{\text{line integral}}
    \;=\;
    \frac{\hbar c}{e}\,\underbrace{\theta(\lvect{x})}_{\text{phase of }\psi\text{ relative to }\psi_0}.
\]
\begin{itemize}
    \item \emph{$\Lambda\leftrightarrow$ line integral.} $\vect{A} = \grad\Lambda$ is inverted by the fundamental theorem of calculus, $\Lambda(\lvect{x}) - \Lambda(\lvect{x}_0) = \int_{\lvect{x}_0}^{\lvect{x}}\grad\Lambda\cdot d\lvect{s} = \int_{\lvect{x}_0}^{\lvect{x}}\vect{A}\cdot d\lvect{s}$. Path independence ($\curl\vect{A} = 0$ plus simple connectivity) is exactly what makes this a function of $\lvect{x}$ alone; the base point $\lvect{x}_0$ is the additive constant in $\Lambda$.
    \item \emph{$\Lambda\leftrightarrow$ phase.} The gauge transformation multiplies $\psi$ by $e^{ie\Lambda/\hbar c}$, so the local phase is $\theta(\lvect{x}) = \frac{e}{\hbar c}\Lambda(\lvect{x})$: each step $d\lvect{s}$ adds $d\theta = \frac{e}{\hbar c}\vect{A}\cdot d\lvect{s}$.
    \item \emph{Why it is unobservable here.} The phase difference between two points, $\theta(\lvect{x}_2) - \theta(\lvect{x}_1) = \frac{e}{\hbar c}\int_{\lvect{x}_1}^{\lvect{x}_2}\vect{A}\cdot d\lvect{s}$, is the same for every path, so two beams arriving at a point carry the \emph{same} extra phase and their interference is unchanged. Equivalently, $\vect{A}$ shifts the canonical momentum by $\hbar\grad\theta = \frac ec\vect{A}$, exactly cancelling the $-\frac ec\vect{A}$ in $\bm{\Pi}$, so $\bm{\Pi}\psi = e^{i\theta}\hat{\lvect{p}}\psi_0$.
\end{itemize}
Around a hole the first item fails: the line integral depends on which side of the hole the path passes, $\Lambda$ becomes multivalued, and the phase difference between the two sides is the loop integral, $\frac{e}{\hbar c}\Phi$. That is the Aharonov--Bohm effect.

\subsection{Loop Integrals Are Gauge Invariant Modulo \texorpdfstring{$2\pi$}{2 pi}}

The previous subsection showed that in a simply connected field-free region every closed loop integral of $\vect{A}$ vanishes. Around a hole it need not, so we now ask what freedom a gauge transformation has to change it. Under $\vect{A}\to\vect{A} + \grad\Lambda$, the integral of $\vect{A}$ around a closed loop $C$ changes by the integral of a gradient, which is the change of $\Lambda$ between the endpoints:
\[
    \frac{e}{\hbar c}\oint_C\vect{A}\cdot d\lvect{s}\ \to\ \frac{e}{\hbar c}\oint_C\vect{A}\cdot d\lvect{s} + \frac{e}{\hbar c}\Delta\Lambda,
    \qquad
    \Delta\Lambda = \Lambda(\text{end}) - \Lambda(\text{start}).
\]

\begin{fact}[Why ``the curl of a gradient vanishes'' is not enough]
Stokes' theorem gives $\oint_C\grad\Lambda\cdot d\lvect{s} = \int_S(\curl\grad\Lambda)\cdot d\lvect{a} = 0$ only if $\Lambda$ is smooth on a surface $S$ bounded by $C$. Around a hole there may be no such surface. In the plane with the origin removed,
\[
    \Lambda = \frac{\Phi}{2\pi}\varphi,
    \qquad
    \grad\Lambda = \frac{\Phi}{2\pi r}\hat{\bm{\varphi}},
    \qquad
    \curl\grad\Lambda = 0\ (r\neq0),
    \qquad
    \oint\grad\Lambda\cdot d\lvect{s} = \Phi .
\]
The correct condition is that $\Lambda$ be \emph{single-valued along the loop}.
\end{fact}

The gauge function itself need not be single-valued: what must be single-valued is the transformed wavefunction, i.e.\ the factor $e^{ie\Lambda/\hbar c}$. Going once around the loop this requires
\[
    \frac{e\,\Delta\Lambda}{\hbar c} = 2\pi N,
    \quad
    N\in\mathbb{Z}
    \qquad\Longleftrightarrow\qquad
    \Delta\Lambda = N\frac{2\pi\hbar c}{e} = \pm N\Phi_0 .
\]

\begin{theorem}[Gauge invariance of loop phases]
Under any allowed (single-valued) gauge transformation,
\[
    \frac{e}{\hbar c}\oint_C\vect{A}\cdot d\lvect{s}\ \to\ \frac{e}{\hbar c}\oint_C\vect{A}\cdot d\lvect{s} + 2\pi N .
\]
The loop phase is defined modulo $2\pi$, and the phase factor $\exp\bigl(\frac{ie}{\hbar c}\oint_C\vect{A}\cdot d\lvect{s}\bigr)$ is gauge invariant. An allowed gauge transformation can add or remove exactly $N$ flux quanta through a hole.
\end{theorem}

\subsection{The Aharonov--Bohm Effect}

An impenetrable cylinder contains flux $\Phi$; outside, $\vect{B} = 0$ but $\vect{A}\neq0$. The field-free region is now the plane with a hole, which is not simply connected. Particles travel from a source $A$ to a point $\lvect{x}$ on a screen, either above or below the cylinder (Figure~\ref{fig:ab-paths}).

Each path carries the phase factor of Section~\ref{sec:field-free}, but now the two line integrals differ: the two paths together form a loop around the hole, and Stokes' theorem applied to a surface that includes the solenoid gives the enclosed flux,
\[
    \chi_{\text{above}}(\lvect{x}) - \chi_{\text{below}}(\lvect{x}) = \oint\vect{A}\cdot d\lvect{s} = \Phi .
\]
The function $\chi$ is multivalued: locally $\vect{A} = \grad\chi$ everywhere, but globally the gauge function needed to remove $\vect{A}$ jumps by $\Phi$ around the hole.

\begin{formula}[Aharonov--Bohm phase]
\[
    \Delta\phi_{AB} = \frac{e}{\hbar c}\oint\vect{A}\cdot d\lvect{s} = \frac{e}{\hbar c}\Phi = \pm2\pi\frac{\Phi}{\Phi_0}.
\]
\end{formula}

It does not depend on where on the screen $\lvect{x}$ is: moving $\lvect{x}$ changes both path ends the same way while the enclosed flux stays $\Phi$. So the total relative phase is the ordinary double-slit phase plus a uniform shift,
\[
    I(x)\propto1 + \cos\Bigl(\frac{kd\,x}{L} + 2\pi\frac{\Phi}{\Phi_0}\Bigr),
\]
and the flux slides the fringe pattern rigidly, one fringe per flux quantum. The particle never enters a region with nonzero $\vect{B}$, and the Lorentz force on it vanishes everywhere it can be, yet the pattern depends on $\Phi$.

\paragraph{Periodicity in the flux.} For $\Phi = N\Phi_0$ the vector potential outside is
\[
    \vect{A} = \frac{N\Phi_0}{2\pi r}\hat{\bm{\varphi}} = \grad\Lambda,
    \qquad
    \Lambda = \frac{N\Phi_0}{2\pi}\varphi,
    \qquad
    e^{ie\Lambda/\hbar c} = e^{\pm iN\varphi},
\]
which is single-valued. So an allowed gauge transformation removes $\vect{A}$ entirely, and every observable is the same as with no flux. For non-integer $\Phi/\Phi_0$ the required phase factor is multivalued, no allowed gauge transformation removes $\vect{A}$, and the effect is physical. Physics depends on $\Phi$ only modulo $\Phi_0$.

\paragraph{Bound-state version: a ring around a solenoid.} Confine the particle to a thin ring of radius $R$ enclosing flux $\Phi$ (Figure~\ref{fig:ring}). On the ring, $\vect{A} = \frac{\Phi}{2\pi R}\hat{\bm{\varphi}}$ and $\hat{p}_\varphi = -\frac{i\hbar}{R}\partial_\varphi$, so
\[
    \hat{H} = \frac{1}{2m}\Bigl(-\frac{i\hbar}{R}\partial_\varphi - \frac{e\Phi}{2\pi Rc}\Bigr)^2 = \frac{\hbar^2}{2mR^2}\Bigl(-i\partial_\varphi - \frac{e\Phi}{2\pi\hbar c}\Bigr)^2 .
\]
Single-valuedness requires $\psi_n = e^{in\varphi}/\sqrt{2\pi}$ with integer $n$.

\begin{formula}[Spectrum of a ring threaded by flux]
\[
    E_n(\Phi) = \frac{\hbar^2}{2mR^2}\Bigl(n - \frac{e\Phi}{2\pi\hbar c}\Bigr)^2,
    \qquad
    n\in\mathbb{Z}.
\]
\end{formula}

\centering

\end{axis}

\caption{Energy levels of a ring threaded by flux (for $e>0$). The set of levels is periodic in $\Phi$ with period $\Phi_0$, but each individual level moves: following one as $\Phi$ increases by $\Phi_0$, $n$ changes by one (spectral flow).}
\end{figure}

\begin{itemize}
    \item The spectrum depends on $\Phi$ although the particle never touches the field.
    \item The \emph{set} of energies is periodic with period $\Phi_0$: shifting $\Phi\to\Phi+\Phi_0$ maps the spectrum to itself with $n\to n\pm1$. This is the bound-state face of the gauge argument above.
    \item Individual levels are not periodic. Following one state continuously as $\Phi$ increases by $\Phi_0$, it ends in a \emph{different} state, $n\to n\pm1$ (spectral flow). This is the mechanism of Laughlin's argument for the quantum Hall effect: threading one flux quantum returns the Hamiltonian to itself but moves the occupied states over by one.
    \item The ground state carries a persistent current $I = -c\,\partial E/\partial\Phi$ that oscillates in $\Phi$ with period $\Phi_0$.
\end{itemize}

\begin{example}[Can the vector potential on the ring be gauged away?]
Locally yes: $\vect{A} = \frac{\Phi}{2\pi R}\hat{\bm{\varphi}} = \grad\Lambda$ with $\Lambda = \Phi\varphi/2\pi$. Globally only if $e^{ie\Lambda/\hbar c} = e^{ie\Phi\varphi/2\pi\hbar c}$ is single-valued, i.e.\ $\Phi = N\Phi_0$. Otherwise the transformation violates single-valuedness, and the flux has observable effects, here the shifted spectrum. Equivalently, one can gauge $\vect{A}$ away at the cost of a twisted boundary condition $\psi(\varphi+2\pi) = e^{-ie\Phi/\hbar c}\psi(\varphi)$: the physics moves from the Hamiltonian into the boundary condition but does not disappear.
\end{example}

\subsection{The Action Picture: Space-Time Loops}

The classical action of a charge contains the potentials through
\[
    S[\lvect{r}(t)] = \int_1^2\Bigl[\frac ec\vect{A}(\lvect{r},t)\cdot d\lvect{r} - e\phi(\lvect{r},t)\,dt\Bigr] + (\text{kinetic terms}),
\]
and enters quantum mechanics through Feynman's amplitude $e^{iS/\hbar}$. Under a gauge transformation the potential terms change by a total derivative,
\[
    S\to S + \frac ec\int_1^2\bigl(\grad\Lambda\cdot d\lvect{r} + \partial_t\Lambda\,dt\bigr) = S + \frac ec\bigl[\Lambda(\lvect{r}_2,t_2) - \Lambda(\lvect{r}_1,t_1)\bigr],
\]
which depends only on the endpoints and therefore cancels between two trajectories with common endpoints. The interference phase between trajectories $\lvect{r}'(t)$ and $\lvect{r}''(t)$ is the action difference, a closed loop in space-time (forward along $\lvect{r}'$, back along $\lvect{r}''$):

\end{tikzpicture}
\caption{Two trajectories with common endpoints form a closed loop in space-time. The gauge-dependent endpoint terms cancel, leaving $\theta = \frac{e}{\hbar c}\oint(\vect{A}\cdot d\lvect{r} - c\phi\,dt)$. A purely spatial loop gives the magnetic AB phase; a loop separated in time by a voltage gives the electric one.}
\label{fig:spacetime-loop}
\end{figure}

\begin{formula}[Gauge-invariant space-time loop phase]
\[
    \theta = \frac{e}{\hbar c}\oint\bigl(\vect{A}\cdot d\lvect{r} - c\,\phi\,dt\bigr).
\]
\end{formula}

\begin{itemize}
    \item \textbf{Static magnetic case} ($\phi = 0$, $\vect{A}$ time-independent): $\theta = \frac{e}{\hbar c}\Phi_B$, the Aharonov--Bohm phase. It is geometric: independent of how fast the particle moves along the loop.
    \item \textbf{Electric, non-stationary case} ($\vect{B} = 0$, time-dependent $\vect{E}$): choose $\vect{A} = 0$. Then $\theta = -\frac e\hbar\oint\phi\,dt = -\frac e\hbar\int V(t)\,dt$, with $V(t)$ the voltage between the paths. This is exactly the cage experiment above: the electric (``Minkowski-rotated'') Aharonov--Bohm effect.
    \item \textbf{Integer pulses.} A voltage pulse with $\int V\,dt = nh/e$ gives $\theta = 2\pi n$ and leaves no interference signature, the electric analogue of an integer number of flux quanta. Such pulses transfer charge $ne$ cleanly (``levitons'').
\end{itemize}

\subsection{Summary and the Bridge to Berry Phase}

\begin{fact}[Potentials as phases]
\leavevmode
\begin{enumerate}
    \item A position-dependent phase shifts the momentum by $\hbar\grad\theta$; a time-dependent one shifts the energy.
    \item Gauge invariance of position, kinetic momentum and normalization forces $\psi\to e^{ie\Lambda/\hbar c}\psi$ under $\vect{A}\to\vect{A}+\grad\Lambda$.
    \item Loop phases $\frac{e}{\hbar c}\oint\vect{A}\cdot d\lvect{s}$ are gauge invariant modulo $2\pi$, because single-valuedness of $\psi$ allows $\Lambda$ to jump by $N\Phi_0$ around a hole.
    \item In a field-free, simply connected region $\vect{A}$ is pure gauge and unobservable. Around a hole it is locally pure gauge but globally not, and the enclosed flux is observable modulo $\Phi_0$: the Aharonov--Bohm effect.
    \item The general gauge-invariant object is the space-time loop phase $\frac{e}{\hbar c}\oint(\vect{A}\cdot d\lvect{r} - c\phi\,dt)$.
\end{enumerate}
\end{fact}

Every item has a Berry-phase counterpart, with the real-space potential replaced by a potential on the space of a Hamiltonian's parameters $\lvect{R}$:

The next step is the magnetic monopole, where no single smooth $\vect{A}$ exists on a sphere surrounding it, and Dirac's quantization condition follows from requiring the loop phase to be well defined.

\begin{fact}[Key results of this chapter]
\leavevmode
\begin{itemize}
\item $U(t,t_0)$ is unitary; $\hat{H} = i\hbar(\partial_tU)U^\dagger$; $U = e^{-i\hat{H}t/\hbar}$ for constant $\hat{H}$, time-ordered exponential in general.

\item Coherent states $\hat{a}|\alpha\rangle = \alpha|\alpha\rangle$: Poisson number distribution, classical motion, minimum uncertainty at all times.
\item Squeezed states trade noise between quadratures; the EM field is a set of oscillators.
\item Potentials enter only as phases: $\psi\to e^{ie\Lambda/\hbar c}\psi$ under gauge transformations, and loop phases $\frac{e}{\hbar c}\oint\vect{A}\cdot d\lvect{s}$ are observable modulo $2\pi$ (Aharonov--Bohm).
\end{itemize}
\end{fact}
