\chapter{Electrons in a Periodic Potential}

An electron in a crystal moves in a potential with the periodicity of the lattice, and that single fact --- before any approximation whatsoever --- already fixes the form of the stationary states. This chapter begins with Bloch's theorem and its immediate consequences: the labelling of states by a crystal momentum and a band index, the notion of a Fermi surface, and the density of levels. It then treats the two complementary limits in which the problem becomes tractable. In the \emph{nearly free electron} limit the periodic potential is a weak perturbation on plane waves, and its only dramatic effect is to open gaps at the Bragg planes. In the \emph{tight-binding} limit we start instead from atomic orbitals localized on each site and let a small overlap between neighbours broaden the discrete atomic levels into bands. We then collect the lattice Fourier identities that both pictures rest on, and close with the semiclassical dynamics of Bloch wavepackets.

A word on notation. Since bold capitals denote vectors throughout these notes, we write the electron position as $\vect{x}$, direct lattice vectors as $\vect{R}$, reciprocal lattice vectors as $\vect{G}$, and wavevectors as $\vect{k}$ and $\vect{q}$.

% ============================================================
\section{Bloch's Theorem}
% ============================================================

Throughout we make two assumptions: that the crystal is perfect, i.e.\ a strictly periodic array of atoms, and that the atomic nuclei are static. The potential is then periodic,
\begin{equation}
    U(\vect{x}+\vect{R}) = U(\vect{x}),
\end{equation}
for every lattice vector $\vect{R}$, and the one-electron Schrödinger equation
\begin{equation}
    \hat{H}\psi = \left( -\frac{\hbar^2}{2m}\nabla^2 + U(\vect{x}) \right)\psi = E\psi
\end{equation}
has a Hamiltonian that is periodic in $\vect{x}$.

\begin{theorem}[Bloch's theorem]
The eigenstates $\psi$ of the one-electron Hamiltonian $H = -\hbar^2\nabla^2/2m + U(\vect{x})$ with $U$ periodic can be chosen to have the form
\begin{equation}
    \psi_{n\vect{k}}(\vect{x}) = e^{i\vect{k}\cdot\vect{x}}\, u_{n\vect{k}}(\vect{x}),
    \qquad u_{n\vect{k}}(\vect{x}+\vect{R}) = u_{n\vect{k}}(\vect{x}),
\end{equation}
equivalently
\begin{equation}
    \psi_{n\vect{k}}(\vect{x}+\vect{R}) = e^{i\vect{k}\cdot\vect{R}}\, \psi_{n\vect{k}}(\vect{x}).
    \label{eq:bloch-condition}
\end{equation}
\end{theorem}

\subsection{First Proof: Translation Operators}

Define the translation operator $T_{\vect{R}}$ by $T_{\vect{R}} f(\vect{x}) \equiv f(\vect{x}+\vect{R})$. Because the Hamiltonian is periodic,
\begin{equation}
    T_{\vect{R}} \hat{H} \psi = \hat{H}(\vect{x}+\vect{R})\psi(\vect{x}+\vect{R}) = H(\vect{x})\psi(\vect{x}+\vect{R}) = \hat{H} T_{\vect{R}}\psi,
\end{equation}
so $T_{\vect{R}}$ commutes with $H$. The translations also commute among themselves,
\begin{equation}
    T_{\vect{R}} T_{\vect{R}'} \psi(\vect{x}) = \psi(\vect{x}+\vect{R}+\vect{R}') = T_{\vect{R}'} T_{\vect{R}} \psi = T_{\vect{R}+\vect{R}'}\psi.
\end{equation}
Hence $\{T_{\vect{R}}\}$ and $H$ form a set of commuting operators and can be simultaneously diagonalized:
\begin{equation}
    \hat{H}\psi = E\psi, \qquad T_{\vect{R}}\psi = C(\vect{R})\psi,
\end{equation}
where the multiplicative property $C(\vect{R}+\vect{R}') = C(\vect{R})C(\vect{R}')$ lets us write the eigenvalues on the primitive vectors as
\begin{equation}
    C(\vect{a}_i) = \exp(2\pi i x_i).
\end{equation}
For an arbitrary lattice vector $\vect{R} = n_1\vect{a}_1 + n_2\vect{a}_2 + n_3\vect{a}_3$,
\begin{equation}
    C(\vect{R}) = C(\vect{a}_1)^{n_1} C(\vect{a}_2)^{n_2} C(\vect{a}_3)^{n_3}
    = \exp\bigl(2\pi i x_1 n_1 + 2\pi i x_2 n_2 + 2\pi i x_3 n_3\bigr).
\end{equation}
Now take $\vect{k} = x_1\vect{b}_1 + x_2\vect{b}_2 + x_3\vect{b}_3$, where the $\vect{b}_i$ are the reciprocal primitive vectors satisfying $\vect{b}_i\cdot\vect{a}_j = 2\pi\delta_{ij}$. Then
\begin{equation}
    \exp(i\vect{k}\cdot\vect{R}) = \exp\bigl(i(x_1\vect{b}_1 + x_2\vect{b}_2 + x_3\vect{b}_3)\cdot(n_1\vect{a}_1 + n_2\vect{a}_2 + n_3\vect{a}_3)\bigr)
    = \exp\bigl(i(2\pi x_1 n_1 + 2\pi x_2 n_2 + 2\pi x_3 n_3)\bigr) = C(\vect{R}),
\end{equation}
so that $T_{\vect{R}}\psi = \psi(\vect{x}+\vect{R}) = C(\vect{R})\psi = e^{i\vect{k}\cdot\vect{R}}\psi$, which is exactly~\eqref{eq:bloch-condition}.

\subsection{Born--von Karman Boundary Conditions}

\begin{definition}[Born--von Karman boundary condition]
\begin{equation}
    \psi(\vect{x} + N_i \vect{a}_i) = \psi(\vect{x}), \qquad i = 1,2,3,
\end{equation}
where $N = N_1 N_2 N_3$ is the total number of primitive cells and $N_i$ is the length of the system in unit cells along a particular direction.
\end{definition}

This constrains $\vect{k}$ to be real and quantizes its allowed values. Applying the Bloch condition,
\begin{equation}
    \psi_{n\vect{k}}(\vect{x}+N_i\vect{a}_i) = \exp(iN_i \vect{k}\cdot\vect{a}_i)\, \psi_{n\vect{k}}(\vect{x}) = \psi(\vect{x})
    \;\Longrightarrow\; \exp(iN_i\vect{k}\cdot\vect{a}_i) = 1
    \;\Longrightarrow\; \vect{k}\cdot\vect{a}_i = \frac{2\pi m}{N_i}, \quad m \in \mathbb{Z}.
\end{equation}
Writing $\vect{k} = x_1\vect{b}_1 + x_2\vect{b}_2 + x_3\vect{b}_3$ gives $2\pi x_i = 2\pi m/N_i$, i.e.\ $x_i = m/N_i$.

\begin{formula}[Allowed wavevectors]
\begin{equation}
    \vect{k} = \sum_{i=1}^{3} \frac{m_i}{N_i}\,\vect{b}_i, \qquad m_i \in \mathbb{Z}.
\end{equation}
\end{formula}

\subsection{Volume per Allowed Wavevector}

The allowed $\vect{k}$ form a lattice whose primitive cell is the parallelepiped with edges $\vect{b}_i/N_i$, so the volume per $\vect{k}$-value is
\begin{equation}
    \Delta k = \frac{\vect{b}_1}{N_1}\cdot\left( \frac{\vect{b}_2}{N_2} \times \frac{\vect{b}_3}{N_3} \right)
    = \frac{1}{N}\,\vect{b}_1\cdot(\vect{b}_2\times\vect{b}_3).
\end{equation}

\begin{fact}
The number of allowed wavevectors in a primitive cell of the reciprocal lattice equals the number of sites in the crystal.
\end{fact}

Since the volume of the reciprocal lattice primitive cell is $(2\pi)^3/v$ with $v = V/N$,
\begin{equation}
    \Delta k = \frac{\text{primitive cell volume}}{N} = \frac{(2\pi)^3}{V}.
\end{equation}

\subsection{Second Proof: Fourier Expansion}

The second proof is more laborious but produces the central equation on which the nearly free electron model is built. By Fourier transform, any function obeying the boundary conditions can be written
\begin{equation}
    \psi(\vect{x}) = \sum_{\vect{q}} c_{\vect{q}}\, e^{i\vect{q}\cdot\vect{x}},
\end{equation}
where $\vect{q}$ runs over the set of BVK wavevectors. The potential, being periodic in the lattice, has a plane-wave expansion containing only reciprocal lattice vectors:
\begin{equation}
    U(\vect{x}) = \sum_{\vect{G}} U_{\vect{G}}\, e^{i\vect{G}\cdot\vect{x}},
    \qquad
    U_{\vect{G}} = \frac{1}{v}\int_{\text{cell}} d\vect{x}\; e^{-i\vect{G}\cdot\vect{x}}\, U(\vect{x}),
\end{equation}
where it suffices to integrate over one cell since $U$ is periodic. We are free to make the constant term vanish,
\begin{equation}
    U_0 = \frac{1}{v}\int_{\text{cell}} d\vect{x}\; U(\vect{x}) = 0.
\end{equation}
Since $U(\vect{x})$ is real, $U_{-\vect{G}} = U^*_{\vect{G}}$; and if the lattice has inversion symmetry, $U(\vect{x}) = U(-\vect{x})$ for a suitable choice of origin, then $U_{\vect{G}} = U^*_{-\vect{G}}$ so that $U_{\vect{G}}$ is real.

Substituting both expansions into the Schrödinger equation, the kinetic term gives
\begin{equation}
    -\frac{\hbar^2}{2m}\nabla^2 \psi = \sum_{\vect{q}} \frac{\hbar^2}{2m} q^2 c_{\vect{q}}\, e^{i\vect{q}\cdot\vect{x}},
\end{equation}
while the potential term gives
\begin{equation}
    U\psi = \left( \sum_{\vect{G}} U_{\vect{G}} e^{i\vect{G}\cdot\vect{x}} \right)\left( \sum_{\vect{q}} c_{\vect{q}} e^{i\vect{q}\cdot\vect{x}} \right)
    = \sum_{\vect{G},\vect{q}} U_{\vect{G}} c_{\vect{q}}\, e^{i(\vect{G}+\vect{q})\cdot\vect{x}}
    = \sum_{\vect{G},\vect{q}'} U_{\vect{G}} c_{\vect{q}'-\vect{G}}\, e^{i\vect{q}'\cdot\vect{x}}.
\end{equation}

\begin{fact}
The potential $U$ only couples a wavevector $\vect{q}$ to wavevectors differing from it by a reciprocal lattice vector, $\vect{q}+\vect{G}$. Everything that follows is a rigorous statement of this fact.
\end{fact}

Collecting terms with a common exponential,
\begin{equation}
    \sum_{\vect{q}} e^{i\vect{q}\cdot\vect{x}} \left\{ \left( \frac{\hbar^2}{2m}q^2 - \varepsilon \right) c_{\vect{q}} + \sum_{\vect{G}'} U_{\vect{G}'}\, c_{\vect{q}-\vect{G}'} \right\} = 0,
\end{equation}
and since the plane waves satisfying the Born--von Karman conditions are orthogonal, each brace must vanish separately:
\begin{equation}
    \left( \frac{\hbar^2}{2m}q^2 - \varepsilon \right) c_{\vect{q}} + \sum_{\vect{G}'} U_{\vect{G}'}\, c_{\vect{q}-\vect{G}'} = 0
    \qquad \text{for all allowed } \vect{q}.
\end{equation}
Rewriting with $\vect{q} = \vect{k}-\vect{G}$, so that $\vect{k}$ lies in the first Brillouin zone, and then shifting $\vect{G}' \to \vect{G}'-\vect{G}$, gives the central equation.

\begin{formula}[Central equation]
\begin{equation}
    \left( \frac{\hbar^2}{2m}(\vect{k}-\vect{G})^2 - \varepsilon \right) c_{\vect{k}-\vect{G}}
    + \sum_{\vect{G}'} U_{\vect{G}'-\vect{G}}\, c_{\vect{k}-\vect{G}'} = 0.
    \label{eq:bloch-central}
\end{equation}
\end{formula}

This relates only the coefficients $c_{\vect{k}}, c_{\vect{k}-\vect{G}}, c_{\vect{k}-\vect{G}'}, \ldots$ to one another, so the problem of determining the solution coefficients splits into $N$ independent subproblems, one for each $\vect{k}$ in the first zone. Organizing the vector of coefficients into reciprocal-space blocks,
\begin{equation}
    \vect{\Psi} = \bigl( \underbrace{c_{\vect{k}},\, c_{\vect{k}-\vect{G}},\, c_{\vect{k}-\vect{G}'},\ldots}_{\vect{k}},\;
    \underbrace{c_{\vect{k}'},\, c_{\vect{k}'-\vect{G}},\, c_{\vect{k}'-\vect{G}'},\ldots}_{\vect{k}'},\; \ldots \bigr),
\end{equation}
the Hamiltonian is block diagonal, with one block $\hat{H}_{\vect{k}}$ for each of the $N$ wavevectors.

\begin{formula}[Reduced matrix equation]
\begin{equation}
    \sum_{\vect{G}'} H_{\vect{G}\vect{G}'}(\vect{k})\, c^n_{\vect{G}'} = E_n(\vect{k})\, c^n_{\vect{G}},
    \qquad
    H_{\vect{G}\vect{G}'}(\vect{k}) \equiv \frac{\hbar^2}{2m}|\vect{k}+\vect{G}|^2 \delta_{\vect{G}\vect{G}'} + U_{\vect{G}-\vect{G}'}.
\end{equation}
\end{formula}

There are thus $N$ independent subproblems of the form $\hat{H}_{\vect{k}}\psi_{\vect{k}} = E\psi_{\vect{k}}$, and the solution for a given $\vect{k}$ is
\begin{equation}
    \psi_{\vect{k}}(\vect{x}) = \sum_{\vect{G}} c^n_{\vect{G}}\, e^{i(\vect{k}+\vect{G})\cdot\vect{x}}
    = e^{i\vect{k}\cdot\vect{x}} \underbrace{\sum_{\vect{G}} c^n_{\vect{G}}\, e^{i\vect{G}\cdot\vect{x}}}_{u_{\vect{k}}(\vect{x})}.
\end{equation}
Since $u^n_{\vect{k}}(\vect{x}+\vect{R}) = \sum_{\vect{G}} c^n_{\vect{G}} e^{i\vect{G}\cdot(\vect{x}+\vect{R})} = \sum_{\vect{G}} c^n_{\vect{G}} e^{i\vect{G}\cdot\vect{x}} = u^n_{\vect{k}}(\vect{x})$, this proves Bloch's theorem a second time.

\section{Consequences of Bloch's Theorem}
% ============================================================

\subsection{General Remarks}

\begin{enumerate}
    \item The wavevector $\vect{k}$ plays the same role as in the free-electron case: it is the quantum number characteristic of the translational symmetry of the periodic potential. But $\hbar\vect{k}$ is \emph{not} the momentum, because the Hamiltonian no longer has full translational symmetry. Indeed, with $\hat{\vect{p}} = \frac{\hbar}{i}\nabla$,
    \begin{equation}
        \frac{\hbar}{i}\nabla \psi_{n\vect{k}} = \frac{\hbar}{i}\nabla\bigl( e^{i\vect{k}\cdot\vect{x}} u_{n\vect{k}}(\vect{x}) \bigr)
        = \hbar\vect{k}\,\psi_{n\vect{k}} + e^{i\vect{k}\cdot\vect{x}}\,\frac{\hbar}{i}\nabla u_{n\vect{k}}(\vect{x}),
    \end{equation}
    which is not proportional to $\psi_{n\vect{k}}$. We call $\hbar\vect{k}$ the \emph{crystal momentum}.
    \item The wavevector in Bloch's theorem can always be confined to the first Brillouin zone. Any $\vect{k}'$ outside it can be written $\vect{k}' = \vect{k}+\vect{G}$ with $\vect{k}$ inside, and then
    \begin{equation}
        \psi(\vect{x}+\vect{R}) = e^{i\vect{k}'\cdot\vect{R}}\psi(\vect{x}) = e^{i(\vect{k}+\vect{G})\cdot\vect{R}}\psi(\vect{x}) = e^{i\vect{k}\cdot\vect{R}}\psi(\vect{x}),
    \end{equation}
    since $e^{i\vect{G}\cdot\vect{R}} = 1$.
    \item For a given $\vect{k}$ there are many solutions of the Schrödinger equation
    \begin{equation}
        H_{\vect{k}} u_{\vect{k}}(\vect{x}) = \left( \frac{\hbar^2}{2m}\Bigl(\frac{1}{i}\nabla + \vect{k}\Bigr)^{\!2} + U(\vect{x}) \right) u_{\vect{k}}(\vect{x}),
    \end{equation}
    a discrete spectrum of eigenvalues $\varepsilon_{n\vect{k}}$ labelled by the \emph{band index} $n$. Each level varies continuously with $\vect{k}$, so we expect a family of continuous functions $\varepsilon_n(\vect{k})$ with the periodicity
    \begin{equation}
        \psi_{n,\vect{k}+\vect{G}} = \psi_{n\vect{k}}(\vect{x}), \qquad \varepsilon_{n,\vect{k}+\vect{G}} = \varepsilon_{n,\vect{k}}.
    \end{equation}
    \item The mean electron group velocity in band $n$ at wavevector $\vect{k}$ is $\vect{v}_n(\vect{k}) = \frac{1}{\hbar}\nabla_{\vect{k}}\varepsilon_n(\vect{k})$ --- derived below in the semiclassical section. Remarkably, an electron can move forever in a perfect periodic potential without any degradation of its mean velocity.
\end{enumerate}

\begin{definition}[Insulator case]
One or more bands are completely filled and the rest are empty. The band gap is
\begin{equation}
    \Delta E = E(\text{lowest unoccupied state}) - E(\text{highest occupied state}).
\end{equation}
Since the number of available levels per band is $2N_{\vect{k}} = 2N_{\text{primitive cells}}$, this case can only occur if the number of electrons per unit cell is even. If $\Delta E \gg k_B T$ the material is an \emph{insulator}; if $\Delta E \sim k_B T$ it is a \emph{semiconductor}.
\end{definition}

\begin{definition}[Metal case]
One or more bands are partially filled. For each such band there is a surface in $\vect{k}$-space separating occupied from unoccupied levels --- a branch of the \emph{Fermi surface} --- defined by
\begin{equation}
    \varepsilon_n(\vect{k}) = \varepsilon_F.
\end{equation}
The solution of this constraint for each $n$ is a $\vect{k}$-space surface with the periodicity of the reciprocal lattice.
\end{definition}

\subsection{Density of Levels}

\begin{definition}[Density of levels]
\begin{equation}
    g(\varepsilon) = \sum_n g_n(\varepsilon),
    \qquad
    g_n(\varepsilon) = \int \frac{d\vect{k}}{4\pi^3}\,\delta\bigl(\varepsilon - \varepsilon_n(\vect{k})\bigr),
    \qquad [g(\varepsilon)] = \frac{1}{[E][V]},
\end{equation}
where the integral runs over any primitive cell.
\end{definition}

Equivalently, $g_n(\varepsilon)\,d\varepsilon$ is $2/V$ times the number of allowed wavevectors in the $n$th band with energy between $\varepsilon$ and $\varepsilon+d\varepsilon$:
\begin{equation}
    g_n(\varepsilon)\, d\varepsilon
    = \frac{2}{V}\cdot\frac{\text{volume of the $\vect{k}$-space region containing } \varepsilon \to \varepsilon+d\varepsilon}{\Delta k}
    = \frac{2}{V}\cdot\frac{V}{(2\pi)^3}\int \frac{d\vect{k}}{4\pi^3}\,\delta\bigl(\varepsilon \leq \varepsilon_n(\vect{k}) \leq \varepsilon+d\varepsilon\bigr).
\end{equation}
Geometrically, $g_n(\varepsilon)\,d\varepsilon$ is the volume between the constant-energy surface $S_n(\varepsilon)$ and the neighbouring surface $S_n(\varepsilon+d\varepsilon)$ (Figure~\ref{fig:bloch-dos-shell}):
\begin{equation}
    g_n(\varepsilon) = \int_{S_n(\varepsilon)} \frac{dS}{4\pi^3}\, \delta k(\vect{k}),
\end{equation}
where the perpendicular separation of the two surfaces follows from $\varepsilon + d\varepsilon = \varepsilon + |\nabla\varepsilon_n(\vect{k})|\,\delta k(\vect{k})$, so that
\begin{equation}
    \delta k(\vect{k}) = \frac{d\varepsilon}{|\nabla\varepsilon_n(\vect{k})|}.
\end{equation}

\begin{formula}[Alternate density of states]
\begin{equation}
    g_n(\varepsilon) = \int_{S_n(\varepsilon)} \frac{dS}{4\pi^3}\,\frac{1}{|\nabla\varepsilon_n(\vect{k})|}.
\end{equation}
\end{formula}

\begin{fact}[Van Hove singularities]
Because $\varepsilon_n(\vect{k})$ is periodic in the $\vect{k}$-space primitive cell and bounded above and below, there must be points where $\nabla\varepsilon_n(\vect{k}) = 0$, and at those points $g_n(\varepsilon)$ diverges.
\end{fact}

\section{Electrons in a Weak Periodic Potential}
% ============================================================

The nearly free electron model starts from the Sommerfeld free electron gas and adds a weak perturbing potential. Two reasons make this a sensible description of real materials: conduction electrons cannot get close to the ions and interact strongly with them, since the core electrons already occupy that region; and in the region where conduction electrons are allowed, they screen the field of the positively charged ions, diminishing the total effective potential.

\subsection{Solving the Schrödinger Equation for a Weak Potential}

We begin from the plane-wave ansatz of the second proof of Bloch's theorem,
\begin{equation}
    \psi_{\vect{k}}(\vect{x}) = \sum_{\vect{G}} c_{\vect{k}-\vect{G}}\, e^{i(\vect{k}-\vect{G})\cdot\vect{x}},
\end{equation}
summed over all reciprocal lattice vectors, with the coefficients $c_{\vect{k}-\vect{G}}$ and $\varepsilon(\vect{k})$ fixed by the central equation~\eqref{eq:bloch-central}, which we now write as
\begin{equation}
    \bigl(\varepsilon - \varepsilon^0_{\vect{k}-\vect{G}}\bigr)\, c_{\vect{k}-\vect{G}}
    = \sum_{\vect{G}'} U_{\vect{G}'-\vect{G}}\; c_{\vect{k}-\vect{G}'},
    \qquad \varepsilon^0_{\vect{k}} \equiv \frac{\hbar^2 k^2}{2m}.
    \label{eq:bloch-central-eps}
\end{equation}
For free electrons $U = 0$ and this collapses to
\begin{equation}
    \bigl( \varepsilon^0_{\vect{k}-\vect{G}} - \varepsilon \bigr)\, c_{\vect{k}-\vect{G}} = 0,
\end{equation}
so for every $\vect{G}$ either $\varepsilon^0_{\vect{k}-\vect{G}} = \varepsilon(\vect{k})$ or $c_{\vect{k}-\vect{G}} = 0$; nontriviality demands at least one nonzero coefficient. Two cases follow:
\begin{enumerate}
    \item \emph{Nondegenerate}: $\varepsilon = \varepsilon^0_{\vect{k}-\vect{G}_1}$ with $\psi_{\vect{k}} \propto e^{i(\vect{k}-\vect{G}_1)\cdot\vect{x}}$, a single plane wave.
    \item \emph{Degenerate}: $\varepsilon^0_{\vect{k}-\vect{G}_1} = \cdots = \varepsilon^0_{\vect{k}-\vect{G}_m}$, i.e.\ several reciprocal lattice vectors sit at the same squared distance from the given $\vect{k}$ --- which happens only at certain points. There are then $m$ independent degenerate plane-wave solutions, and any linear combination is again a solution, so one has complete freedom in choosing the coefficients $c_{\vect{k}-\vect{G}}$ for $\vect{G} = \vect{G}_1,\ldots,\vect{G}_m$.
\end{enumerate}

Turning on a weak but nonzero $U_{\vect{G}}$ resolves this freedom. The free-electron parabolas centred on neighbouring reciprocal lattice points intersect, and it is only in the neighbourhood of those intersections --- the region where $|\varepsilon^0_{\vect{k}-\vect{G}_1} - \varepsilon^0_{\vect{k}-\vect{G}}| \sim O(U)$ --- that the potential matters at all (Figure~\ref{fig:bloch-parabolas}).

\subsection{Case 1: The Nondegenerate Case}

Fix $\vect{k}$ and consider a $\vect{G}_1$ such that
\begin{equation}
    \bigl| \varepsilon^0_{\vect{k}-\vect{G}_1} - \varepsilon^0_{\vect{k}-\vect{G}} \bigr| \gg U
    \qquad \text{for all } \vect{G} \neq \vect{G}_1,
\end{equation}
so that one energy level --- typically the smallest --- is isolated from all the others by an amount much larger than $U$. We want a solution that approaches the free-electron result $\varepsilon = \varepsilon^0_{\vect{k}-\vect{G}_1}$, $c_{\vect{k}-\vect{G}} = 0$ ($\vect{G}\neq\vect{G}_1$) as $U \to 0$. Hence $c_{\vect{k}-\vect{G}_1}$ is of order unity while the remaining $c_{\vect{k}-\vect{G}}$ are first order in $U$. Rearranging~\eqref{eq:bloch-central-eps} for a general $\vect{G}$,
\begin{equation}
    c_{\vect{k}-\vect{G}} = \frac{U_{\vect{G}_1-\vect{G}}\, c_{\vect{k}-\vect{G}_1}}{\varepsilon - \varepsilon^0_{\vect{k}-\vect{G}}}
    + \sum_{\vect{G}' \neq \vect{G}_1} \frac{U_{\vect{G}'-\vect{G}}\, c_{\vect{k}-\vect{G}'}}{\varepsilon - \varepsilon^0_{\vect{k}-\vect{G}}}
    = \frac{U_{\vect{G}_1-\vect{G}}\, c_{\vect{k}-\vect{G}_1}}{\varepsilon - \varepsilon^0_{\vect{k}-\vect{G}}} + O(U^2),
\end{equation}
having chosen $U_0 = 0$. Plugging this back into the equation for $c_{\vect{k}-\vect{G}_1}$,
\begin{equation}
    \bigl(\varepsilon - \varepsilon^0_{\vect{k}-\vect{G}_1}\bigr) c_{\vect{k}-\vect{G}_1}
    = \sum_{\vect{G}} \frac{|U_{\vect{G}-\vect{G}_1}|^2}{\varepsilon - \varepsilon^0_{\vect{k}-\vect{G}}}\, c_{\vect{k}-\vect{G}_1} + O(U^3).
\end{equation}
Replacing $\varepsilon \approx \varepsilon^0_{\vect{k}-\vect{G}_1}$ in the denominator --- legitimate because $\varepsilon = \varepsilon^0_{\vect{k}-\vect{G}_1} + \Delta E$ with $\Delta E = O(U^2)$ --- gives the perturbed level.

\begin{formula}[Nondegenerate weakly perturbed energy levels]
\begin{equation}
    \varepsilon = \varepsilon^0_{\vect{k}-\vect{G}_1}
    + \sum_{\vect{G}} \frac{|U_{\vect{G}-\vect{G}_1}|^2}{\varepsilon^0_{\vect{k}-\vect{G}_1} - \varepsilon^0_{\vect{k}-\vect{G}}} + O(U^3).
\end{equation}
\end{formula}

The intuition is that the exact state is still mostly a single plane wave; the periodic potential merely admixes small pieces of the other $\vect{k}-\vect{G}$ plane waves, and the resulting energy shift is small.

\subsection{Case 2: The Degenerate Case}

Now suppose there are $m$ reciprocal lattice vectors $\vect{G}_1,\ldots,\vect{G}_m$ whose free-electron energies $\varepsilon^0_{\vect{k}-\vect{G}_1},\ldots,\varepsilon^0_{\vect{k}-\vect{G}_m}$ all lie within $O(U)$ of each other, but are far from every other level on the scale of $U$:
\begin{equation}
    \bigl|\varepsilon^0_{\vect{k}-\vect{G}} - \varepsilon^0_{\vect{k}-\vect{G}_i}\bigr| \gg U,
    \qquad i = 1,\ldots,m, \quad \vect{G} \neq \vect{G}_1,\ldots,\vect{G}_m.
\end{equation}
There is one constraint for each member of the degenerate set, so we get $m$ coupled equations. Separating the large coefficients from the small ones,
\begin{equation}
    \bigl(\varepsilon - \varepsilon^0_{\vect{k}-\vect{G}_i}\bigr) c_{\vect{k}-\vect{G}_i}
    = \underbrace{\sum_{j=1}^{m} U_{\vect{G}_j-\vect{G}_i}\, c_{\vect{k}-\vect{G}_j}}_{\text{large coefficients}}
    + \sum_{\vect{G} \neq \vect{G}_1\ldots\vect{G}_m} U_{\vect{G}-\vect{G}_i}\, c_{\vect{k}-\vect{G}},
    \qquad i = 1,\ldots,m.
    \label{eq:bloch-degen}
\end{equation}
For the remaining $\vect{G} \neq \vect{G}_1,\ldots,\vect{G}_m$ the same equation gives
\begin{equation}
    c_{\vect{k}-\vect{G}} = \frac{1}{\varepsilon - \varepsilon^0_{\vect{k}-\vect{G}}}
    \left( \sum_{j=1}^{m} U_{\vect{G}_j-\vect{G}}\, c_{\vect{k}-\vect{G}_j}
    + \sum_{\vect{G}' \neq \vect{G}_1\ldots\vect{G}_m} U_{\vect{G}'-\vect{G}}\, c_{\vect{k}-\vect{G}'} \right)
    = \frac{1}{\varepsilon - \varepsilon^0_{\vect{k}-\vect{G}}} \sum_{j=1}^{m} U_{\vect{G}_j-\vect{G}}\, c_{\vect{k}-\vect{G}_j} + O(U^2).
\end{equation}
Substituting this back into~\eqref{eq:bloch-degen},
\begin{equation}
    \bigl(\varepsilon - \varepsilon^0_{\vect{k}-\vect{G}_i}\bigr) c_{\vect{k}-\vect{G}_i}
    = \sum_{j=1}^{m} U_{\vect{G}_j-\vect{G}_i}\, c_{\vect{k}-\vect{G}_j}
    + \sum_{j=1}^{m} \left( \sum_{\vect{G} \neq \vect{G}_1\ldots\vect{G}_m}
    \frac{U_{\vect{G}-\vect{G}_i} U_{\vect{G}_j-\vect{G}}}{\varepsilon - \varepsilon^0_{\vect{k}-\vect{G}}} \right) c_{\vect{k}-\vect{G}_j} + O(U^3),
\end{equation}
so that to leading order the second sum drops out.

\begin{formula}[System of equations for $\varepsilon(\vect{k}-\vect{G})$]
\begin{equation}
    \bigl(\varepsilon - \varepsilon^0_{\vect{k}-\vect{G}_i}\bigr)\, c_{\vect{k}-\vect{G}_i}
    = \sum_{j=1}^{m} U_{\vect{G}_j-\vect{G}_i}\, c_{\vect{k}-\vect{G}_j}.
    \label{eq:bloch-degen-system}
\end{equation}
\end{formula}

\subsection{The Bragg Plane Condition}

When are two free-electron levels degenerate in the required sense? Suppose $\varepsilon^0_{\vect{q}} \approx \varepsilon^0_{\vect{q}-\vect{G}}$ while $|\varepsilon^0_{\vect{q}} - \varepsilon^0_{\vect{q}-\vect{G}'}| \gg U$ for all other $\vect{G}' \neq \vect{G}, 0$. The exact condition $\varepsilon^0_{\vect{q}} = \varepsilon^0_{\vect{q}-\vect{G}}$ holds precisely when $|\vect{q}| = |\vect{q}-\vect{G}|$, i.e.\ when $\vect{q}$ lies on the plane bisecting $\vect{G}$ --- the \emph{Bragg plane} (Figure~\ref{fig:bloch-bragg-plane}).

A weak periodic potential has its most prominent effects only at those free-electron energy levels whose $\vect{k}$-values lie close to the planes that cause Bragg reflections.
\end{fact}

\subsection{Energy Levels Near a Single Bragg Plane}

\begin{example}[Two nearly degenerate levels]
Take $m = 2$, i.e.\ two electron levels within $U$ of each other. Then~\eqref{eq:bloch-degen-system} reduces to the pair
\begin{align}
    \bigl(\varepsilon - \varepsilon^0_{\vect{k}-\vect{G}_1}\bigr) c_{\vect{k}-\vect{G}_1} &= U_{\vect{G}_2-\vect{G}_1}\, c_{\vect{k}-\vect{G}_2}, \\
    \bigl(\varepsilon - \varepsilon^0_{\vect{k}-\vect{G}_2}\bigr) c_{\vect{k}-\vect{G}_2} &= U_{\vect{G}_1-\vect{G}_2}\, c_{\vect{k}-\vect{G}_1}.
\end{align}
Letting $\vect{q} = \vect{k} - \vect{G}_1$ and $\vect{G} = \vect{G}_2 - \vect{G}_1$, a nontrivial solution requires
\begin{equation}
    \det \begin{pmatrix} \varepsilon - \varepsilon^0_{\vect{q}} & -U_{\vect{G}} \\[2pt] -U^*_{\vect{G}} & \varepsilon - \varepsilon^0_{\vect{q}-\vect{G}} \end{pmatrix} = 0,
\end{equation}
whose roots are
\begin{equation}
    \varepsilon = \tfrac{1}{2}\bigl(\varepsilon^0_{\vect{q}} + \varepsilon^0_{\vect{q}-\vect{G}}\bigr)
    \pm \left[ \left( \frac{\varepsilon^0_{\vect{q}} - \varepsilon^0_{\vect{q}-\vect{G}}}{2} \right)^{\!2} + |U_{\vect{G}}|^2 \right]^{1/2}.
    \label{eq:bloch-two-level}
\end{equation}
Note that when $\varepsilon^0_{\vect{q}} = \varepsilon^0_{\vect{q}-\vect{G}}$,
\begin{equation}
    \pdv{\varepsilon}{\vect{q}} = \frac{\hbar^2}{m}\bigl(\vect{q} - \tfrac{1}{2}\vect{G}\bigr),
\end{equation}
so the constant-energy surfaces meet the Bragg plane at right angles. If $\vect{q}$ lies exactly on the Bragg plane the two free-electron energies coincide and~\eqref{eq:bloch-two-level} collapses to
\begin{equation}
    \varepsilon = \varepsilon^0_{\vect{q}} \pm |U_{\vect{G}}|.
\end{equation}
\end{example}

The two branches are therefore split by $2|U_{\vect{G}}|$ where they would otherwise have crossed, as sketched in Figure~\ref{fig:bloch-gap}.

\subsection{Wavefunctions at a Single Bragg Plane}

Plugging $\varepsilon = \varepsilon^0_{\vect{q}} \pm |U_{\vect{G}}|$ back into the two-level system fixes the ratio of the coefficients,
\begin{equation}
    c_{\vect{q}} = \pm \operatorname{sgn}(U_{\vect{G}})\, c_{\vect{q}-\vect{G}},
\end{equation}
so for $U_{\vect{G}} > 0$ the two solutions are the even and odd combinations of the two plane waves:
\begin{align}
    \psi(\vect{x}) &\propto e^{i\frac{\vect{G}}{2}\cdot\vect{x}} + e^{-i\frac{\vect{G}}{2}\cdot\vect{x}}
    \;\Longrightarrow\; |\psi(\vect{x})|^2 \propto \cos^2\!\left(\tfrac{\vect{G}}{2}\cdot\vect{x}\right),
    && \varepsilon = \varepsilon^0_{\vect{q}} + |U_{\vect{G}}| \quad \text{($s$-like)}, \\
    \psi(\vect{x}) &\propto e^{i\frac{\vect{G}}{2}\cdot\vect{x}} - e^{-i\frac{\vect{G}}{2}\cdot\vect{x}}
    \;\Longrightarrow\; |\psi(\vect{x})|^2 \propto \sin^2\!\left(\tfrac{\vect{G}}{2}\cdot\vect{x}\right),
    && \varepsilon = \varepsilon^0_{\vect{q}} - |U_{\vect{G}}| \quad \text{($p$-like)}.
\end{align}
The lower-energy ($p$-like) state piles its charge away from the ion cores, the higher-energy ($s$-like) state piles it on them.

\subsection{Energy--Wavevector Curves in Three Dimensions}

In three dimensions the free-electron band structure is highly degenerate, and group theory determines how many bands a weak potential splits each degeneracy into. The generic statement that survives is about the size of the splitting.

\begin{fact}[Energy gap]
The energy gap opened at a Bragg plane is at least $2|U_{\vect{G}}|$.
\end{fact}

\subsection{Brillouin Zones and Fermi Levels}

\begin{definition}[$n$th Brillouin zone]
The $n$th Brillouin zone is the set of points that can be reached from the origin by crossing exactly $n-1$ Bragg planes. Each Brillouin zone is a primitive unit cell of the reciprocal lattice.
\end{definition}

This gives an intuitive construction of Fermi surfaces in a weak periodic potential: draw the free-electron Fermi sphere about each reciprocal lattice point, and then note that wherever these spheres intersect a Bragg plane the potential rounds off the sharp corners, since the constant-energy surfaces must meet the zone boundary at right angles (Figure~\ref{fig:bloch-fermi}).

% ============================================================
\section{The Tight-Binding Method}
% ============================================================

The weak-potential treatment above starts from plane waves and perturbs them with the ions. The opposite starting point is the isolated atom, which is appropriate whenever the solid is composed of weakly interacting neutral atoms.

\subsection{Assumptions}

We assume the solid is built from weakly interacting neutral atoms. Lower orbitals ($1s$, $2s$, $2p$, etc.) have less overlap between neighbouring atoms, so although the overlap is non-negligible we may still use single-atom atomic orbitals as a basis. The method is especially useful for describing $d$-shells of transition metals and for insulators.

\subsection{General Formulation}

Assume that in the vicinity of an atom the full Hamiltonian $H$ is well approximated by the single-atom Hamiltonian $H_{\text{at}}$,
\begin{equation}
    H_{\text{at}} \psi_n = E_n \psi_n,
\end{equation}
and require $\psi_n(\vect{x})$ to be very small once $|\vect{x}|$ exceeds the lattice constant $a$. Writing the crystal Hamiltonian as
\begin{equation}
    H = H_{\text{at}} + \Delta U(\vect{x}),
\end{equation}
we have
\begin{equation}
    H \psi_n(\vect{x}) = H_{\text{at}} \psi_n(\vect{x}) + \Delta U(\vect{x}) \psi_n(\vect{x})
    = E_n \psi_n(\vect{x}) + \boxed{\Delta U(\vect{x})\psi_n(\vect{x})}.
\end{equation}
For $\psi_n$ to satisfy the crystal Hamiltonian the boxed term must vanish, which happens either because $\Delta U(\vect{x})$ is small (close to the atom) or because $\psi_n(\vect{x})$ is small (far from the atom). Hence an atomic solution $\psi_n(\vect{x} - \vect{R})$ can be placed at each atomic site $\vect{R}$ of the lattice.

These site-localized functions do not individually satisfy the Bloch condition $\psi(\vect{x}+\vect{R}) = e^{i\vect{k}\cdot\vect{R}}\psi(\vect{x})$, so we take linear combinations,
\begin{equation}
    \psi_{n\vect{k}}(\vect{x}) = \sum_{\vect{R}} e^{i\vect{k}\cdot\vect{R}}\, \psi_n(\vect{x} - \vect{R}),
    \label{eq:bloch-tb-sum}
\end{equation}
where $\vect{k}$ lies in the first Brillouin zone with Born--von Karman boundary conditions.

\begin{theorem}[\eqref{eq:bloch-tb-sum} satisfies Bloch's theorem]
Let $\vect{R}'' = \vect{R}' - \vect{R}$, which is again a lattice vector by the periodic boundary conditions. Then
\begin{align}
    \psi(\vect{x}+\vect{R}) &= \sum_{\vect{R}'} e^{i\vect{k}\cdot\vect{R}'}\, \psi_n(\vect{x} + \vect{R} - \vect{R}')
    = \sum_{\vect{R}''} e^{i\vect{k}\cdot(\vect{R}''+\vect{R})}\, \psi_n(\vect{x} - \vect{R}'') \\
    &= e^{i\vect{k}\cdot\vect{R}} \sum_{\vect{R}''} e^{i\vect{k}\cdot\vect{R}''}\, \psi_n(\vect{x} - \vect{R}'')
    = e^{i\vect{k}\cdot\vect{R}}\, \psi(\vect{x}).
\end{align}
\end{theorem}

However, acting with $H$ on this state,
\begin{equation}
    H\psi_{n\vect{k}} = \sum_{\vect{R}} e^{i\vect{k}\cdot\vect{R}} H \psi_n(\vect{x}-\vect{R}) = E_n \psi_{n\vect{k}}(\vect{x}),
\end{equation}
gives no modification to the atomic energies at all --- which is not realistic. The fix is to assume instead that the product $\Delta U(\vect{x})\psi_n(\vect{x})$ is exceedingly small but nonzero.

\begin{formula}[Tight-binding ansatz]
Assume a single-atom-like wavefunction $\phi(\vect{x})$ close to the true $\psi(\vect{x})$,
\begin{equation}
    \phi(\vect{x}) = \sum_n b_n \psi_n(\vect{x}) \qquad \text{(cf.\ Wannier functions)},
\end{equation}
and build the crystal state as
\begin{equation}
    \psi_{\vect{k}}(\vect{x}) = \sum_{\vect{R}} e^{i\vect{k}\cdot\vect{R}}\, \phi(\vect{x}-\vect{R}),
    \qquad b_n = \braket{\psi_n | \phi}.
\end{equation}
\end{formula}

\subsection{Finding the Coefficients}

We need the $b_n(\vect{k})$. Start from $H\ket{\psi_{\vect{k}}} = (H_{\text{at}} + \Delta U)\ket{\psi_{\vect{k}}} = \varepsilon(\vect{k})\ket{\psi_{\vect{k}}}$ and project onto an atomic orbital. Since $\bra{\psi_m} H_{\text{at}} \ket{\psi_{\vect{k}}} = E_m \braket{\psi_m | \psi_{\vect{k}}}$,
\begin{equation}
    \bra{\psi_m} H \ket{\psi_{\vect{k}}} = E_m \braket{\psi_m|\psi_{\vect{k}}} + \bra{\psi_m}\Delta U\ket{\psi_{\vect{k}}} = \varepsilon(\vect{k}) \braket{\psi_m|\psi_{\vect{k}}},
\end{equation}
so that
\begin{equation}
    \bigl(\varepsilon(\vect{k}) - E_m\bigr) \braket{\psi_m | \psi_{\vect{k}}} = \bra{\psi_m} \Delta U \ket{\psi_{\vect{k}}}.
\end{equation}
Now insert $\ket{\psi_{\vect{k}}} = \sum_{\vect{R}} e^{i\vect{k}\cdot\vect{R}} \sum_n b_n \ket{\psi_n(\vect{x}-\vect{R})}$ and use $\braket{\psi_m|\psi_n} = \delta_{mn}$:
\begin{equation}
    \bigl(\varepsilon(\vect{k}) - E_m\bigr) \sum_n b_n \sum_{\vect{R}} e^{i\vect{k}\cdot\vect{R}}
    \underbrace{\braket{\psi_m | \psi_n(\vect{x}-\vect{R})}}_{\displaystyle \alpha_{mn}(\vect{R})}
    = \sum_n b_n \sum_{\vect{R}} e^{i\vect{k}\cdot\vect{R}}
    \underbrace{\bra{\psi_m} \Delta U \ket{\psi_n(\vect{x}-\vect{R})}}_{\displaystyle \gamma_{mn}(\vect{R})}.
\end{equation}
Rearranging gives the general constraint
\begin{equation}
    \varepsilon(\vect{k}) \sum_n b_n \sum_{\vect{R}} e^{i\vect{k}\cdot\vect{R}} \alpha_{mn}(\vect{R})
    = \sum_n b_n \sum_{\vect{R}} e^{i\vect{k}\cdot\vect{R}} \bigl( E_m \alpha_{mn}(\vect{R}) + \gamma_{mn}(\vect{R}) \bigr).
    \label{eq:bloch-tb-constraint}
\end{equation}

\subsection{The Tight-Binding Equation}

Assume now that $\alpha_{mn}(\vect{R})$ and $\gamma_{mn}(\vect{R})$ are negligible for $|\vect{R}|$ beyond the nearest neighbours, so that the lattice sums run over a finite set. Define
\begin{equation}
    \alpha_{mn}(\vect{k}) = \sum_{\text{n.n.}} e^{i\vect{k}\cdot\vect{R}} \alpha_{mn}(\vect{R}),
    \qquad
    H_{mn}(\vect{k}) = \sum_{\text{n.n.}} e^{i\vect{k}\cdot\vect{R}} H_{mn}(\vect{R}),
    \qquad
    H_{mn}(\vect{R}) = E_m \alpha_{mn}(\vect{R}) + \gamma_{mn}(\vect{R}).
\end{equation}
Then~\eqref{eq:bloch-tb-constraint} reads $\varepsilon(\vect{k}) \sum_n b_n \alpha_{mn}(\vect{k}) = \sum_n b_n H_{mn}(\vect{k})$.

\begin{formula}[Tight-binding equation]
\begin{equation}
    \varepsilon(\vect{k})\, \alpha(\vect{k})\, \vect{c} = H(\vect{k})\, \vect{c},
    \qquad \vect{c} = (b_1, \ldots, b_n)^T.
\end{equation}
\end{formula}

\subsection{Application to an $s$-Band}

\begin{example}[Single $s$-band]
If there is only one band then $\alpha(\vect{k}) = \mathbb{1}$ and $\vect{c} \to 1$, so $\varepsilon(\vect{k})\mathbb{1} = H(\vect{k})$. With $\alpha_{mn}(\vect{R}) = \delta_{mn}\delta_{\vect{R},0}$,
\begin{equation}
    H(\vect{k}) = \sum_{\text{n.n.}} e^{i\vect{k}\cdot\vect{R}} \bigl( E_m \delta_{\vect{R},0} + \gamma(\vect{R}) \bigr)
    = E_m - \beta + \sum_{\text{n.n.}} e^{i\vect{k}\cdot\vect{R}} \gamma(\vect{R}),
\end{equation}
where $-\beta = \gamma(0)$ is the on-site term. By inversion symmetry of the lattice $\Delta U(-\vect{R}) = \Delta U(\vect{R})$, so $\gamma(\vect{R}) = \gamma$ is the same for all nearest neighbours and the exponentials combine into cosines:
\begin{equation}
    \varepsilon(\vect{k}) = E_s - \beta + \sum_{\text{n.n.}} \gamma \cos(\vect{k}\cdot\vect{R}).
\end{equation}
\end{example}

\begin{formula}[Single-orbital tight-binding]
\begin{equation}
    \varepsilon(\vect{k}) = E_s - \beta - \sum_{\text{n.n.}} \gamma(\vect{R}) \cos(\vect{k}\cdot\vect{R}),
    \qquad
    \gamma \equiv -\bra{\phi(\vect{x})} \Delta U(\vect{x}) \ket{\phi(\vect{x}-\vect{R})},
\end{equation}
where ``n.n.'' means only nearest-neighbour atomic separations are summed. The assumptions are:
\begin{enumerate}
    \item \emph{Orthogonality}: $\alpha(\vect{R}) = \int d\vect{x}\; \phi^*(\vect{x})\phi(\vect{x}-\vect{R}) = 0$;
    \item only nearest-neighbour separations give appreciable overlap integrals.
\end{enumerate}
\end{formula}

\begin{example}[Tight-binding for an FCC crystal]
The 12 nearest neighbours of the origin in an FCC lattice sit at
\begin{equation}
    \vect{R} = \tfrac{a}{2}(\pm 1, \pm 1, 0), \quad \tfrac{a}{2}(\pm 1, 0, \pm 1), \quad \tfrac{a}{2}(0, \pm 1, \pm 1),
\end{equation}
so $\vect{k}\cdot\vect{R} = \tfrac{a}{2}(\pm k_i \pm k_j)$ with $ij = xy,\, yz,\, xz$. Summing the twelve exponentials gives
\begin{equation}
    \varepsilon(\vect{k}) = E_s - \beta - 4\gamma \sum_{ij} \cos(k_i a)\cos(k_j a).
\end{equation}
The intuition to carry away: the smaller the orbital overlap integral, the narrower the band.
\end{example}

\subsection{Remarks on the Tight-Binding Method}

\begin{enumerate}
    \item The number of atomic levels kept in the expansion determines the number of bands, and equals the dimension of the matrix problem --- as one sees from the indices of $H_{mn}$.
    \item The width of a band (lowest to highest energy within it) scales with the overlap integrals $\gamma_{mn}(\vect{R})$. Rule of thumb: as the energy of a given atomic level increases, so does the spatial extent of its wavefunction.
    \item The more overlap, the greater $\partial\varepsilon/\partial \vect{k}$, which means the electrons have a larger mean velocity.
    \item For materials with more than one atom in the basis, use
    \begin{equation}
        \psi(\vect{x}) = \sum_{\vect{R}} e^{i\vect{k}\cdot\vect{R}}\bigl( a\,\phi(\vect{x}-\vect{R}) + b\,\phi(\vect{x}-\vect{d}-\vect{R}) \bigr).
    \end{equation}
    \item When spin--orbit coupling is present (as in heavier elements), include in $\Delta U(\vect{x})$ the electron's interaction with neighbouring atomic spins, and approximate $\phi$ as a linear combination of the same orbital wavefunctions but of opposite spin --- which doubles the size of the tight-binding matrix.
    \item The method still assumes the independent-electron approximation, and so fails for magnetic materials.
\end{enumerate}

\subsection{Wannier Functions}

Wannier functions form a basis for the atomic wavefunctions $\phi$. Any Bloch function can be written in the form
\begin{equation}
    \psi_{n\vect{k}}(\vect{x}) = \sum_{\vect{R}} e^{i\vect{k}\cdot\vect{R}} f_n(\vect{R}, \vect{x}),
    \qquad
    f_n(\vect{R},\vect{x}) = \frac{1}{v_0} \int d\vect{k}\; e^{-i\vect{k}\cdot\vect{R}} \psi_{n\vect{k}}(\vect{x}),
\end{equation}
and by Bloch's theorem it must be that $f_n(\vect{R},\vect{x}) = \phi_n(\vect{x}-\vect{R})$.

\begin{fact}[Orthogonality]
\begin{equation}
    \braket{\phi_n | \phi_m} = \braket{\phi_n(\vect{x}-\vect{R}_a) | \phi_n(\vect{x}-\vect{R}_b)} = 0.
\end{equation}
\end{fact}

Because they are localized, Wannier functions are important for modelling semiclassical electron transport and defect states in semiconductors.

% ============================================================
\section{Lattice Fourier Transforms and Identities}
% ============================================================

Both models above lean on a handful of lattice Fourier identities, which we now collect. The starting point is that the plane waves $e^{i\vect{k}\cdot\vect{x}}$ form a complete basis for any well-behaved function on $\mathbb{R}^3$.

\subsection{Functions Periodic in Real Space}

If $f(\vect{x}+\vect{R}) = f(\vect{x})$, then only those plane waves periodic in the lattice can appear --- that is, those with reciprocal lattice wavevectors:
\begin{equation}
    f(\vect{x}) = \sum_{\vect{G}} f_{\vect{G}}\, e^{i\vect{G}\cdot\vect{x}},
    \qquad
    f_{\vect{G}} = \frac{1}{v}\int_C d\vect{x}\; e^{-i\vect{G}\cdot\vect{x}} f(\vect{x}),
\end{equation}
where the integral runs over any primitive cell $C$ of volume $v$.

\begin{theorem}[Vanishing of the cell integral]
For a nonzero reciprocal lattice vector $\vect{G}$, $\int_C d\vect{x}\, e^{i\vect{G}\cdot\vect{x}} = 0$. Translating the unit cell does not change the integral, so
\begin{equation}
    \int_C d\vect{x}\; e^{i\vect{G}\cdot\vect{x}} = \int_C d\vect{x}\; e^{i\vect{G}\cdot(\vect{x}+\vect{d})}
    \;\Longrightarrow\;
    \left(\int_C d\vect{x}\; e^{i\vect{G}\cdot\vect{x}}\right)\bigl(e^{i\vect{G}\cdot\vect{d}} - 1\bigr) = 0.
\end{equation}
The second factor is never zero for generic $\vect{d}$, so the integral vanishes.
\end{theorem}

\subsection{Functions Periodic in Reciprocal Space}

Dually, a function periodic in reciprocal space is expanded on direct lattice vectors:
\begin{equation}
    \phi(\vect{k}) = \sum_{\vect{R}} e^{i\vect{R}\cdot\vect{k}}\, \phi_{\vect{R}},
    \qquad
    \phi_{\vect{R}} \equiv V \int \frac{d\vect{k}}{(2\pi)^3}\, e^{-i\vect{R}\cdot\vect{k}}\, \phi(\vect{k}).
\end{equation}

\subsection{Functions Periodic under Born--von Karman Boundary Conditions}

The BVK boundary condition is $f(\vect{x} + N_i \vect{a}_i) = f(\vect{x})$, which restricts the allowed wavevectors to
\begin{equation}
    \vect{k} = \sum_{i=1}^{3} \frac{m_i}{N_i}\, \vect{b}_i, \qquad m_i \in \mathbb{Z}.
\end{equation}
The expansion then takes the form
\begin{equation}
    f(\vect{x}) = \sum_{\vect{k}} f_{\vect{k}}\, e^{i\vect{k}\cdot\vect{x}},
    \qquad
    f_{\vect{k}} \equiv \frac{1}{V}\int d\vect{x}\; e^{-i\vect{k}\cdot\vect{x}} f(\vect{x}),
\end{equation}
and again $\int_V d\vect{x}\, e^{i\vect{k}\cdot\vect{x}} = 0$ for $\vect{k} \neq 0$.

\begin{theorem}[Sum of exponentials identity]
For $\vect{R}$ running over the lattice sites $\sum_{i=1}^3 n_i \vect{a}_i$ with $0 \leq n_i < N_i$ and $N_1 N_2 N_3 = N$,
\begin{equation}
    \sum_{\vect{R}} e^{i\vect{k}\cdot\vect{R}} = N \delta_{\vect{k},0}.
\end{equation}
The sum is invariant under displacing every site by a lattice vector $\vect{R}_0$,
\begin{equation}
    \sum_{\vect{R}} e^{i\vect{k}\cdot(\vect{R}+\vect{R}_0)} = e^{i\vect{k}\cdot\vect{R}_0} \sum_{\vect{R}} e^{i\vect{k}\cdot\vect{R}} = \sum_{\vect{R}} e^{i\vect{k}\cdot\vect{R}},
\end{equation}
so that $\bigl(e^{i\vect{k}\cdot\vect{R}_0} - 1\bigr)\bigl(\sum_{\vect{R}} e^{i\vect{k}\cdot\vect{R}}\bigr) = 0$. If $\vect{k} = 0$ the sum is $\sum_{\vect{R}} 1 = N$; if $\vect{k} \neq 0$ the sum must vanish.
\end{theorem}

\begin{formula}[BVK identity]
\begin{equation}
    \sum_{\vect{k}} e^{i\vect{k}\cdot\vect{R}} = N \delta_{\vect{R},0}.
\end{equation}
\end{formula}

\section{Wave Packets of Bloch Electrons}
% ============================================================

In the semiclassical picture electrons do \emph{not} collide with the ions --- the mean free path is far too long for that --- but rather with imperfections in the crystal. To describe their response to applied fields we build wavepackets out of Bloch states.

\begin{definition}[Bloch wavepacket]
\begin{equation}
    \psi_n(\vect{x},t) = \sum_{\vect{k}'} g(\vect{k}')\, \psi_{n\vect{k}'}(\vect{x})\, \exp\!\left(-\tfrac{i}{\hbar}\varepsilon_n(\vect{k}')t\right),
\end{equation}
where the weight $g(\vect{k}') \approx 0$ once $|\vect{k}'-\vect{k}| > \Delta k$. The packet must obey the uncertainty principle $\Delta k\, \Delta x \gtrsim 1$.
\end{definition}

\subsection{Width of the Wavepacket}

Assume $\Delta k$ is small compared with the dimensions of the Brillouin zone, and examine the amplitude at points a lattice vector away:
\begin{equation}
    \psi_n(\vect{x}_0 + \vect{R}, t) = \sum_{\vect{k}'} \underbrace{\bigl[ g(\vect{k}')\psi_{n\vect{k}'}(\vect{x}_0) \bigr]}_{\text{weight function}}
    \exp\!\left[ i\Bigl( \underbrace{\vect{k}'\cdot\vect{R}}_{\text{from the Bloch condition}} - \tfrac{1}{\hbar}\varepsilon_n(\vect{k}')t \Bigr) \right].
\end{equation}
By the properties of the Fourier transform, if $g(\vect{k})$ is appreciable within $\Delta k$ then $g(\vect{x})$ is appreciable within $\Delta R \approx 1/\Delta k$. Since $\Delta k$ is small compared with the Brillouin zone dimension, of order $1/a$, the spread $\Delta R$ is large compared with the lattice constant $a$: wavepackets are spread over many primitive unit cells.

\subsection{Group Velocity of Bloch Electrons}

Expand the band energy to second order in a small shift $\vect{q}$:
\begin{equation}
    \varepsilon_n(\vect{k}+\vect{q}) = \varepsilon_n(\vect{k}) + \sum_i \pdv{\varepsilon_n}{k_i} q_i
    + \frac{1}{2}\sum_{ij} \frac{\partial^2 \varepsilon_n}{\partial k_i \partial k_j} q_i q_j + O(q^3).
    \label{eq:bloch-expand}
\end{equation}
The same shift can be treated by perturbation theory: since $H_{\vect{k}+\vect{q}}\psi^n_{\vect{k}+\vect{q}} = \varepsilon^n_{\vect{k}+\vect{q}}\psi^n_{\vect{k}+\vect{q}}$,
\begin{equation}
    H_{\vect{k}+\vect{q}} = H_{\vect{k}} + \underbrace{\frac{\hbar^2}{m}\,\vect{q}\cdot\left(\frac{1}{i}\nabla + \vect{k}\right)}_{\displaystyle V(\vect{q})} + \frac{\hbar^2}{2m}q^2 .
    \label{eq:bloch-Hexpand}
\end{equation}
If the unperturbed problem is $H_0 \psi_n = E^0_n \psi_n$, then to second order in $V$,
\begin{equation}
    E_n = E^0_n + \int d\vect{x}\; \psi_n^* V \psi_n + \sum_{n' \neq n} \frac{\bigl| \int d\vect{x}\; \psi_n^* V \psi_{n'} \bigr|^2}{E^0_n - E^0_{n'}}.
    \label{eq:bloch-pert}
\end{equation}
Comparing~\eqref{eq:bloch-expand} with~\eqref{eq:bloch-pert} at first order,
\begin{equation}
    \sum_i \pdv{\varepsilon_n}{k_i} q_i = \int d\vect{x}\; \psi_n^* V \psi_n,
\end{equation}
and comparing~\eqref{eq:bloch-Hexpand} with~\eqref{eq:bloch-pert} we need the part of $V(\vect{q})$ linear in $\vect{q}$, so that
\begin{equation}
    \sum_i \pdv{\varepsilon_n}{k_i} q_i = \frac{\hbar^2}{m} \sum_i \int d\vect{x}\; u^*_{n\vect{k}}\, q_i \left(\frac{1}{i}\nabla + \vect{k}\right)_{\!i} u_{n\vect{k}}.
\end{equation}
Recalling the momentum operator $\hat{p}_i \equiv \frac{\hbar}{i}\pdv{}{x_i}$,
\begin{equation}
    \frac{1}{\hbar}\pdv{\varepsilon_n}{k_i} = \frac{1}{m} \Bigl\langle \frac{\hbar}{i}\pdv{}{x_i} + \hbar k_i \Bigr\rangle_{u_{n\vect{k}}}
    = \frac{1}{m}\bigl\langle \hat{p}_i + \hbar k_i \bigr\rangle_{u_{n\vect{k}}}.
\end{equation}
Finally, using $\psi_{n\vect{k}} = e^{i\vect{k}\cdot\vect{x}} u_{n\vect{k}}(\vect{x})$,
\begin{equation}
    \frac{1}{\hbar}\pdv{\varepsilon_n}{k_i}
    = \frac{1}{m}\int d\vect{x}\; u^*_{n\vect{k}} (\hat{p}_i + \hbar k_i) u_{n\vect{k}}
    = \frac{1}{m}\int d\vect{x}\; e^{i\vect{k}\cdot\vect{x}} \psi^*_{n\vect{k}} (\hat{p}_i + \hbar k_i) e^{-i\vect{k}\cdot\vect{x}} \psi_{n\vect{k}},
\end{equation}
and since
\begin{equation}
    \hat{p}_i \bigl[ e^{-i\vect{k}\cdot\vect{x}} \psi_{n\vect{k}}(\vect{x}) \bigr]
    = \frac{\hbar}{i}\pdv{}{x_i}\bigl[ e^{-i\vect{k}\cdot\vect{x}} \psi_{n\vect{k}}(\vect{x}) \bigr]
    = -\hbar k_i\, e^{-i\vect{k}\cdot\vect{x}}\psi_{n\vect{k}}(\vect{x}) + e^{-i\vect{k}\cdot\vect{x}}\, \hat{p}_i \psi_{n\vect{k}}(\vect{x}),
\end{equation}
the $\hbar k_i$ terms cancel, leaving
\begin{equation}
    \frac{1}{\hbar}\pdv{\varepsilon_n}{k_i}
    = \frac{1}{m}\int d\vect{x}\; e^{i\vect{k}\cdot\vect{x}} \psi^*_{n\vect{k}} e^{-i\vect{k}\cdot\vect{x}} \hat{p}_i \psi_{n\vect{k}}
    = \frac{1}{m}\langle \hat{p}_i \rangle_{\psi_{n\vect{k}}} = \langle \hat{v}_i \rangle_{\psi_{n\vect{k}}}.
\end{equation}

\begin{formula}[Group velocity]
\begin{equation}
    \langle \vect{v}_n(\vect{k}) \rangle = \frac{1}{\hbar} \pdv{\varepsilon_n(\vect{k})}{\vect{k}} = \pdv{\omega}{\vect{k}}.
\end{equation}
\end{formula}

\section{The Semiclassical Model}
% ============================================================

The semiclassical model describes the response of electrons to externally applied electric and magnetic fields that vary slowly on the scale $\Delta R$ of the wavepacket. Given the band structure $\varepsilon_n(\vect{k})$, it associates with each electron a position $\vect{x}$, a wavevector $\vect{k}$, and a band index $n$.

\begin{fact}[Assumptions of the semiclassical model]
\begin{enumerate}
    \item The electron stays within its band: there are no interband transitions.
    \item The time evolution is governed by
    \begin{equation}
        \dot{\vect{x}} = \vect{v}_n(\vect{k}) = \frac{1}{\hbar}\pdv{\varepsilon_n(\vect{k})}{\vect{k}},
        \qquad
        \hbar \dot{\vect{k}} = -e\left[ \vect{E}(\vect{x},t) + \frac{1}{c}\, \vect{v}_n(\vect{k}) \times \vect{H}(\vect{x},t) \right],
    \end{equation}
    where $\vect{H}$ is the magnetic field. Note that the crystal momentum $\hbar\vect{k}$ is \emph{not} the total electron momentum.
    \item All distinct wavevectors for a single band lie within a single primitive cell of the reciprocal lattice, with occupation given by the Fermi distribution
    \begin{equation}
        f\bigl(\varepsilon_n(\vect{k})\bigr) \frac{d\vect{k}}{4\pi^3}
        = \frac{d\vect{k}/4\pi^3}{e^{(\varepsilon_n(\vect{k})-\mu)/k_B T}+1}.
    \end{equation}
\end{enumerate}
\end{fact}

A useful corollary is the \emph{many-carrier} statement: one need only consider the types of carriers lying within $\sim k_B T$ of the Fermi level.

\subsection{Conditions of Validity}

There must be some minimum strength of the periodic potential below which the model breaks down, because interband transitions become possible. The conditions are
\begin{equation}
    e |\vect{E}| a \ll \frac{\bigl[\varepsilon_{\text{gap}}(\vect{k})\bigr]^2}{\varepsilon_F},
    \qquad
    \hbar\omega_c \ll \frac{\bigl[\varepsilon_{\text{gap}}(\vect{k})\bigr]^2}{\varepsilon_F},
\end{equation}
where $\omega_c$ is the cyclotron frequency: both the work done by the field over a lattice constant and the cyclotron energy quantum must be small compared with the scale set by the gap.
