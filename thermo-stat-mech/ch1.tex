\section{Partial Derivatives}
% ============================================================

The partial derivative of $F(x_1, x_2, \ldots)$ with respect to $x_1$ holds all other arguments fixed:
\begin{equation}
    \pdv{F}{x_1} = \lim_{\Delta x_1 \to 0} \frac{F(x_1 + \Delta x_1,\, x_2, \ldots) - F(x_1, x_2, \ldots)}{\Delta x_1}.
\end{equation}

\subsection{Relations among First Partials}

For $x(y, z)$, the total differential is $dx = \left.\pdv{x}{y}\right|_z dy + \left.\pdv{x}{z}\right|_y dz$. The inverse-function theorem gives:
\begin{formula}
\begin{equation}
    \left.\pdv{x}{y}\right|_z = \left(\left.\pdv{y}{x}\right|_z\right)^{-1}.
\end{equation}
\end{formula}

Setting $dx = 0$ gives $dz/dy = -(\partial x/\partial y)_z / (\partial x/\partial z)_y = (\partial z/\partial y)_x$, yielding the \emph{triple product rule}:
\begin{formula}
\begin{equation}
    \left.\pdv{x}{y}\right|_z \left.\pdv{y}{z}\right|_x \left.\pdv{z}{x}\right|_y = -1.
\end{equation}
\end{formula}
The minus sign is a recurring source of error — three cyclic partial derivatives multiply to $-1$, not $+1$.

\subsection{Mixed Second Partials}

For $x(y, z)$, provided $x$ is twice continuously differentiable, mixed partials commute:
\begin{equation}
    \left.\pdv{f_1}{z}\right|_y = \left.\pdv{f_2}{y}\right|_z = \frac{\partial^2 x}{\partial y\,\partial z},
\end{equation}
where $f_1 = (\partial x/\partial y)_z$ and $f_2 = (\partial x/\partial z)_y$. This symmetry underpins all Maxwell relations.

\subsection{Exact and Inexact Differentials}

The expression $df = a(y,z)\,dy + b(y,z)\,dz$ is an \emph{exact differential} (i.e.\ $f$ is a well-defined function of state) if and only if
\begin{equation}
    \left.\pdv{a}{z}\right|_y = \left.\pdv{b}{y}\right|_z.
\end{equation}

When exact, $f$ is recovered by integrating $a$ with respect to $y$ and fixing the $z$-dependence:
\begin{equation}
    f(y,z) = \int^y a(y', z)\,dy' + g(z), \qquad g'(z) = b(y,z) - \int^y \left.\pdv{a(y',z)}{z}\right|_{y'} dy'.
\end{equation}

When inexact, the integral is path-dependent. For two paths $C$, $C'$ between the same endpoints,
\begin{equation}
    f_C - f_{C'} = \oint d\vec{x}\cdot\vec{V} = \int_A d\vec{A}\cdot(\nabla\times\vec{V}), \qquad
    \nabla\times\vec{V} = \left.\pdv{a}{z}\right|_y - \left.\pdv{b}{y}\right|_z,
\end{equation}
which vanishes iff $df$ is exact. Inexact differentials in thermodynamics ($\dbar Q$, $\dbar W$) are path-dependent in precisely this sense.

\begin{definition}[Dirac delta function]
The Dirac delta $\delta(x - a)$ satisfies
\begin{equation}
    \delta(x - a) = 0 \quad \text{for } x \neq a, \qquad
    \int_{x_0}^{x_1} \delta(x - a)\,dx =
    \begin{cases} 1 & a \in (x_0, x_1), \\ 0 & \text{otherwise.} \end{cases}
\end{equation}
It is a distribution, not a classical function — always understood inside an integral.
\end{definition}

The sifting property for a smooth test function $f(x)$:
\begin{formula}
\begin{equation}
    \int_{x_0}^{x_1} f(x)\,\delta(x - a)\,dx = f(a), \qquad a \in (x_0, x_1).
\end{equation}
\end{formula}

\subsection{Composite Arguments}

For a linear argument, substituting $u = cx$:
\begin{equation}
    \int f(x)\,\delta(cx - a)\,dx = \frac{1}{|c|}\,f\!\left(\frac{a}{c}\right).
\end{equation}
For a nonlinear argument $g(x)$, each simple zero $x_j$ of $g$ contributes:
\begin{formula}
\begin{equation}
    \int f(x)\,\delta(g(x))\,dx = \sum_j \frac{f(x_j)}{|g'(x_j)|}.
\end{equation}
\end{formula}

\subsection{Representations}

The \emph{Gaussian representation} $\delta_\varepsilon(x) = ({\sqrt{2\pi}\,\varepsilon})^{-1}e^{-x^2/2\varepsilon^2}$ has unit area for all $\varepsilon > 0$ and converges to $\delta(x)$ in the distributional sense as $\varepsilon \to 0$.

\section{Probability}
% ============================================================

\begin{definition}[Random variable]
A random variable $X$ is a quantity with a definite set of possible values $\{x_j\}$ whose outcome cannot be predicted with certainty. It may be discrete (coin flip, die face) or continuous (molecular speed). Randomness may be classical (incomplete information) or quantum mechanical.
\end{definition}

For a discrete variable taking values $\{x_j\}$, $j = 1,\ldots,M$:
\begin{equation}
    P_j = \lim_{N\to\infty} \frac{n_j(N)}{N}, \qquad P_j \ge 0, \qquad \sum_j P_j = 1.
\end{equation}
For continuous variables, the probability density function (PDF) $p(x)$ is defined by:
\begin{definition}[Probability density]
\begin{equation}
    dP = p(x)\,dx, \qquad p(x) \ge 0, \qquad \int_{-\infty}^\infty p(x)\,dx = 1.
\end{equation}
\end{definition}

The \emph{cumulative distribution function} (CDF) and its relation to the PDF:
\begin{definition}[CDF]
\begin{equation}
    \mathcal{P}(x) = \int_{-\infty}^{x} p(\xi)\,d\xi = \Pr[\text{value} < x], \qquad p(x) = \dv{\mathcal{P}}{x}.
\end{equation}
\end{definition}

A discrete distribution $\{x_j, P_j\}$ is encoded as a continuous density via:
\begin{formula}
\begin{equation}
    p(x) = \sum_j P_j\,\delta(x - x_j).
\end{equation}
\end{formula}

\subsection{Expectation Values and Moments}

\begin{definition}[Mean]
\begin{equation}
    \langle X \rangle = \int_{-\infty}^\infty x\,p(x)\,dx.
\end{equation}
\end{definition}

\begin{definition}[Variance and standard deviation]
\begin{equation}
    \mathrm{Var}(X) = \langle (X - \langle X\rangle)^2 \rangle = \langle X^2 \rangle - \langle X \rangle^2, \qquad \sigma(X) = \sqrt{\mathrm{Var}(X)}.
\end{equation}
\end{definition}
Higher moments: skewness $\langle[(X-\langle X\rangle)/\sigma]^3\rangle$ (asymmetry) and kurtosis $\langle[(X-\langle X\rangle)/\sigma]^4\rangle$ (tail weight).

\subsection{The Gaussian Distribution}

\begin{definition}[Gaussian distribution]
\begin{equation}
    p(x) = \frac{1}{\sqrt{2\pi\sigma^2}}\exp\!\left\{-\frac{(x-a)^2}{2\sigma^2}\right\},
\end{equation}
where $a$ is the mean and $\sigma^2$ is the variance.
\end{definition}

Normalization follows from $\int_{-\infty}^{\infty} e^{-u^2}\,du = \sqrt{\pi}$. Higher moments via differentiation under the integral sign:
\begin{equation}
    \int_{-\infty}^{\infty} u^{2n} e^{-u^2}\,du = \frac{1 \cdot 3 \cdot 5 \cdots (2n-1)}{2^n}\,\sqrt{\pi}.
\end{equation}

The CDF involves the complementary error function:
\begin{equation}
    P(\bar{x}) = 1 - \frac{1}{2}\operatorname{erfc}\!\left(\frac{\bar{x}-a}{\sqrt{2}\,\sigma}\right),
\end{equation}
\begin{definition}[Complementary error function]
\begin{equation}
    \operatorname{erfc}(z) = \frac{1}{\sqrt{\pi}}\int_{z}^{\infty} e^{-u^2}\,du.
\end{equation}
\end{definition}

Integrating over symmetric intervals about the mean:
\begin{align}
    \Pr(a - \sigma < x < a + \sigma) &\approx 0.68, \\
    \Pr(a - 2\sigma < x < a + 2\sigma) &\approx 0.95, \\
    \Pr(a - 5\sigma < x < a + 5\sigma) &\approx 1 - 5.7\times 10^{-7}.
\end{align}
The $5\sigma$ criterion is the standard discovery threshold in particle physics.

\subsection{The Poisson Distribution}

\begin{definition}[Poisson process]
A Poisson process models discrete events in continuous time. The probability of an event in $dt$ is $\lambda\,dt$, with $\lambda$ constant.
\end{definition}

Dividing observation time $T$ into $N\gg1$ bins each of probability $\lambda T/N$ and taking $N\to\infty$ yields:
\begin{formula}[Poisson distribution]
\begin{equation}
    P_n(T) = \frac{(\lambda T)^n}{n!}\,e^{-\lambda T}.
\end{equation}
\end{formula}

The mean follows from $n/n! = 1/(n-1)!$:
\begin{equation}
    \langle n\rangle = e^{-z}\sum_{n=0}^{\infty} n\,\frac{z^n}{n!} = z = \lambda T.
\end{equation}
Computing $\langle n(n-1)\rangle = z^2$ gives $\langle n^2\rangle = z^2 + z$, so:
\begin{formula}
\begin{equation}
    \mathrm{Var}(n) = \lambda T = \langle n\rangle, \qquad \sigma(n) = \sqrt{\lambda T}.
\end{equation}
\end{formula}

\begin{fact}
Variance equals mean for the Poisson distribution. The relative spread $\sigma/\langle n\rangle = 1/\sqrt{\lambda T}$ vanishes for large $\lambda T$.
\end{fact}

For large $\lambda T$, Stirling's approximation applied to $\log P_n = n\log z - \log n! - z$ gives the Gaussian limit:
\begin{equation}
    P_n(z) \approx \frac{1}{\sqrt{2\pi z}}\,e^{-(n-z)^2/2z},
\end{equation}
a Gaussian with mean and variance $z = \lambda T$ — an instance of the Central Limit Theorem.

\subsection{The Binomial Distribution}

The binomial distribution models $N$ independent steps each moving right (prob.\ $a$) or left (prob.\ $b = 1-a$), with $n$ counting right steps.

\begin{formula}[Binomial distribution]
\begin{equation}
    P_N(n) = \binom{N}{n}(1-a)^{N-n}\,a^n.
\end{equation}
\end{formula}

Normalization from the binomial theorem: $\sum_{n=0}^{N} P_N(n) = (a+b)^N = 1$. Using the differentiation trick $\langle n\rangle = a\,\partial_a(a+b)^N\big|_{a+b=1}$:
\begin{formula}
\begin{equation}
    \langle n\rangle = Na, \qquad \mathrm{Var}(n) = Na(1-a) = Nab, \qquad \sigma(n) = \sqrt{Nab}.
\end{equation}
\end{formula}
The relative spread $\sigma/\langle n\rangle \sim 1/\sqrt{N} \to 0$ as $N\to\infty$. Stirling's approximation gives the Gaussian limit:
\begin{equation}
    P_N(n) \approx \frac{1}{\sqrt{2\pi Na(1-a)}}\,e^{-(n-Na)^2/2Na(1-a)}.
\end{equation}

\subsection{Joint Distributions, Marginals, and Statistical Independence}

\begin{definition}[Joint probability density]
For continuous $X$ and $Y$, the joint PDF $p(x,y)$ satisfies
\begin{equation}
    p(x,y)\,dx\,dy \equiv \Pr(x \le X \le x+dx \text{ and } y \le Y \le y+dy).
\end{equation}
\end{definition}
The marginal distribution for $x$ is $p(x) = \int_{-\infty}^{\infty} p(x,y)\,dy$. If $p(x,y) = p(x)\,p(y)$, then $X$ and $Y$ are \emph{statistically independent} and all joint moments factorise: $\langle f(x)\,g(y)\rangle = \langle f(x)\rangle\langle g(y)\rangle$.

\subsection{Conditional Probability and Bayes' Theorem}

From $p(x,y)\,dx\,dy = p(y)\,dy\cdot p(x|y)\,dx$:
\begin{equation}
    p(x|y) = \frac{p(x,y)}{p(y)}.
\end{equation}
Symmetry $p(x,y) = p(x|y)\,p(y) = p(y|x)\,p(x)$ gives:
\begin{theorem}[Bayes' Theorem]
\begin{equation}
    p(y|x) = \frac{p(x|y)\,p(y)}{p(x)}.
\end{equation}
\end{theorem}

\begin{example}[Medical Testing]
A disease affects $\Pr(D) = 10^{-6}$ of the population. A test has sensitivity $\Pr(+|D) = 99\%$ and false-positive rate $\Pr(+|\bar{D}) = 1\%$. For a randomly chosen person testing positive:
\begin{equation}
    \Pr(+) \approx 0.99\times 10^{-6} + 0.01 \approx 0.01,
    \qquad
    \Pr(D|+) \approx \frac{0.99\times 10^{-6}}{0.01} \approx 10^{-4}.
\end{equation}
Despite 99\% accuracy, a positive result implies only $\sim0.01\%$ chance of disease — almost all positives are false alarms because the disease is so rare.
\end{example}

\section{Functions of Random Variables}
\label{sec:functions-rv}
% ============================================================

Given $X$ with density $p(x)$ and $Y = f(X)$, the density of $Y$ is found systematically via the delta function, which handles multi-valuedness automatically.

\subsection{The Delta-Function Method}

Inserting the identity $p(x) = \int p(k)\,\delta(x-k)\,dk$ into the transformation $Y = f(X)$:
\begin{equation}
    p(y) = \int_{-\infty}^{\infty} p(x)\,\delta\!\bigl(y - f(x)\bigr)\,dx.
\end{equation}
Applying $\delta(g(x)) = \sum_{x_i} |g'(x_i)|^{-1}\delta(x-x_i)$:
\begin{formula}[Density of a function of a random variable]
\begin{equation}
    p(y) = \sum_{x_i:\,f(x_i)=y} \frac{p(x_i)}{|f'(x_i)|}.
\end{equation}
\end{formula}

\subsection{Worked Example: Kinetic Energy from the Maxwell Speed Distribution}

\begin{example}[Kinetic energy from the Maxwell speed distribution]
For $p(v) = A\,e^{-Mv^2/2k_BT}$ and $E = \tfrac{1}{2}mv^2$, the map is two-to-one: $v_\pm = \pm\sqrt{2E/m}$, $f'(v) = mv$. The delta-function formula gives
\begin{align}
    p(E) &= \int_{-\infty}^{\infty} dv\; p(v)\,\delta\!\bigl(E - \tfrac{1}{2}mv^2\bigr) \notag \\
         &= \sum_{v_i = \pm\sqrt{2E/m}} \frac{p(v_i)}{m|v_i|}
          = \frac{2A\,e^{-E/k_BT}}{m\sqrt{2E/m}}
          = \frac{A}{\sqrt{2mE}}\cdot 2\,e^{-E/k_BT}.
\end{align}
Both roots contribute, supplying the factor of 2 that a naive chain-rule approach misses.
\end{example}

\subsection{Worked Example: Sum of Two Uniform Variables}

\begin{example}[Sum of two uniform random variables]
For $X, Y \sim \mathrm{Uniform}[0,1]$ independently and $Z = X + Y$:
\begin{equation}
    p(z) = \int_0^1 dx\int_{-\infty}^{\infty} dy\;\theta(y)\,\theta(1-y)\,\delta(z-x-y).
\end{equation}
Setting $y = z - x$, the step functions restrict the $x$-integral to $[\max(0,z-1),\min(1,z)]$:
\begin{align}
    0 < z < 1: &\quad p(z) = z, \\
    1 < z < 2: &\quad p(z) = 2-z.
\end{align}
The result is the \emph{triangular distribution} on $[0,2]$.

\section{Sums of Random Variables and the Central Limit Theorem}
% ============================================================

\subsection{Mean and Variance of Sums}

Let $S_n = \sum_{j=1}^{n} x_j$ with independent $x_j$ drawn from $p_j(x_j)$. The density is
\begin{equation}
    p(S_n) = \int \prod_{j=1}^{n} dx_j\, p_j(x_j)\; \delta\!\left(S_n - \sum_{j=1}^{n} x_j\right). \label{eq:sum_dist}
\end{equation}
The mean is $\langle S_n\rangle = \sum_{k}\langle x_k\rangle$. Independence factorises cross terms in $\langle S_n^2\rangle$:

\begin{theorem}[Additivity of Variance]
If $x_1, \ldots, x_n$ are statistically independent, then
\begin{equation}
    \mathrm{Var}\!\left(\sum_{k=1}^{n} x_k\right) = \sum_{k=1}^{n} \mathrm{Var}[x_k].
\end{equation}
\end{theorem}
For identically distributed variables, the sample mean $\bar{x} = S_n/n$ satisfies $\sigma(\bar{x}) = \sigma(x)/\sqrt{n}$: averaging $n$ measurements reduces the standard deviation by $\sqrt{n}$.

\subsection{Convolutions}

The density in \eqref{eq:sum_dist} is an $n$-fold convolution. For $n=2$:
\begin{equation}
    p(S_2) = \int dx\; p_1(x)\, p_2(S_2 - x) \equiv (p_1 \circledast p_2)(S_2).
\end{equation}

\begin{definition}[Convolution]
\begin{equation}
    (f \circledast g)(z) = \int dx\; f(x)\, g(z - x).
\end{equation}
\end{definition}

\begin{fact}
The PDF of the sum of independent random variables is the convolution of their individual PDFs.
\end{fact}

Three families closed under convolution: \textbf{Gaussians} (sum is Gaussian with additive mean and variance); \textbf{Poissonians} (sum is Poissonian); \textbf{Lorentzians}
\begin{equation}
    p(x) = \frac{1}{\pi}\,\frac{\Gamma}{(x-a)^2 + \Gamma^2},
\end{equation}
which has \emph{infinite} variance — so the Lorentzian is excluded from the CLT.

\subsection{The Central Limit Theorem}

\begin{theorem}[Central Limit Theorem]
Let $S_n = x_1 + \cdots + x_n$ be a sum of i.i.d.\ random variables with mean $\langle x\rangle$ and finite variance $\sigma^2(x)$. As $n \to \infty$:
\begin{equation}
    p(S_n) \xrightarrow{n\to\infty} \frac{1}{\sqrt{2\pi n\,\mathrm{Var}(x)}}\;\exp\!\left(-\frac{(S_n - n\langle x\rangle)^2}{2n\,\mathrm{Var}(x)}\right).
\end{equation}
Finite variance is essential; distributions violating it (e.g.\ Lorentzian) do not obey the CLT.
\end{theorem}

\begin{example}[Total energy of an ideal gas]
For a 1D ideal gas atom, $\langle E\rangle = \tfrac{1}{2}k_B T$ and $\mathrm{Var}(E) = \tfrac{1}{4}(k_B T)^2$. For $N$ atoms, $\langle E_{\text{tot}}\rangle = \tfrac{1}{2}Nk_B T$ and by the CLT:

\begin{center}

  \draw[->] (-3.5,0) -- (3.5,0) node[right] {$E_{\text{tot}}$};
  \draw[->] (0,-0.2) -- (0,2.8) node[above] {$p(E_{\text{tot}})$};
  \draw[thick,domain=-3.2:3.2,smooth,samples=100]
    plot (\x, {2.5*exp(-\x*\x/0.04)});
  \draw[dashed] (0,0) -- (0,2.5);
  \node[below] at (0,0) {$\langle E_{\text{tot}}\rangle$};

\end{tikzpicture}\\[3pt]

\end{center}

\begin{equation}
    \frac{\sigma(E_{\text{tot}})}{\langle E_{\text{tot}}\rangle} = \frac{1}{\sqrt{N}} \approx 10^{-12} \quad \text{for } N \sim 10^{24}.
\end{equation}
For $\sim10^{24}$ particles the total energy is essentially deterministic — this ``concentration of measure'' is why thermodynamics deals in certainties.
\end{example}

\section{The Gamma Function}
% ============================================================

\begin{definition}[Gamma function]
\begin{equation}
    \Gamma(n) = \int_0^\infty x^{n-1} e^{-x}\,dx.
\end{equation}
\end{definition}
Defined for $n > 0$, extended by analytic continuation to $\mathbb{C}$ excluding non-positive integers.

\subsection{Recurrence Relation and Connection to Factorials}

Integration by parts with $u = x^{n+1}$, $dv = e^{-x}dx$ gives $\Gamma(n+2) = (n+1)\Gamma(n+1)$, and $\Gamma(1) = 1$, so by induction:
\begin{formula}
\begin{equation}
    \Gamma(n+1) = n!, \qquad n = 0, 1, 2, \ldots
\end{equation}
\end{formula}
A useful corollary (from differentiating $\int_0^\infty e^{-tx}\,dx = 1/t$ under the integral):
\begin{formula}
\begin{equation}
    \int_0^\infty x^n e^{-tx}\,dx = \frac{n!}{t^{n+1}}, \qquad t > 0.
\end{equation}
\end{formula}

\subsection{The Special Value $\Gamma(1/2) = \sqrt{\pi}$}

Substituting $u = \sqrt{x}$:
\begin{equation}
    \Gamma\!\left(\tfrac{1}{2}\right) = 2\int_0^\infty e^{-u^2}\,du \equiv 2I.
\end{equation}
Setting $f(t) = \int_0^\infty e^{-t^2(1+x^2)}/(1+x^2)\,dx$ and differentiating: $f'(t) = -2e^{-t^2}I$. Integrating from $0$ to $\infty$: $f(\infty) - f(0) = -2I^2$, with $f(\infty) = 0$ and $f(0) = \pi/4$, giving $I^2 = \pi/4$, hence:
\begin{formula}
\begin{equation}
    \Gamma\!\left(\tfrac{1}{2}\right) = \sqrt{\pi}.
\end{equation}
\end{formula}
Corollaries: $\Gamma(\tfrac{3}{2}) = \tfrac{1}{2}\sqrt{\pi}$, $\Gamma(-\tfrac{1}{2}) = -2\sqrt{\pi}$.

\subsection{Stirling's Approximation}

Expanding the integrand of $\Gamma(n+1)$ around the saddle point $x = n$:
\begin{formula}[Stirling's approximation]
\begin{equation}
    \log n! \approx n\log n - n + \tfrac{1}{2}\log(2\pi n).
\end{equation}
\end{formula}
In statistical mechanics with $n \sim 10^{23}$, the correction $\tfrac{1}{2}\log(2\pi n)$ is negligible and $\log n! \approx n\log n - n$ suffices.

\subsection{Measuring Randomness}

An \emph{ensemble} $\{S_i, p_i\}$ specifies symbols and their probabilities. The natural measure of randomness is the minimum average number of bits needed to specify a drawn symbol. If one outcome is certain, zero bits suffice; if all outcomes are equally likely, more information is required. Variable-length codes assign shorter codewords to more probable symbols. Shannon's theorem identifies the fundamental limit.

\begin{theorem}[Shannon's Theorem]
The minimum average bits per symbol from an ensemble $\{S_i, p_i\}$ is
\begin{equation}
    \sigma = -\sum_{j=1}^{N} p_j \log_2 p_j.
\end{equation}
\end{theorem}

\begin{example}[Biased coin]
For $p_A = \tfrac{3}{4}$, $p_B = \tfrac{1}{4}$: $\sigma = -\tfrac{3}{4}\log_2\tfrac{3}{4} - \tfrac{1}{4}\log_2\tfrac{1}{4} \approx 0.811$ bits/symbol. Encoding pairs with the prefix-free code ($AA\to0$, $AB\to10$, $BA\to110$, $BB\to111$) achieves $27/16 \approx 1.688$ bits/pair $\approx 0.844$ bits/symbol.

\subsection{Derivation of the Shannon Formula}

For an alphabet of $N$ letters in a message of length $M\gg N$, with letter $j$ occurring $n_j$ times, the number of distinct messages with the same frequency composition is
\begin{equation}
    \Omega = \frac{M!}{n_1!\, n_2!\, \cdots\, n_N!}.
\end{equation}
Transmitting one particular message requires $B = \log_2\Omega$ bits. Applying Stirling's approximation:
\begin{align}
    B \log 2 &= \log M! - \sum_j \log n_j!
    \approx M\log M - \sum_j n_j \log n_j
    = -M\sum_j \frac{n_j}{M}\log\frac{n_j}{M}.
\end{align}
With $p_j = n_j/M$, the minimum bits per symbol is
\begin{equation}
    \frac{B}{M} = -\sum_j p_j \log_2 p_j = \sigma.
\end{equation}

\subsection{Properties of Shannon Entropy}

\begin{definition}[Shannon Entropy]
For an ensemble $\{S_i, p_i\}_{i=1}^{N}$, the Shannon entropy (in bits) is
\begin{equation}
    \sigma = -\sum_{j=1}^{N} p_j \log_2 p_j.
\end{equation}
\end{definition}

Minimum $\sigma = 0$ when one outcome is certain; maximum $\sigma = \log_2 N$ at the uniform distribution $p_j = 1/N$.

\begin{fact}
The uniform distribution maximises Shannon entropy subject to normalization.
\end{fact}
This follows from the Lagrange condition $-(\log p_i + 1) + \lambda = 0$, which is $i$-independent, giving $p_i = 1/N$.

\subsection{Connection to Thermodynamic Entropy}

In thermodynamics, a macrostate $(N, V, U)$ is compatible with many microstates. The thermodynamic entropy is
\begin{equation}
    S = -k_B \sum_{j=1}^{\Omega} p_j \log p_j = k_B \ln 2\cdot \sigma,
\end{equation}
with natural logarithms and the prefactor $k_B$ chosen so that $TS$ has dimensions of energy.

The equilibrium $\{p_j\}$ maximises $S$ subject to the macroscopic constraints (ergodic hypothesis). For an isolated system with fixed $N$, $V$, $U$, all $\Omega$ compatible microstates are equally probable, giving:
\begin{formula}[Boltzmann's Entropy]
\begin{equation}
    S = k_B \log \Omega,
\end{equation}
where $\Omega$ is the multiplicity of the macrostate.
\end{formula}
The general relation $S = -k_B\sum_j p_j\log p_j$ is more fundamental; Boltzmann's formula is its special case for isolated equilibrium systems.

Entropy is \emph{extensive}: $\Omega_{\mathrm{tot}} = \Omega_A\cdot\Omega_B$ for independent subsystems gives $S = S_A + S_B$, since the logarithm converts multiplicative counting into an additive quantity.
