\section{Microstates, Macrostates, and Multiplicity}
% ============================================================

Irreversible processes are not logically forbidden --- they are simply overwhelmingly probable. To understand why, we count the number of ways a system can be arranged.

\begin{definition}
A \textbf{microstate} is a complete specification of the system. A \textbf{macrostate} is a coarser description specifying only macroscopic quantities. The \textbf{multiplicity} $\Omega$ of a macrostate is the number of microstates consistent with it.
\end{definition}

For $N$ distinguishable objects of which $n$ are in a designated state:
\[
    \Omega(N, n) = \binom{N}{n} = \frac{N!}{n!\,(N-n)!}.
\]

\begin{example}
Three coins: ``two heads'' has $\Omega(3,2) = 3$ (HHT, HTH, THH); ``zero heads'' has $\Omega(3,0) = 1$. For large $N$ such disparities become astronomical.
\end{example}

\begin{definition}[Fundamental Assumption of Statistical Mechanics]
In an isolated system in thermal equilibrium, every accessible microstate is equally probable.
\end{definition}

This implies any process $X \to Y$ and its time-reverse occur at the same rate. Macrostates with more microstates are simply more probable.

\subsection{Two-State Paramagnet}

In the idealised \textbf{two-state paramagnet}, each dipole points either parallel ($\uparrow$) or antiparallel ($\downarrow$) to an applied field. With $N$ total dipoles and $N_\uparrow$ up-spins, $N_\downarrow = N - N_\uparrow$, and the multiplicity is
\[
    \Omega(N, N_\uparrow) = \binom{N}{N_\uparrow} = \frac{N!}{N_\uparrow!\, N_\downarrow!}.
\]
Specifying $N_\uparrow$ determines both the energy and the multiplicity.

The number of ways to distribute $q$ indistinguishable energy units among $N$ oscillators is a stars-and-bars problem:

\begin{formula}[Multiplicity of an Einstein Solid]
\[
    \Omega(N,q) = \binom{q+N-1}{q} = \frac{(q+N-1)!}{q!\,(N-1)!}.
\]
\end{formula}

\begin{example}
Check: $\Omega(4,5) = \binom{8}{5} = 56$.
\end{example}

\subsection{Interacting Einstein Solids and Equilibrium}

Two Einstein solids A and B in thermal contact share total energy $q = q_A + q_B$. Their multiplicities multiply:

\[
    \Omega_A = \binom{q_A + N_A - 1}{q_A}, \qquad
    \Omega_B = \binom{q_B + N_B - 1}{q_B}, \qquad
    \Omega_{\text{tot}} = \Omega_A \cdot \Omega_B.
\]
The distribution peaks sharply near $q_A = q_B = q/2$. This is the statistical content of the second law: systems evolve toward the macrostate of greatest multiplicity.

\section{Large Systems and Stirling's Approximation}
% ============================================================

Using Stirling's approximation ($\ln N! \approx N\ln N - N$), we obtain closed-form expressions for $\ln\Omega$ in limiting cases.

\subsection{High-Temperature Limit ($q \gg N$)}

When $q \gg N$, use $\ln(q+N) \approx \ln q + N/q$ and Stirling's approximation:
\begin{align*}
    \ln\Omega
    &= (q+N)\ln(q+N) - q\ln q - N\ln N + \cdots \\
    &\approx N\ln\!\left(\frac{q}{N}\right) + N.
\end{align*}

\begin{formula}[Einstein Solid: High-Temperature Limit]
For $q \gg N$,
\[
    \Omega(N,q) \approx \left(\frac{eq}{N}\right)^N.
\]
\end{formula}

\subsection{Low-Temperature Limit ($q \ll N$)}

For $q \ll N$, similarly:
\[
    \Omega(N,q) \approx \left(\frac{eN}{q}\right)^q.
\]

\subsection{Sharpness of the Multiplicity Peak}

\begin{example}
Two equal solids ($N$ oscillators each) sharing $q$ units in the high-$T$ limit:
\[
    \Omega_{\text{tot}} = \left(\frac{e}{N}\right)^{2N}(q_A q_B)^N,
\]
maximised at $q_A = q_B = q/2$:
\[
    \Omega_{\max} = \left(\frac{eq}{2N}\right)^{2N}.
\]
Writing $q_A = q/2 + x$, $q_B = q/2 - x$:

\begin{formula}[Gaussian Form of the Multiplicity Peak]
\[
    \Omega_{\text{tot}}(x) = \Omega_{\max}\, \exp\!\left[-N\!\left(\frac{2x}{q}\right)^2\right].
\]
\end{formula}

The $1/e$ half-width is $\Delta x = q/\sqrt{N}$, giving relative width $\Delta x/q = 1/\sqrt{N}$. For $N \sim 10^{23}$, this is $\sim 10^{-11.5}$: the peak is vanishingly narrow. The second law is a statement about probability in the thermodynamic limit.
\end{example}

\section{Density of States and Semi-Classical Phase Space}
% ============================================================

For a continuous system like an ideal gas, we need a semi-classical procedure to count states.

\subsection{State Counting and One Quantum State Per $h^3$}

For a single particle in a box of side $L$, energy eigenvalues are
\[
    E(n_1, n_2, n_3) = \frac{\pi^2\hbar^2}{2mL^2}(n_1^2 + n_2^2 + n_3^2).
\]
Since level spacing $\ll k_B T$, replace the lattice sum by an integral over the positive octant:
\[
    N(E) = \frac{1}{8} \cdot \frac{4\pi}{3} C^3 = \frac{1}{6\pi^2}\left(\frac{2m}{\hbar^2}\right)^{3/2} V E^{3/2},
\]
where $V = L^3$. The density of states:
\begin{formula}
\[
    \rho(E) \equiv \dv{N}{E} = \frac{1}{4\pi^2}\left(\frac{2m}{\hbar^2}\right)^{3/2} V\sqrt{E}.
\]
\end{formula}

\subsection{Semi-Classical Phase Space and the $h^3$ Cell}

Matching the classical phase-space integral to the quantum count determines the volume per state:
\[
    N(E) = \frac{1}{\Delta}\int d^3x\int d^3p\;\Theta\!\bigl(E - E(\vb{x}, \vb{p})\bigr)
         = \frac{V}{\Delta}\cdot\frac{4\pi}{3}(2mE)^{3/2}.
\]

\begin{formula}[Phase Space Volume Per Quantum State]
\[
    \Delta = (2\pi\hbar)^3 = h^3.
\]
\end{formula}
This follows from the Heisenberg uncertainty principle: $\Delta x\,\Delta p_x \sim \hbar$ per direction.

\subsection{General $N$-Particle System}

For $N$ particles, each state occupies $h^{3N}$ in $6N$-dimensional phase space:
\begin{formula}
\[
    N(U) = \frac{1}{(2\pi\hbar)^{3N}}\int d^{3N}x\int d^{3N}p\;\Theta\!\bigl(U - H(\vb{x}, \vb{p})\bigr).
\]
\end{formula}
For identical particles, divide by $N!$ to avoid over-counting permutations.

\section{The Multiplicity of an Ideal Gas: Sackur-Tetrode}
% ============================================================

\subsection{Volume of the $3N$-Dimensional Momentum Sphere}

For $N$ non-interacting particles, the spatial integral gives $V^N$ and the momentum integral is a $3N$-dimensional ball of radius $\sqrt{2mU}$:
\[
    \Omega(U, V, N) \approx \frac{1}{N!}\cdot\frac{V^N}{h^{3N}}\cdot\frac{\pi^{3N/2}}{\left(\tfrac{3N}{2}\right)!}\,(2mU)^{3N/2}.
\]

\subsection{Derivation of the Sackur-Tetrode Equation}

Applying Stirling's approximation to $\ln N!$ and $\ln\!\left(\tfrac{3N}{2}\right)!$:
\begin{align*}
    \ln\Omega
    &= N\!\left[-\ln N + 1 + \ln V - 3\ln h + \frac{3}{2}\ln\pi + \frac{3}{2}\ln(2mU)
       - \frac{3}{2}\ln\frac{3N}{2} + \frac{3}{2}\right] \\
    &= N\!\left[\ln\frac{V}{N} + \frac{3}{2}\ln\!\left(\frac{4\pi mU}{3Nh^2}\right) + \frac{5}{2}\right].
\end{align*}

\begin{formula}[Sackur-Tetrode Equation]
\[
    \boxed{S(N,V,U) = Nk\!\left[\ln\!\left(\frac{V}{N}\left(\frac{4\pi m U}{3Nh^2}\right)^{3/2}\right) + \frac{5}{2}\right].}
\]
\end{formula}

With species-dependent constant $c_0 = \tfrac{3}{2}\ln\!\left(\tfrac{4\pi m}{3h^2}\right) + \tfrac{5}{2}$:
\[
    S = Nk\!\left[\ln\frac{V}{N} + \frac{3}{2}\ln\frac{U}{N} + c_0\right].
\]

\subsection{The Gibbs Paradox and the $1/N!$ Factor}

Without the $1/N!$ factor, doubling the system $(U,V,N) \to (2U,2V,2N)$ gives
\[
    S(2U, 2V, 2N) = 2S(U, V, N) + 2kN\ln 2 \neq 2S(U,V,N).
\]
The spurious $2kN\ln 2$ is the \textbf{Gibbs paradox}. The resolution: identical particles are genuinely indistinguishable, so swapping two produces no new microstate. With the correct $1/N!$, the Sackur-Tetrode entropy is exactly extensive.

\section{Entropy: $S = k\ln\Omega$}
% ============================================================

\begin{definition}[Boltzmann's Entropy]
\[
    S \equiv k\ln\Omega,
\]
with $k = 1.38\times10^{-23}$~J/K. Since $\Omega_{\text{tot}} = \Omega_A\Omega_B$, entropy is \textbf{additive}: $S_{\text{tot}} = S_A + S_B$.
\end{definition}

\begin{theorem}[Second Law of Thermodynamics]
The total entropy of an isolated system tends to increase, because a randomly chosen microstate overwhelmingly belongs to a high-multiplicity macrostate.
\end{theorem}

\subsection{Entropy of Mixing}

Two different ideal gases initially at the same $T$ and $P$ in volumes $(1-x)V$ and $xV$ mix into volume $V$:
\[
    \Delta S_A = (1-x)Nk\ln\frac{1}{1-x}, \qquad \Delta S_B = xNk\ln\frac{1}{x}.
\]

\begin{formula}[Entropy of Mixing]
\[
    \Delta S_{\text{mix}} = -Nk\bigl[(1-x)\ln(1-x) + x\ln x\bigr].
\]
\end{formula}

Non-negative for all $0 \leq x \leq 1$; vanishes only for pure species. Equivalently, $\Delta S_{\text{mix}} = k\ln\binom{N}{N_A}$ by Stirling's approximation.

\section{Temperature and Pressure from Entropy}
\label{sec:temperature-from-entropy}
% ============================================================

\subsection{The Statistical Definition of Temperature}

Two subsystems A and B in thermal contact with $U_A + U_B$ fixed. The condition $\partial S_{\text{tot}}/\partial U_A = 0$ gives
\[
    \left.\pdv{S_A}{U_A}\right|_{V_A,N_A} = \left.\pdv{S_B}{U_B}\right|_{V_B,N_B}.
\]

A steeper $\partial S/\partial U$ corresponds to a colder object (more entropy per unit energy received).

\begin{definition}[Statistical Definition of Temperature]
\[
    \frac{1}{T} \equiv \left(\pdv{S}{U}\right)_{V,N}.
\]
\end{definition}

\subsection{Pressure from Entropy}

For systems exchanging volume at fixed total $V$ and fixed energies:
\begin{formula}
\[
    \frac{P}{T} = \left(\pdv{S}{V}\right)_{U,N}.
\]
\end{formula}

\subsection{Verification Against the Ideal Gas}

\begin{example}
From the Sackur-Tetrode entropy, $S \propto \tfrac{3}{2}Nk\ln U$:
\[
    \frac{1}{T} = \left(\pdv{S}{U}\right)_{V,N} = \frac{3Nk}{2U}
    \qquad\Longrightarrow\qquad \boxed{U = \tfrac{3}{2}NkT.}
\]
From $S \propto Nk\ln V$:
\[
    \frac{P}{T} = \left(\pdv{S}{V}\right)_{U,N} = \frac{Nk}{V}
    \qquad\Longrightarrow\qquad \boxed{PV = NkT.}
\]
Both laws emerge from microstate counting alone.
\end{example}

\subsection{Connecting to Clausius: $dS = \dbar Q/T$}

At constant $V$, $dU = \dbar Q$ and $dS = dU/T = \dbar Q/T$ --- exactly the Clausius entropy. For a reversible process:
\begin{formula}
\[
    \Delta S = \int_{T_i}^{T_f}\frac{C_V}{T}\,dT.
\]
\end{formula}

\subsection{The Zeroth Law as a Corollary}

\begin{theorem}[Zeroth Law of Thermodynamics]
If $A$ is in equilibrium with $C$, and $B$ is in equilibrium with $C$, then $A$ and $B$ are in mutual equilibrium. This follows directly: $(\partial S_A/\partial U_A)_{V,N} = (\partial S_C/\partial U_C)_{V,N}$ and $(\partial S_B/\partial U_B)_{V,N} = (\partial S_C/\partial U_C)_{V,N}$ imply $(\partial S_A/\partial U_A)_{V,N} = (\partial S_B/\partial U_B)_{V,N}$.
\end{theorem}

The Third Law also emerges: at $T \to 0$, $\Omega \to 1$ and $S \to k\ln 1 = 0$.

\section{Paramagnetism}
% ============================================================

\subsection{Setup and Multiplicity}

$N$ non-interacting spin-$\tfrac{1}{2}$ dipoles in field $B$, with energies
\[
    \varepsilon_{\uparrow} = -\mu B, \qquad \varepsilon_{\downarrow} = +\mu B.
\]
Total energy and magnetization:
\[
    U = \mu B(N - 2N_\uparrow), \qquad M \equiv \mu(N_\uparrow - N_\downarrow) = -\frac{U}{B}.
\]

\begin{formula}
\[
    \Omega(N_\uparrow) = \binom{N}{N_\uparrow} = \frac{N!}{N_\uparrow!\,N_\downarrow!}.
\]
\end{formula}

\subsection{Temperature, Energy, and Magnetization}

Using Stirling's approximation on $S = k\ln\Omega$ and differentiating with respect to $U$:
\[
    \frac{1}{T} = \pdv{S}{U} = \frac{k}{2\mu B}\ln\frac{N_\downarrow}{N_\uparrow}.
\]
Solving gives $N_\uparrow/N_\downarrow = e^{2\mu B/kT}$, hence:

\begin{formula}
\[
    U(T) = -N\mu B\tanh\!\left(\frac{\mu B}{kT}\right), \qquad
    M(T) = N\mu\tanh\!\left(\frac{\mu B}{kT}\right).
\]
\end{formula}

\subsection{Limiting Behaviour}

\begin{example}
At $kT \ll \mu B$: $\tanh \to 1$, so $U \to -N\mu B$ and $M \to N\mu$ (full saturation). At $kT \gg \mu B$: $\tanh(x) \approx x$, giving the \textbf{Curie law}:
\[
    M \approx \frac{N\mu^2 B}{kT}.
\]
\end{example}

\subsection{Entropy of the Paramagnet}

Substituting $N_\uparrow/N = [1 + \tanh(\mu B/kT)]/2$ into $S = k\ln\Omega$:
\[
    \frac{S}{Nk} = -\frac{1+t}{2}\ln\frac{1+t}{2} - \frac{1-t}{2}\ln\frac{1-t}{2}, \quad t \equiv \tanh\!\left(\frac{\mu B}{kT}\right).
\]
At $T = 0$: $\Omega = 1$, so $S = 0$ (Third Law). At $T \to \infty$: $\Omega \to 2^N$, so $S \to Nk\ln 2$.
