\chapter{Thermodynamic Potentials and Free Energy}

\section{Spontaneous Processes at Fixed Constraints}

The second law states $\Delta S \geq 0$ for an isolated system, with equilibrium at maximum $S$. Most laboratory systems are not isolated — they contact a reservoir at fixed $T$. Different constraints yield different free energies, each giving a criterion for spontaneous change intrinsic to the system.

\subsection*{At Fixed $T$ and $V$: Helmholtz Free Energy}

For a system at fixed $V$ in contact with a reservoir at $T$: energy conservation gives $\Delta Q_{\text{in}} = \Delta U + W$, so $\Delta S_{\text{bath}} = -(\Delta U + W)/T$. The total entropy condition $\Delta S_{\text{bath}} + \Delta S_{\text{sys}} \geq 0$ gives:
\begin{align}
    -\frac{\Delta U + W}{T} + \Delta S &\geq 0 \notag \\
    \Rightarrow\quad \Delta U - T\Delta S + W &\leq 0.
\end{align}

\begin{definition}[Helmholtz Free Energy]
\begin{equation}
    F \;\defeq\; U - TS.
\end{equation}
\end{definition}

The spontaneity condition becomes $\Delta F + W \leq 0$:
\begin{formula}
\begin{equation}
    W \leq -\Delta F.
\end{equation}
\end{formula}
Maximum isothermal work is $-\Delta F$. With $W = 0$, $\Delta F \leq 0$: the system drifts to lower $F$.

\begin{fact}
At fixed $T$ and $V$ with no external work, the system evolves to minimum $F$. The competition $F = U - TS$ is weighted by $T$: energy dominates at low $T$, entropy at high $T$.
\end{fact}

\subsection*{At Fixed $T$ and $P$: Gibbs Free Energy}

With both $T$ and $P$ fixed, energy conservation gives $\Delta Q_{\text{in}} = \Delta U + P\Delta V + W_{\text{other}}$. The total entropy condition becomes:
\begin{equation}
    \frac{-\Delta U - P\Delta V - W_{\text{other}}}{T} + \Delta S \geq 0
    \quad\Rightarrow\quad
    \Delta(U + PV - TS) + W_{\text{other}} \leq 0.
\end{equation}

\begin{definition}[Gibbs Free Energy]
\begin{equation}
    G \;\defeq\; U + PV - TS \;=\; H - TS \;=\; F + PV,
\end{equation}
where $H = U + PV$ is the enthalpy.
\end{definition}

\begin{formula}
\begin{equation}
    W_{\text{other}} \leq -\Delta G.
\end{equation}
\end{formula}
Maximum non-$pV$ work is $-\Delta G$. With no other work, $\Delta G \leq 0$.

\begin{fact}
At fixed $T$ and $P$, the system evolves to minimum $G$. Phase transitions occur when $G$ of one phase falls below another (e.g.\ water freezes below $0\,^\circ\text{C}$ at atmospheric pressure).
\end{fact}

\section{The Four Thermodynamic Potentials}

The four potentials arise from $U$ by Legendre transforms on $dU = T\,dS - P\,dV + \mu\,dN$, swapping extensive natural variables for their conjugate intensive ones.

\begin{definition}[Chemical Potential]
\begin{equation}
    \mu \;\defeq\; \left.\pdv{U}{N}\right|_{S,V} = \left.\pdv{G}{N}\right|_{T,P}.
\end{equation}
\end{definition}

\begin{center}
\begin{tabular}{llll}
    \hline
    Potential & Definition & Natural variables & Exact differential \\
    \hline
    Internal energy $U$ & --- & $S,\,V,\,N$ & $dU = T\,dS - P\,dV + \mu\,dN$ \\[4pt]
    Enthalpy $H$ & $U + PV$ & $S,\,P,\,N$ & $dH = T\,dS + V\,dP + \mu\,dN$ \\[4pt]
    Helmholtz free energy $F$ & $U - TS$ & $T,\,V,\,N$ & $dF = -S\,dT - P\,dV + \mu\,dN$ \\[4pt]
    Gibbs free energy $G$ & $U - TS + PV$ & $T,\,P,\,N$ & $dG = -S\,dT + V\,dP + \mu\,dN$ \\
    \hline
\end{tabular}
\end{center}

The conjugate relationships:
\begin{align}
    T = \left.\pdv{U}{S}\right|_{V,N} = \left.\pdv{H}{S}\right|_{P,N}, \qquad
    P = -\left.\pdv{U}{V}\right|_{S,N} = -\left.\pdv{F}{V}\right|_{T,N}, \\[6pt]
    V = \left.\pdv{H}{P}\right|_{S,N} = \left.\pdv{G}{P}\right|_{T,N}, \qquad
    S = -\left.\pdv{F}{T}\right|_{V,N} = -\left.\pdv{G}{T}\right|_{P,N}.
\end{align}

\section{Maxwell Relations}

Since all potentials are state functions, mixed second partials commute, yielding one Maxwell relation per potential (at fixed $N$):
\begin{align}
    dU = T\,dS - P\,dV \quad&\Rightarrow\quad \left.\pdv{T}{V}\right|_S = -\left.\pdv{P}{S}\right|_V, \tag{from $U$} \\[6pt]
    dH = T\,dS + V\,dP \quad&\Rightarrow\quad \left.\pdv{T}{P}\right|_S = \phantom{+}\left.\pdv{V}{S}\right|_P, \tag{from $H$} \\[6pt]
    dF = -S\,dT - P\,dV \quad&\Rightarrow\quad \left.\pdv{S}{V}\right|_T = \phantom{+}\left.\pdv{P}{T}\right|_V, \tag{from $F$} \\[6pt]
    dG = -S\,dT + V\,dP \quad&\Rightarrow\quad \left.\pdv{S}{P}\right|_T = -\left.\pdv{V}{T}\right|_P. \tag{from $G$}
\end{align}

\begin{theorem}[Maxwell Relations]
\begin{align}
    \left.\pdv{T}{V}\right|_S = -\left.\pdv{P}{S}\right|_V, \qquad
    \left.\pdv{T}{P}\right|_S &= \left.\pdv{V}{S}\right|_P, \notag \\[6pt]
    \left.\pdv{S}{V}\right|_T = \left.\pdv{P}{T}\right|_V, \qquad
    \left.\pdv{S}{P}\right|_T &= -\left.\pdv{V}{T}\right|_P.
\end{align}
Derive each relation as needed from the relevant differential rather than memorising signs.
\end{theorem}

These relations connect experimentally inaccessible quantities to measurable ones.

\subsection*{Application: Internal Pressure}

\begin{example}
From $dU = T\,dS - P\,dV$, dividing by $dV$ at constant $T$:
\begin{equation}
    \left.\pdv{U}{V}\right|_T = T\left.\pdv{S}{V}\right|_T - P.
\end{equation}
The Maxwell relation from $dF$ gives $\left.\pdv{S}{V}\right|_T = \left.\pdv{P}{T}\right|_V$, so:

\begin{formula}
\begin{equation}
    \left.\pdv{U}{V}\right|_T = T\left.\pdv{P}{T}\right|_V - P.
\end{equation}
\end{formula}

For an ideal gas, $\left.\pdv{P}{T}\right|_V = P/T$, giving zero internal pressure — $U$ is independent of $V$. For real gases with attractive interactions, the internal pressure is positive.
\end{example}

\section{Example: The Hydrogen Fuel Cell}

\begin{example}
A hydrogen fuel cell at fixed $T$ and $P$ converts chemical energy to electrical work. The reaction:
\begin{equation}
    \mathrm{H_2} + \tfrac{1}{2}\mathrm{O_2} \;\longrightarrow\; \mathrm{H_2O}.
\end{equation}
At $T = 300\,\mathrm{K}$:
\begin{align}
    \Delta H &= -286\,\mathrm{kJ\,mol^{-1}}, \\
    \Delta S &= -163\,\mathrm{J\,K^{-1}\,mol^{-1}}.
\end{align}
\begin{equation}
    \Delta G = \Delta H - T\Delta S = -286 - (300)(-0.163) \approx -237\,\mathrm{kJ\,mol^{-1}}.
\end{equation}
Of $286\,\mathrm{kJ\,mol^{-1}}$ released, only $237\,\mathrm{kJ\,mol^{-1}}$ is available as work; the rest is expelled as heat ($\Delta S < 0$). Efficiency:
\begin{equation}
    \eta_{\mathrm{fc}} = \frac{|\Delta G|}{|\Delta H|} = \frac{237}{286} \approx 83\%.
\end{equation}

For comparison, burning hydrogen in a Carnot engine between $T_+ \approx 1760\,\mathrm{K}$ and $T_- = 300\,\mathrm{K}$:
\begin{equation}
    \eta_{\mathrm{Carnot}} = 1 - \frac{T_-}{T_+} = 1 - \frac{300}{1760} \approx 83\%.
\end{equation}
The same efficiency, but from different physics: the Carnot bound depends on temperature ratios; the fuel cell bound on $|\Delta G|/|\Delta H|$.
\end{example}
