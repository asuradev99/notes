\chapter{The Canonical Ensemble}

\section{Derivation of the Boltzmann Distribution}

Systems held at fixed temperature rather than fixed energy are described by the \emph{canonical ensemble}. Consider system $S$ in thermal contact with reservoir $R$ at temperature $T$. The combined system is isolated, so all total microstates are equally probable:
\begin{equation}
    U_{\text{tot}} = U_R + U_S, \qquad S_{\text{tot}} = S_R + S_S.
\end{equation}

\begin{center}

  \node[align=center] at (1.9,1.2) {Reservoir at $T$};
  \draw[thick, fill=orange!15] (5.4,0.5) rectangle (7.8,1.9);
  \node[align=center, font=\small] at (6.6,1.2) {System $S$\\energy $E_j$};
  \draw[decorate, decoration={snake, amplitude=3pt, segment length=6pt},

    (3.95,1.2) -- (5.25,1.2)
    node[midway, above=4pt, font=\small, black] {heat};
\end{tikzpicture}\\[3pt]
{\small\itshape Canonical ensemble: system $S$ in thermal contact with a large reservoir at $T$.}
\end{center}

When the system is in microstate $j$ (energy $E_j$), the reservoir occupies $\Omega_R(U_{\text{tot}} - E_j)$ states, so $P_j \propto \Omega_R(U_{\text{tot}} - E_j) = e^{S_R(U_{\text{tot}}-E_j)/k_B}$. Since $E_j \ll U_{\text{tot}}$, Taylor-expand $S_R$ to first order:
\begin{equation}
    S_R\!\left(U_{\text{tot}} - E_j\right)
    \approx S_R(U_{\text{tot}}) - \frac{E_j}{T},
\end{equation}
using $(\partial S_R/\partial U)_V = 1/T$. Absorbing the constant into normalisation gives the \textbf{Boltzmann factor}:

\begin{formula}[Boltzmann factor]
\begin{equation}
    P_j \;\propto\; e^{-E_j / k_B T}.
    \label{eq:boltzmann_factor}
\end{equation}
\end{formula}

\subsection*{The Partition Function and Normalisation}

Writing $\beta \equiv 1/(k_B T)$, normalisation $\sum_j P_j = 1$ defines:

\begin{formula}[Partition function]
\begin{equation}
    Z(T, V, N) \;=\; \sum_j e^{-\beta E_j}, \qquad P_j = \frac{e^{-\beta E_j}}{Z}.
    \label{eq:partition_function}
\end{equation}
\end{formula}

High-energy states are exponentially suppressed by $E_j/k_BT$. With degeneracy $g(E_i)$ or a continuous spectrum, the sum becomes an integral over the density of states. For a single classical particle in 3D:
\begin{equation}
    Z_1 \;=\; \int \frac{d^3x\,d^3p}{(2\pi\hbar)^3}\;e^{-\beta E(\vec{x},\vec{p})}.
\end{equation}

\begin{definition}[Canonical ensemble]
The \textbf{canonical ensemble} assigns probability $P_j = e^{-\beta E_j}/Z$ to each microstate of a system in equilibrium with a heat reservoir at $T$. All energies are included (unlike microcanonical).
\end{definition}

\section{Thermodynamics from the Partition Function}

\subsection*{Internal energy}

Since $E_j e^{-\beta E_j} = -\partial e^{-\beta E_j}/\partial\beta$:

\begin{formula}[Internal energy from $Z$]
\begin{equation}
    U = \langle E \rangle = -\pdv{\log Z}{\beta}.
    \label{eq:U_from_Z}
\end{equation}
\end{formula}

\subsection*{Helmholtz free energy and entropy}

From the Gibbs entropy $S = -k_B \sum_j P_j \log P_j$ with $\log P_j = -\beta E_j - \log Z$:
\begin{equation}
    S = \frac{U}{T} + k_B \log Z,
\end{equation}
which immediately gives:

\begin{formula}[Helmholtz free energy]
\begin{equation}
    F = U - TS = -k_B T \log Z.
    \label{eq:F_from_Z}
\end{equation}
\end{formula}

All remaining quantities follow from $dF = -S\,dT - P\,dV$:

\begin{formula}[Entropy and pressure from $Z$]
\begin{equation}
    S = -\left.\pdv{F}{T}\right|_V = \frac{\partial(k_B T \log Z)}{\partial T}\bigg|_V,
    \qquad
    P = -\left.\pdv{F}{V}\right|_T.
\end{equation}
\end{formula}

\subsection*{Heat capacity}

Using $\partial/\partial T = -k_B \beta^2 \,\partial/\partial\beta$:

\begin{formula}[Heat capacity from $Z$]
\begin{equation}
    C_V = \left.\pdv{U}{T}\right|_V = k_B\beta^2\,\pdv[2]{\log Z}{\beta}.
\end{equation}
\end{formula}

\subsection*{Energy fluctuations}

Differentiating $U = -\partial\log Z/\partial\beta$ again gives $\partial^2\log Z/\partial\beta^2 = \langle E^2\rangle - \langle E\rangle^2$, so:

\begin{formula}[Energy fluctuations]
\begin{equation}
    \mathrm{Var}(E) = k_B T^2 C_V.
    \label{eq:var_E}
\end{equation}
\end{formula}

\begin{example}
For an ideal gas ($C_V = \tfrac{3}{2}Nk_B$, $U = \tfrac{3}{2}Nk_BT$):
\begin{equation}
    \frac{\sigma(U)}{U} = \sqrt{\frac{3}{2}}\cdot\frac{1}{\sqrt{N}}.
\end{equation}
Relative fluctuations vanish as $1/\sqrt{N}$, justifying equivalence of canonical and microcanonical ensembles in the thermodynamic limit.
\end{example}

\begin{theorem}[Canonical ensemble thermodynamics recipe]
Given $Z(T, V, N)$:
\begin{alignat}{3}
    &\text{(1)}\quad U = -\pdv{\log Z}{\beta}, \qquad
    &&\text{(2)}\quad P = -\left.\pdv{F}{V}\right|_T, \qquad
    &&\text{(3)}\quad F = -k_B T \log Z,\\[4pt]
    &\text{(4)}\quad S = \frac{U}{T} + k_B \log Z, \qquad
    &&\text{(5)}\quad C_V = k_B\beta^2\pdv[2]{\log Z}{\beta}, \qquad
    &&\text{(6)}\quad \mathrm{Var}(U) = k_B T^2 C_V.
\end{alignat}
\end{theorem}

\subsection*{Equipartition as a corollary}

For any degree of freedom $\xi$ entering the Hamiltonian as $H = a\xi^2 + \cdots$:
\begin{equation}
    Z \propto \int_{-\infty}^{\infty} d\xi\;e^{-\beta a\xi^2} = \sqrt{\frac{\pi}{\beta a}},
    \qquad
    U_\xi = -\pdv{}{\beta}\!\left[-\tfrac{1}{2}\log(\beta a)\right] = \frac{1}{2}k_BT.
\end{equation}

\begin{fact}
\textbf{Equipartition theorem.} Each quadratic degree of freedom contributes $\tfrac{1}{2}k_BT$ to the average energy, independent of $a$.
\end{fact}

\section{Two-Level System and the Schottky Anomaly}

\begin{example}
A spin-$\tfrac{1}{2}$ particle in field $B$ has levels $E_{1,2} = \mp\mu_B B$. The partition function, internal energy, and entropy are:
\begin{equation}
    Z = 2\cosh(\beta\mu_B B), \qquad
    U = -\mu_B B \tanh(\beta\mu_B B),
\end{equation}
\begin{equation}
    S = k_B\log\!\left(2\cosh(\beta\mu_B B)\right) - \frac{\mu_B B}{T}\tanh(\beta\mu_B B).
\end{equation}
As $T \to 0$: $S \to 0$ (third law). As $T \to \infty$: $S \to k_B\log 2$ (two equally accessible states).

\subsection*{Heat capacity and the Schottky anomaly}

\begin{equation}
    C_V = \frac{(\mu_B B)^2}{k_B T^2}\,\mathrm{sech}^2(\beta\mu_B B).
\end{equation}

The \textbf{Schottky anomaly}: $C_V$ peaks at $k_BT \sim \mu_B B$, falls exponentially at low $T$ (excited state frozen out) and as $T^{-2}$ at high $T$ (states equally occupied). A universal signature of finite level spacing.

The level probabilities:
\begin{equation}
    P_1 = \frac{1}{1 + e^{-2\beta\mu_B B}}, \qquad
    P_2 = \frac{1}{1 + e^{2\beta\mu_B B}},
\end{equation}
are related to the Fermi-Dirac function.
\end{example}

\section{The Ideal Gas from the Partition Function}

\subsection*{Single-particle partition function}

For $N$ monatomic particles in volume $V$, the Hamiltonian factorises, so $Z_N = (Z_1)^N$ (distinguishable). The spatial integral gives $V$; the momentum integral:
\begin{equation}
    Z_1 = \int \frac{d^3x\,d^3p}{(2\pi\hbar)^3}\;e^{-\beta p^2/2m}, \qquad
    \int d^3p\;e^{-\beta p^2/2m} = (2\pi mk_BT)^{3/2}.
\end{equation}

\begin{formula}[Single-particle ideal gas partition function]
\begin{equation}
    Z_1 = \frac{V}{\lambda_{\text{th}}^3} \propto V T^{3/2}, \qquad
    \lambda_{\text{th}} = \frac{h}{\sqrt{2\pi mk_BT}}.
\end{equation}
\end{formula}

\subsection*{$N$-particle partition function and the Gibbs factor}

Treating particles as distinguishable gives entropy $\propto N\log V$ (non-extensive) — the \textbf{Gibbs paradox}. Identical particles are indistinguishable; dividing by $N!$ fixes extensivity:

\begin{formula}[Ideal gas partition function for indistinguishable particles]
\begin{equation}
    Z_N = \frac{(Z_1)^N}{N!} = \frac{1}{N!}\left(\frac{V}{\lambda_{\text{th}}^3}\right)^N.
    \label{eq:ideal_gas_ZN}
\end{equation}
Valid in the classical limit where typical single-state occupancy $\ll 1$.
\end{formula}

\subsection*{Recovery of the Sackur--Tetrode entropy}

From $F = -k_BT\log Z_N$, $S = -(\partial F/\partial T)_V$, with Stirling's approximation:

\begin{formula}[Sackur--Tetrode equation]
\begin{equation}
    S = k_BN\!\left[\log\frac{V}{N\lambda_{\text{th}}^3} + \frac{5}{2}\right].
    \label{eq:sackur_tetrode}
\end{equation}
\end{formula}

Properly extensive. The internal energy $U = \frac{3}{2}Nk_BT$ follows from $-\partial\log Z_N/\partial\beta$.

\section{The Maxwell--Boltzmann Distribution}

The single-particle phase-space density is $\rho \propto e^{-\beta p^2/2m}$.

\subsection*{Energy distribution}

Inserting a delta function to select energy $E$:
\begin{equation}
    P(E) = \frac{V}{Z_1}\cdot\frac{4\pi}{(2\pi\hbar)^3}
            \int_0^\infty p^2\,e^{-\beta p^2/2m}\;\delta\!\left(E - \frac{p^2}{2m}\right)dp.
\end{equation}

\begin{formula}[Maxwell--Boltzmann energy distribution]
\begin{equation}
    P(E) = 2\pi\!\left(\frac{1}{\pi k_BT}\right)^{3/2}\sqrt{E}\;e^{-\beta E}.
    \label{eq:MB_energy}
\end{equation}
\end{formula}

Peaks at $E = \tfrac{1}{2}k_BT$; mean $\langle E\rangle = \tfrac{3}{2}k_BT$.

\subsection*{Speed distribution}

Change variables $E = \tfrac{1}{2}mv^2$, conserving probability ($dE = mv\,dv$):

\begin{formula}[Maxwell--Boltzmann speed distribution]
\begin{equation}
    P(v) = 4\pi\!\left(\frac{m}{2\pi k_BT}\right)^{3/2} v^2\,e^{-mv^2/2k_BT}.
    \label{eq:MB_speed}
\end{equation}
\end{formula}

\begin{equation}
    \langle v \rangle = \sqrt{\frac{8k_BT}{\pi m}}, \qquad
    v_{\text{rms}} = \sqrt{\frac{3k_BT}{m}}.
\end{equation}
The $v^2$ factor (phase-space volume) competes with the Boltzmann suppression to produce the asymmetric peak.

\section{The Einstein Solid and Quantum Oscillators}

Model a monatomic solid as $N$ independent 3D harmonic oscillators of frequency $\omega$ (distinguishable, $Z_N = (Z_1)^N$).

\subsection*{Single-oscillator partition function}

With levels $\varepsilon = \hbar\omega(n_1+n_2+n_3+\tfrac{3}{2})$, the partition function factorises:
\begin{align}
    Z_1 &= e^{-3\beta\hbar\omega/2}
           \left(\frac{1}{1 - e^{-\beta\hbar\omega}}\right)^3, \qquad
    Z_N = \left[\frac{e^{-\beta\hbar\omega/2}}{1 - e^{-\beta\hbar\omega}}\right]^{3N}.
\end{align}

\subsection*{Internal energy and the Planck factor}

From $U_N = -\partial\log Z_N/\partial\beta$:

\begin{formula}[Einstein solid internal energy]
\begin{equation}
    U_N = 3N\hbar\omega\!\left[\frac{1}{2} + \bar{n}\right],
    \qquad \bar{n} = \frac{1}{e^{\beta\hbar\omega} - 1}.
    \label{eq:einstein_U}
\end{equation}
\end{formula}

Zero-point energy $\tfrac{3}{2}N\hbar\omega$; thermal excitation $3N\hbar\omega\,\bar{n}$.

\begin{example}
\textbf{Low-$T$:} $\bar{n} \approx e^{-\beta\hbar\omega} \to 0$; all modes frozen out:
\begin{equation}
    C_V \approx 3Nk_B\!\left(\frac{\hbar\omega}{k_BT}\right)^2 e^{-\hbar\omega/k_BT} \to 0.
\end{equation}

\textbf{High-$T$:} $\bar{n} \approx 1/\beta\hbar\omega$, giving $U_N \approx 3Nk_BT$ (classical equipartition).
\end{example}

\subsection*{Heat capacity}

\begin{equation}
    C_V = 3Nk_B\!\left(\frac{\hbar\omega}{k_BT}\right)^2
    \frac{e^{\beta\hbar\omega}}{\left(e^{\beta\hbar\omega} - 1\right)^2}.
\end{equation}

\begin{fact}
\textbf{Dulong--Petit Law.} At high $T$, $C_V \to 3Nk_B = 3R$ per mole.
\end{fact}

\section{Final Example: Adiabatic Cooling}

\begin{example}
$N$ classical indistinguishable ultrarelativistic particles ($E = c|\vec{p}|$) in a linear trap $V(\vec{x}) = \alpha|\vec{x}|$. Using $\int d^3u\, e^{-a|\vec{u}|} = 8\pi/a^3$:
\begin{equation}
    Z_1 = \frac{1}{(2\pi\hbar)^3}(8\pi)^2
    \left(\frac{k_BT}{c}\right)^3\!\left(\frac{k_BT}{\alpha}\right)^3
    = \frac{8}{\pi}\!\left(\frac{k_BT}{\hbar}\right)^6 \frac{1}{(c\alpha)^3}.
\end{equation}

With $Z_N = Z_1^N/N!$:
\begin{align}
    U_N &= 6Nk_BT \quad\text{(6 effective dof per particle)},\\
    S_N &= 6Nk_B + k_B\!\left[N\log Z_1 - N\log N + N\right].
\end{align}

\paragraph{Adiabatic cooling.} At fixed $S_N$, since $Z_1 \propto T^6/\alpha^3$, we need $Z_1 = \text{const}$:
\begin{equation}
    \frac{T^6}{\alpha^3} = \text{const}
    \quad\Longrightarrow\quad \boxed{T \sim \sqrt{\alpha}}.
\end{equation}
Reducing $\alpha$ adiabatically cools the gas. The same $T \propto \alpha^{m/n}$ scaling holds for any $Z_1 \propto T^n/\alpha^m$.
\end{example}
